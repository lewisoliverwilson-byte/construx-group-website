## Email template — React Email

```tsx
// emails/order-confirmation.tsx

export default function OrderConfirmation({ name, orderId, amount, items }: OrderConfirmationProps) {
  return (
    <Html>
      <Head />

      <Body style={{ backgroundColor: '#f6f9fc', fontFamily: 'system-ui, sans-serif' }}>
        <Container style={{ maxWidth: '560px', margin: '40px auto', backgroundColor: '#fff', borderRadius: '8px', padding: '40px' }}>
          <Img src="https://construxgroup.io/logo.png" width={120} height={32} alt="Construx" />

          <Heading style={{ fontSize: '24px', fontWeight: '700', marginTop: '32px' }}>
            Order confirmed
          </Heading>

          <Text style={{ color: '#6b7280' }}>Hi {name},</Text>
          <Text style={{ color: '#374151' }}>
            Your order <strong>{orderId}</strong> has been confirmed. Here's a summary:
          </Text>

          <Section style={{ background: '#f9fafb', borderRadius: '6px', padding: '16px', marginBottom: '24px' }}>
            {items.map((item, i) => (
              <div key={i} style={{ display: 'flex', justifyContent: 'space-between', marginBottom: '8px' }}>
                <Text style={{ margin: 0, color: '#374151' }}>{item.name} × {item.qty}</Text>
                <Text style={{ margin: 0, color: '#374151' }}>£{(item.price / 100).toFixed(2)}</Text>

            ))}

            <div style={{ display: 'flex', justifyContent: 'space-between' }}>
              <Text style={{ margin: 0, fontWeight: '600' }}>Total</Text>
              <Text style={{ margin: 0, fontWeight: '600' }}>£{(amount / 100).toFixed(2)}</Text>
            </div>
          </Section>

          <Button
            href={`https://construxgroup.io/orders/${orderId}`}
            style={{ background: '#F97316', color: '#fff', borderRadius: '6px', padding: '12px 24px', fontWeight: '600' }}

            View order
          </Button>

          <Hr style={{ borderColor: '#e5e7eb', margin: '32px 0' }} />
          <Text style={{ fontSize: '12px', color: '#9ca3af' }}>
            Construx Group · <Link href="https://construxgroup.io/unsubscribe">Unsubscribe</Link>
          </Text>

    </Html>

}
```

## Send with Resend SDK

```ts
// lib/email.ts
import { Resend }              from 'resend';
import { render }              from '@react-email/render';
import OrderConfirmation       from '@/emails/order-confirmation';

const resend = new Resend(process.env.RESEND_API_KEY);

export async function sendOrderConfirmation(
  to:    string,
  props: Parameters<typeof OrderConfirmation>[0],
) {
  const html = await render(<OrderConfirmation {...props} />);
  const text = await render(<OrderConfirmation {...props} />, { plainText: true });

  const { data, error } = await resend.emails.send({
    from:    'orders@construxgroup.io',
    to,
    subject: `Order ${props.orderId} confirmed`,
    html,
    text,
    tags: [
      { name: 'category', value: 'order-confirmation' },
      { name: 'order_id', value: props.orderId },
    ],
  });

  if (error) throw new Error(`Resend error: ${error.message}`);
  return data!.id;
}
```

## Webhooks — delivery events

```ts
// api/webhooks/resend/route.ts
import { Webhook } from 'svix';

const wh = new Webhook(process.env.RESEND_WEBHOOK_SECRET!);

export async function POST(req: Request) {
  const payload = await req.text();
  const headers = Object.fromEntries(req.headers);

  const event = wh.verify(payload, headers) as ResendWebhookEvent;

  switch (event.type) {
    case 'email.delivered':
      await db.emails.update({ where: { id: event.data.email_id }, data: { deliveredAt: new Date() } });
      break;
    case 'email.bounced':
      await markEmailBounced(event.data.to[0]);
      break;
    case 'email.complained':
      await unsubscribeEmail(event.data.to[0]);
      break;
  }

  return new Response('ok');
}
```

Resend vs SendGrid: SendGrid has more enterprise features (dedicated IPs, ISP reputation management, marketing email). Resend is developer-first — better DX, React Email integration, cleaner API, reasonable pricing for transactional volume. The Construx default: Resend for all product transactional email; revisit SendGrid only if deliverability becomes an issue at scale.
