Local development should mirror production. Docker Compose makes that easy without committing to Kubernetes.

## compose.yaml

```yaml
name: construx

services:
  postgres:
    image: postgres:16-alpine
    restart: unless-stopped
    environment:
      POSTGRES_DB:       construx
      POSTGRES_USER:     construx
      POSTGRES_PASSWORD: localdev
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD", "pg_isready", "-U", "construx"]
      interval: 5s
      timeout:  5s
      retries:  5

  redis:
    image: redis:7-alpine
    restart: unless-stopped
    ports:
      - "6379:6379"
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout:  5s
      retries:  5

  app:
    build:
      context: .
      dockerfile: Dockerfile.dev
    restart: unless-stopped
    ports:
      - "3000:3000"
    volumes:
      - .:/app
      - /app/node_modules  # prevent host node_modules from overwriting container's
      - /app/.next
    environment:
      DATABASE_URL: postgresql://construx:localdev@postgres:5432/construx
      REDIS_URL:    redis://redis:6379
      NODE_ENV:     development
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy

  mailpit:
    image: axllent/mailpit
    ports:
      - "1025:1025"   # SMTP
      - "8025:8025"   # Web UI
    environment:
      MP_SMTP_AUTH_ACCEPT_ANY:  1
      MP_SMTP_AUTH_ALLOW_INSECURE: 1

volumes:
  postgres_data:
```

## Dockerfile.dev

```dockerfile
FROM node:22-alpine

WORKDIR /app

# Install dependencies layer — cached unless package files change
COPY package.json package-lock.json ./
RUN npm install

# Source code is mounted as a volume — not copied
EXPOSE 3000
CMD ["npm", "run", "dev"]
```

## Useful commands

```bash
# Start all services (detached)
docker compose up -d

# View logs for all services
docker compose logs -f

# View logs for a specific service
docker compose logs -f app

# Restart a specific service after config changes
docker compose restart app

# Run a command in a service
docker compose exec postgres psql -U construx

# Connect to Redis
docker compose exec redis redis-cli

# Stop all services
docker compose down

# Stop and delete volumes (reset database)
docker compose down -v
```

## .env file

```bash
# .env.local (not committed — git ignored)
DATABASE_URL=postgresql://construx:localdev@localhost:5432/construx
REDIS_URL=redis://localhost:6379
NEXT_PUBLIC_APP_URL=http://localhost:3000
STRIPE_SECRET_KEY=sk_test_...

# Mailpit SMTP (local email testing)
SMTP_HOST=localhost
SMTP_PORT=1025
SMTP_FROM=noreply@construxgroup.local
```

## The workflow

```bash
git clone git@github.com:construx/website.git && cd website
cp .env.example .env.local
# fill in STRIPE_SECRET_KEY and other real keys

docker compose up -d
npm run db:migrate
npm run db:seed

open http://localhost:3000
open http://localhost:8025  # Mailpit UI for email testing
```

First checkout to running app in under 2 minutes. No manual postgres installation, no Redis configuration, no "which port is postgres running on?".
