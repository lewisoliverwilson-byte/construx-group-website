Workers sit at the network edge before your origin server. Every request passes through a Worker if you deploy one. This means you can: authenticate requests before they hit origin, rewrite URLs, A/B test at the CDN layer, serve personalised content with zero origin load, and rate-limit or block before the request costs you anything.

## Hello World

```ts
// src/index.ts
export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);

    if (url.pathname === '/health') {
      return Response.json({ status: 'ok', region: request.cf?.colo });
    }

    return new Response('Hello from the edge', {
      headers: { 'Content-Type': 'text/plain' },
    });
  },
} satisfies ExportedHandler<Env>;
```

## KV: global key-value store

```ts
interface Env {
  CACHE: KVNamespace;
}

export default {
  async fetch(request: Request, env: Env) {
    const key = new URL(request.url).pathname;

    // Try cache first
    const cached = await env.CACHE.get(key);
    if (cached) {
      return new Response(cached, {
        headers: { 'X-Cache': 'HIT', 'Content-Type': 'application/json' },
      });
    }

    // Fetch from origin
    const res  = await fetch(`https://api.construxgroup.io${key}`);
    const body = await res.text();

    // Cache for 60 seconds
    await env.CACHE.put(key, body, { expirationTtl: 60 });

    return new Response(body, {
      headers: { 'X-Cache': 'MISS', 'Content-Type': 'application/json' },
    });
  },
} satisfies ExportedHandler<Env>;
```

KV replicates globally. Reads are fast from any PoP; writes propagate within ~60 seconds. Good for caching, feature flags, and rate-limit counters.

## Durable Objects: stateful edge

```ts
// Durable Object: a single-threaded stateful process
export class RateLimiter {
  private state:    DurableObjectState;
  private requests: number = 0;
  private windowStart: number = Date.now();

  constructor(state: DurableObjectState) {
    this.state = state;
  }

  async fetch(request: Request): Promise<Response> {
    const now = Date.now();

    if (now - this.windowStart > 60_000) {
      this.requests   = 0;
      this.windowStart = now;
    }

    this.requests++;

    if (this.requests > 100) {
      return new Response('Rate limited', { status: 429 });
    }

    return new Response(JSON.stringify({ remaining: 100 - this.requests }));
  }
}
```

```ts
// Worker: route to a Durable Object by IP
interface Env {
  RATE_LIMITER: DurableObjectNamespace;
}

export default {
  async fetch(request: Request, env: Env) {
    const ip = request.headers.get('cf-connecting-ip') ?? 'unknown';
    const id  = env.RATE_LIMITER.idFromName(ip);
    const obj = env.RATE_LIMITER.get(id);
    return obj.fetch(request);
  },
} satisfies ExportedHandler<Env>;
```

Durable Objects give you consistent state at the edge — a single authoritative instance per key, colocated with the first request.

## R2: object storage

```ts
interface Env {
  ASSETS: R2Bucket;
}

export default {
  async fetch(request: Request, env: Env) {
    const key = new URL(request.url).pathname.slice(1);  // strip leading /

    if (request.method === 'PUT') {
      await env.ASSETS.put(key, request.body, {
        httpMetadata: { contentType: request.headers.get('Content-Type') ?? '' },
      });
      return new Response('Uploaded', { status: 201 });
    }

    const object = await env.ASSETS.get(key);
    if (!object) return new Response('Not found', { status: 404 });

    return new Response(object.body, {
      headers: {
        'Content-Type':  object.httpMetadata?.contentType ?? 'application/octet-stream',
        'Cache-Control': 'public, max-age=31536000, immutable',
        'ETag':          object.httpEtag,
      },
    });
  },
} satisfies ExportedHandler<Env>;
```

R2 is S3-compatible with no egress fees. Serve assets directly from Workers without hitting your origin.

## Middleware pattern

```ts
type Handler = (req: Request, env: Env) => Promise<Response>;

function withAuth(next: Handler): Handler {
  return async (req, env) => {
    const token = req.headers.get('authorization')?.replace('Bearer ', '');
    if (!token || token !== env.API_TOKEN) {
      return new Response('Unauthorized', { status: 401 });
    }
    return next(req, env);
  };
}

function withCors(next: Handler): Handler {
  return async (req, env) => {
    if (req.method === 'OPTIONS') {
      return new Response(null, {
        headers: {
          'Access-Control-Allow-Origin':  '*',
          'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE',
          'Access-Control-Allow-Headers': 'Content-Type, Authorization',
        },
      });
    }
    const res = await next(req, env);
    res.headers.set('Access-Control-Allow-Origin', '*');
    return res;
  };
}

const handler = withCors(withAuth(async (req, env) => {
  return Response.json({ data: 'protected' });
}));
```
