Server-Sent Events (SSE) are a native browser API for receiving a stream of updates from the server over a single long-lived HTTP connection. Unlike WebSockets, SSE is one-directional (server to client only), which makes it dramatically simpler to implement and scale.

For most real-time use cases — live activity feeds, AI streaming responses, progress updates, notifications — SSE is the right tool. WebSockets add bidirectional complexity you usually don't need.

## The protocol

SSE uses plain HTTP with `Content-Type: text/event-stream`. The server sends newline-delimited events:

```
data: {"type":"update","value":42}\n\n
event: status\ndata: {"status":"complete"}\n\n
```

Each event ends with a blank line. The browser's `EventSource` API handles parsing, reconnection, and event routing automatically.

## Next.js Route Handler

```ts
// app/api/stream/route.ts
export async function GET(req: Request) {
  const encoder = new TextEncoder();

  const stream = new ReadableStream({
    async start(controller) {
      const send = (data: unknown, event?: string) => {
        const lines = event ? `event: ${event}\ndata: ${JSON.stringify(data)}\n\n` : `data: ${JSON.stringify(data)}\n\n`;
        controller.enqueue(encoder.encode(lines));
      };

      // Send initial state
      send({ type: 'connected', ts: Date.now() });

      // Stream events
      for (let i = 0; i < 10; i++) {
        await new Promise((r) => setTimeout(r, 1000));
        send({ type: 'tick', count: i });
      }

      send({ type: 'done' }, 'close');
      controller.close();
    },
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      Connection: 'keep-alive',
    },
  });
}
```

## Client-side EventSource

```ts
// hooks/useStream.ts
import { useEffect, useRef, useState } from 'react';

type StreamEvent<T> = { type: string; data: T };

export function useStream<T>(url: string) {
  const [events, setEvents] = useState<StreamEvent<T>[]>([]);
  const [status, setStatus] = useState<'connecting' | 'open' | 'closed'>('connecting');
  const sourceRef = useRef<EventSource | null>(null);

  useEffect(() => {
    const source = new EventSource(url);
    sourceRef.current = source;

    source.onopen = () => setStatus('open');
    source.onerror = () => setStatus('closed');

    source.onmessage = (e) => {
      const data = JSON.parse(e.data) as T;
      setEvents((prev) => [...prev, { type: 'message', data }]);
    };

    // Named event listeners
    source.addEventListener('close', () => {
      setStatus('closed');
      source.close();
    });

    return () => source.close();
  }, [url]);

  return { events, status };
}
```

## AI streaming responses

The most common SSE use case is streaming LLM output token by token:

```ts
// app/api/chat/route.ts
import Anthropic from '@anthropic-ai/sdk';

const client = new Anthropic();

export async function POST(req: Request) {
  const { messages } = await req.json();
  const encoder = new TextEncoder();

  const stream = new ReadableStream({
    async start(controller) {
      const send = (chunk: string) => {
        controller.enqueue(encoder.encode(`data: ${JSON.stringify({ chunk })}\n\n`));
      };

      const response = await client.messages.create({
        model: 'claude-opus-4-8',
        max_tokens: 1024,
        messages,
        stream: true,
      });

      for await (const event of response) {
        if (event.type === 'content_block_delta' && event.delta.type === 'text_delta') {
          send(event.delta.text);
        }
      }

      controller.enqueue(encoder.encode(`data: ${JSON.stringify({ done: true })}\n\n`));
      controller.close();
    },
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
    },
  });
}
```

Client-side hook for AI streaming:

```ts
async function streamChat(messages: Message[], onChunk: (text: string) => void) {
  const res = await fetch('/api/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ messages }),
  });

  const reader = res.body!.getReader();
  const decoder = new TextDecoder();

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    const text = decoder.decode(value);
    const lines = text.split('\n').filter((l) => l.startsWith('data: '));

    for (const line of lines) {
      const data = JSON.parse(line.slice(6));
      if (data.chunk) onChunk(data.chunk);
      if (data.done) return;
    }
  }
}
```

## Authentication

`EventSource` doesn't support custom headers, so pass auth via query params (short-lived token) or cookies:

```ts
// Generate a short-lived stream token
const streamToken = await createStreamToken(userId, { expiresIn: '1m' });
const source = new EventSource(`/api/stream?token=${streamToken}`);

// Validate in the route handler
export async function GET(req: Request) {
  const token = new URL(req.url).searchParams.get('token');
  const userId = await validateStreamToken(token);
  if (!userId) return new Response('Unauthorized', { status: 401 });
  // ...
}
```

## When to use WebSockets instead

Use WebSockets when you need bidirectional communication: collaborative editing, multiplayer games, chat where the client also sends messages frequently. For everything read-heavy — dashboards, feeds, AI responses, progress bars — SSE is simpler and scales better on serverless infrastructure.
