Validate at the edge. Trust everywhere inside.

## Schema basics

```ts
import { z } from 'zod';

// Primitive schemas
const name   = z.string().min(1).max(100).trim();
const age    = z.number().int().min(0).max(150);
const email  = z.string().email().toLowerCase();
const url    = z.string().url();
const uuid   = z.string().uuid();
const date   = z.coerce.date();   // coerces strings and numbers to Date

// Object schema
const UserSchema = z.object({
  id:        z.string().uuid(),
  name:      z.string().min(1).max(100),
  email:     z.string().email(),
  role:      z.enum(['admin', 'user', 'guest']),
  createdAt: z.coerce.date(),
  settings:  z.object({
    theme:         z.enum(['light', 'dark']).default('dark'),
    notifications: z.boolean().default(true),
  }).optional(),
});

// Infer TypeScript type from schema
type User = z.infer<typeof UserSchema>;
// {
//   id: string; name: string; email: string;
//   role: 'admin' | 'user' | 'guest'; createdAt: Date;
//   settings?: { theme: 'light' | 'dark'; notifications: boolean }
// }
```

## Parse and safeParse

```ts
// parse — throws ZodError on failure
const user = UserSchema.parse(rawData);
// user is fully typed: User

// safeParse — returns { success, data } | { success: false, error }
const result = UserSchema.safeParse(rawData);
if (!result.success) {
  console.error(result.error.issues);
  // [{ code: 'invalid_type', path: ['email'], message: 'Invalid email' }]
  return;
}
const user = result.data;  // typed: User
```

## Transform — parse and shape in one step

```ts
const CreateOrderSchema = z.object({
  items: z.array(z.object({
    productId: z.string().uuid(),
    quantity:  z.number().int().min(1).max(99),
    priceP:    z.number().int().min(0),  // price in pence
  })).min(1),
  coupon:   z.string().toUpperCase().optional(),
  currency: z.enum(['GBP', 'USD', 'EUR']).default('GBP'),
}).transform((data) => ({
  ...data,
  totalPence: data.items.reduce((s, i) => s + i.priceP * i.quantity, 0),
  itemCount:  data.items.length,
}));

type CreateOrder = z.infer<typeof CreateOrderSchema>;
// totalPence and itemCount are computed on parse — always consistent
```

## Refine — custom validation

```ts
const PasswordSchema = z.string()
  .min(12, 'Password must be at least 12 characters')
  .refine((pw) => /[A-Z]/.test(pw), { message: 'Must contain uppercase' })
  .refine((pw) => /[0-9]/.test(pw), { message: 'Must contain a number' })
  .refine((pw) => /[^a-zA-Z0-9]/.test(pw), { message: 'Must contain a symbol' });

// Cross-field validation
const SignupSchema = z.object({
  password:        PasswordSchema,
  confirmPassword: z.string(),
}).refine((data) => data.password === data.confirmPassword, {
  message: 'Passwords do not match',
  path:    ['confirmPassword'],
});
```

## Environment variables — validate at startup

```ts
// env.ts — parse process.env at startup, fail fast if invalid
const EnvSchema = z.object({
  DATABASE_URL:     z.string().url(),
  RESEND_API_KEY:   z.string().min(1),
  POSTHOG_API_KEY:  z.string().startsWith('phc_'),
  JWT_SECRET:       z.string().min(32),
  NODE_ENV:         z.enum(['development', 'production', 'test']).default('development'),
  PORT:             z.coerce.number().int().default(3000),
  NEXT_PUBLIC_URL:  z.string().url().optional(),
});

export const env = EnvSchema.parse(process.env);
// env.PORT is number (coerced from string env var)
// env.NODE_ENV is 'development' | 'production' | 'test'
```

## API route validation

```ts
// app/api/orders/route.ts
import { z } from 'zod';

const CreateOrderBodySchema = z.object({
  items:  z.array(z.object({ productId: z.string().uuid(), qty: z.number().int().min(1) })).min(1),
  coupon: z.string().optional(),
});

export async function POST(req: Request) {
  const body   = await req.json();
  const result = CreateOrderBodySchema.safeParse(body);

  if (!result.success) {
    return Response.json({ errors: result.error.flatten().fieldErrors }, { status: 422 });
  }

  const { items, coupon } = result.data;  // typed and validated
  const order = await createOrder({ items, coupon, userId: 'usr_123' });
  return Response.json(order, { status: 201 });
}
```

Zod vs Yup vs io-ts: Yup has a similar API but weaker TypeScript inference. io-ts has stronger guarantees but much more verbose syntax. Zod hits the pragmatic middle: excellent TypeScript integration, readable API, fast enough for request validation. The Construx default: Zod for all API input validation, env var parsing, and external API response parsing.
