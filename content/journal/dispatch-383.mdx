Feature flags let you ship code without activating it. The flag is the release — not the deploy. This lets you:
- Deploy unfinished features (behind a flag) without breaking production
- Do percentage rollouts: 1% → 10% → 100% without re-deploying
- Target specific users, orgs, or cohorts
- Kill-switch problematic features instantly

## The simplest approach: database-backed flags

For most teams early on, a table in your database is all you need:

```ts
// db/schema.ts
export const featureFlags = pgTable('feature_flags', {
  id: text('id').primaryKey(),  // 'new-editor', 'redesigned-checkout'
  enabled: boolean('enabled').notNull().default(false),
  rolloutPercent: integer('rollout_percent').notNull().default(0),  // 0-100
  allowlist: text('allowlist').array().notNull().default(sql`'{}'`),  // user IDs
  updatedAt: timestamp('updated_at').notNull().defaultNow(),
});
```

```ts
// lib/flags.ts
import { cache } from 'react';

// Cached per-request — doesn't hit the DB on every component render
export const getFlags = cache(async () => {
  const flags = await db.query.featureFlags.findMany();
  return Object.fromEntries(flags.map((f) => [f.id, f]));
});

export async function isEnabled(flagId: string, userId?: string): Promise<boolean> {
  const flags = await getFlags();
  const flag = flags[flagId];

  if (!flag) return false;
  if (!flag.enabled) return false;

  // Allowlist check (specific users always get the feature)
  if (userId && flag.allowlist.includes(userId)) return true;

  // Percentage rollout via stable hash
  if (userId && flag.rolloutPercent > 0 && flag.rolloutPercent < 100) {
    const hash = await stableHash(`${flagId}:${userId}`);
    return hash % 100 < flag.rolloutPercent;
  }

  return flag.rolloutPercent === 100;
}

async function stableHash(input: string): Promise<number> {
  const encoded = new TextEncoder().encode(input);
  const hashBuffer = await crypto.subtle.digest('SHA-256', encoded);
  const hashArray = new Uint8Array(hashBuffer);
  return hashArray[0] + hashArray[1] * 256;
}
```

```ts
// Cache flags in Redis to avoid DB hits on every request
export async function getFlagsCached() {
  const cached = await redis.get<Record<string, FeatureFlag>>('feature-flags');
  if (cached) return cached;

  const flags = await db.query.featureFlags.findMany();
  const flagMap = Object.fromEntries(flags.map((f) => [f.id, f]));

  await redis.setex('feature-flags', 60, JSON.stringify(flagMap));  // 60s TTL
  return flagMap;
}

// After updating a flag in admin, invalidate the cache
export async function setFlag(id: string, updates: Partial<FeatureFlag>) {
  await db.update(featureFlags).set({ ...updates, updatedAt: new Date() })
    .where(eq(featureFlags.id, id));
  await redis.del('feature-flags');
}
```

## Using flags in Next.js

```tsx
// Server component — evaluate on the server
export default async function EditorPage() {
  const session = await auth();
  const newEditorEnabled = await isEnabled('new-editor', session?.user?.id);

  return newEditorEnabled ? <NewEditor /> : <LegacyEditor />;
}
```

```ts
// Middleware — edge-level flag evaluation for routing
export async function middleware(req: NextRequest) {
  const userId = req.cookies.get('user-id')?.value;
  const flags = await getFlagsFromKV();  // Vercel KV or Cloudflare KV

  if (flags['redesigned-checkout']?.enabled && isInRollout(flags['redesigned-checkout'], userId)) {
    return NextResponse.rewrite(new URL('/checkout-v2', req.url));
  }
}
```

## LaunchDarkly: managed flags for complex targeting

When you need user-attribute targeting, A/B test analytics, or compliance audit trails:

```bash
npm install @launchdarkly/node-server-sdk
```

```ts
// lib/launchdarkly.ts
import * as ld from '@launchdarkly/node-server-sdk';

let ldClient: ld.LDClient;

export async function getLdClient() {
  if (!ldClient) {
    ldClient = ld.init(process.env.LAUNCHDARKLY_SDK_KEY!);
    await ldClient.waitForInitialization({ timeout: 5 });
  }
  return ldClient;
}
```

```ts
// Evaluate a flag with user context
export async function evaluate<T>(
  flagKey: string,
  context: { userId: string; email?: string; plan?: string },
  defaultValue: T,
): Promise<T> {
  const client = await getLdClient();

  const ldContext: ld.LDContext = {
    kind: 'user',
    key: context.userId,
    email: context.email,
    custom: { plan: context.plan },
  };

  return client.variation(flagKey, ldContext, defaultValue);
}
```

```tsx
// Usage in a Server Component
export default async function PricingPage() {
  const session = await auth();

  const showAnnualDefault = await evaluate(
    'pricing-annual-default',
    { userId: session?.user?.id ?? 'anonymous', plan: session?.user?.plan },
    false,
  );

  return <Pricing defaultBilling={showAnnualDefault ? 'annual' : 'monthly'} />;
}
```

LaunchDarkly handles: percentage rollouts, user attribute targeting (plan, company size, geography), A/B test assignment, flag change audit trails.

## Unleash: self-hosted alternative

Unleash is open-source and self-hostable — same concept, no vendor lock-in:

```bash
docker run -p 4242:4242 unleashorg/unleash-server
npm install unleash-client
```

```ts
// lib/unleash.ts
import { initialize, isEnabled } from 'unleash-client';

initialize({
  url: process.env.UNLEASH_URL!,
  appName: 'construx-app',
  customHeaders: { Authorization: process.env.UNLEASH_API_TOKEN! },
});

export { isEnabled };
```

## Flag lifecycle management

Flags accumulate. Old flags that are 100% enabled or permanently disabled become dead code that confuses future developers. Enforce cleanup:

```ts
// db/schema.ts — add a planned removal date to every flag
export const featureFlags = pgTable('feature_flags', {
  // ...
  deprecatedAfter: date('deprecated_after'),  // when to clean up this flag
  jira: text('jira'),  // ticket to track cleanup
});
```

```yaml
# .github/workflows/flag-audit.yml
# Run weekly — flag any flags past their deprecation date
- run: npx tsx scripts/check-stale-flags.ts
```

The rule: every flag gets a `deprecatedAfter` date when it's created. When a flag reaches 100% rollout, schedule the code cleanup.

## When to use each

- **Database-backed**: early stage, small team, simple enable/disable
- **LaunchDarkly**: complex targeting, A/B tests, compliance requirements, large team
- **Unleash**: open-source preference, self-hosted, no vendor dependency
