One framework. Runs everywhere. Type-safe end to end.

## Basic server

```ts
import { Hono } from 'hono';

const app = new Hono();

app.get('/', (c) => c.json({ status: 'ok', version: '1.0.0' }));

app.get('/users/:id', async (c) => {
  const id   = c.req.param('id');
  const user = await db.users.findUnique({ where: { id } });
  if (!user) return c.json({ error: 'Not found' }, 404);
  return c.json(user);
});

app.post('/users', async (c) => {
  const body = await c.req.json<{ email: string; name: string }>();
  const user = await db.users.create({ data: body });
  return c.json(user, 201);
});

export default app;
```

## Middleware

```ts
import { Hono }         from 'hono';
import { cors }         from 'hono/cors';
import { jwt }          from 'hono/jwt';
import { rateLimiter }  from 'hono-rate-limiter';
import { logger }       from 'hono/logger';
import { secureHeaders } from 'hono/secure-headers';

const app = new Hono();

app.use('*', logger());
app.use('*', secureHeaders());
app.use('/api/*', cors({ origin: ['https://construxgroup.io'] }));

// Rate limit: 100 req/min per IP
app.use('/api/*', rateLimiter({
  windowMs: 60_000,
  limit:    100,
  keyGenerator: (c) => c.req.header('CF-Connecting-IP') ?? 'unknown',
}));

// JWT protected routes
app.use('/api/v1/*', jwt({ secret: process.env.JWT_SECRET! }));

app.get('/api/v1/me', (c) => {
  const payload = c.get('jwtPayload');
  return c.json({ userId: payload.sub });
});
```

## Zod validator

```ts
import { zValidator } from '@hono/zod-validator';
import { z }          from 'zod';

const createUserSchema = z.object({
  email: z.string().email(),
  name:  z.string().min(1).max(100),
  role:  z.enum(['admin', 'user']).default('user'),
});

app.post(
  '/api/users',
  zValidator('json', createUserSchema),
  async (c) => {
    const body = c.req.valid('json');  // Fully typed
    const user = await db.users.create({ data: body });
    return c.json(user, 201);
  },
);
```

## RPC — type-safe client from routes

```ts
// server/routes/users.ts
import { Hono } from 'hono';

const users = new Hono()
  .get('/:id', async (c) => {
    const user = await getUser(c.req.param('id'));
    return c.json(user);
  })
  .post('/', zValidator('json', createUserSchema), async (c) => {
    const body = c.req.valid('json');
    return c.json(await createUser(body), 201);
  });

export type UsersRouter = typeof users;
export default users;
```

```ts
// client/api.ts — generated type-safe client
import { hc } from 'hono/client';
import type { UsersRouter } from '../server/routes/users';

const client = hc<UsersRouter>('https://api.construxgroup.io');

// Fully typed — IDE autocomplete on routes, params, and response
const user = await client.users[':id'].$get({ param: { id: '123' } });
const data = await user.json();  // data is typed as User
```

## Cloudflare Workers

```ts
// src/index.ts
import { Hono }  from 'hono';
import { cache } from 'hono/cache';

const app = new Hono<{ Bindings: Env }>();

app.use('/static/*', cache({ cacheName: 'static-assets', cacheControl: 'max-age=86400' }));

app.get('/api/config', async (c) => {
  // Access Cloudflare bindings directly
  const value = await c.env.KV.get('config');
  return c.json(JSON.parse(value ?? '{}'));
});

app.post('/api/queue', async (c) => {
  const body = await c.req.json();
  await c.env.QUEUE.send(body);
  return c.json({ queued: true });
});

export default app;
```

## Testing

```ts
import { describe, it, expect } from 'bun:test';
import app from './index';

describe('users API', () => {
  it('GET /users/:id — returns user', async () => {
    const res  = await app.request('/users/123');
    const body = await res.json();

    expect(res.status).toBe(200);
    expect(body).toMatchObject({ id: '123' });
  });

  it('POST /users — validates email', async () => {
    const res = await app.request('/users', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({ email: 'not-an-email', name: 'Test' }),
    });

    expect(res.status).toBe(400);
  });
});
```

Hono vs Express vs Fastify: Express has the largest ecosystem but runs only in Node.js and is not typed. Fastify is faster than Express and has schemas, but still Node.js-only. Hono runs everywhere, is as fast as Fastify, and has first-class TypeScript. The tradeoff: Hono's ecosystem is smaller than Express. The Construx default: Hono for all new backend services; Express only for existing services we're not migrating.
