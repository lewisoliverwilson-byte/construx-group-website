The naive file upload flow: user selects file → browser POST to your server → server streams to S3. You pay to transfer the file through your server, your server handles the load, and uploads are limited by your server's bandwidth.

The correct flow: user requests an upload URL from your server → server generates a presigned URL → browser uploads directly to S3 → browser confirms completion to your server. Your server is involved for access control (deciding *who* can upload) and post-upload processing, but never handles the bytes.

## Generating presigned PUT URLs with AWS SDK

```bash
npm install @aws-sdk/client-s3 @aws-sdk/s3-request-presigner
```

```ts
// lib/storage.ts
import { S3Client, PutObjectCommand, GetObjectCommand } from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';

const s3 = new S3Client({
  region: process.env.AWS_REGION!,
  credentials: {
    accessKeyId: process.env.AWS_ACCESS_KEY_ID!,
    secretAccessKey: process.env.AWS_SECRET_ACCESS_KEY!,
  },
});

const BUCKET = process.env.S3_BUCKET_NAME!;

export async function getUploadUrl(key: string, contentType: string, maxSizeBytes = 10_485_760) {
  const command = new PutObjectCommand({
    Bucket: BUCKET,
    Key: key,
    ContentType: contentType,
    ContentLength: maxSizeBytes,  // enforced by S3 — upload fails if file is larger
  });

  const url = await getSignedUrl(s3, command, { expiresIn: 300 });  // 5 min expiry
  return { url, key };
}

export async function getDownloadUrl(key: string) {
  const command = new GetObjectCommand({ Bucket: BUCKET, Key: key });
  return getSignedUrl(s3, command, { expiresIn: 3600 });  // 1 hour
}
```

## Route handler to vend upload URLs

```ts
// app/api/uploads/route.ts
import { NextRequest } from 'next/server';
import { auth } from '@/auth';
import { getUploadUrl } from '@/lib/storage';
import crypto from 'crypto';

const ALLOWED_TYPES = ['image/jpeg', 'image/png', 'image/webp', 'application/pdf'];
const MAX_SIZE = 10 * 1024 * 1024;  // 10MB

export async function POST(req: NextRequest) {
  const session = await auth();
  if (!session?.user) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const { contentType, fileName } = await req.json();

  if (!ALLOWED_TYPES.includes(contentType)) {
    return Response.json({ error: 'File type not allowed' }, { status: 400 });
  }

  // Namespaced key — user can only upload to their own prefix
  const key = `users/${session.user.id}/${crypto.randomUUID()}-${fileName}`;
  const { url } = await getUploadUrl(key, contentType, MAX_SIZE);

  return Response.json({ url, key });
}
```

## Client-side upload flow

```tsx
'use client';

import { useState } from 'react';

export function FileUpload({ onComplete }: { onComplete: (key: string) => void }) {
  const [progress, setProgress] = useState(0);
  const [uploading, setUploading] = useState(false);

  const handleFile = async (file: File) => {
    setUploading(true);

    // 1. Get presigned URL from our API
    const { url, key } = await fetch('/api/uploads', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ contentType: file.type, fileName: file.name }),
    }).then((r) => r.json());

    // 2. Upload directly to S3 with progress tracking
    await new Promise<void>((resolve, reject) => {
      const xhr = new XMLHttpRequest();

      xhr.upload.addEventListener('progress', (e) => {
        if (e.lengthComputable) {
          setProgress(Math.round((e.loaded / e.total) * 100));
        }
      });

      xhr.addEventListener('load', () => {
        if (xhr.status === 200) resolve();
        else reject(new Error(`Upload failed: ${xhr.status}`));
      });

      xhr.addEventListener('error', reject);

      xhr.open('PUT', url);
      xhr.setRequestHeader('Content-Type', file.type);
      xhr.send(file);
    });

    // 3. Tell your server what was uploaded
    onComplete(key);
    setUploading(false);
  };

  return (
    <div>
      <input
        type="file"
        disabled={uploading}
        onChange={(e) => e.target.files?.[0] && handleFile(e.target.files[0])}
      />
      {uploading && <progress value={progress} max={100}>{progress}%</progress>}
    </div>
  );
}
```

`XMLHttpRequest` gives you upload progress events — `fetch` doesn't support request streaming progress.

## Cloudflare R2 (S3-compatible, no egress cost)

R2 is S3-compatible with zero egress fees — reads don't cost per-GB. Good if your objects are frequently downloaded.

```ts
// lib/storage.ts — R2 variant
const r2 = new S3Client({
  region: 'auto',
  endpoint: `https://${process.env.CF_ACCOUNT_ID}.r2.cloudflarestorage.com`,
  credentials: {
    accessKeyId: process.env.R2_ACCESS_KEY_ID!,
    secretAccessKey: process.env.R2_SECRET_ACCESS_KEY!,
  },
});
```

The SDK call is identical — swap the client config, keep the rest.

## Post-upload processing

After upload, trigger background processing:

```ts
// In the route that receives the completion notification
export async function PUT(req: NextRequest) {
  const session = await auth();
  const { key } = await req.json();

  // Verify the key belongs to this user
  if (!key.startsWith(`users/${session?.user?.id}/`)) {
    return Response.json({ error: 'Forbidden' }, { status: 403 });
  }

  // Record the upload
  const upload = await db.insert(uploads)
    .values({ userId: session.user.id, key, status: 'pending' })
    .returning()
    .then(([u]) => u);

  // Enqueue processing (resize, virus scan, text extraction, etc.)
  await enqueueJob('process-upload', { uploadId: upload.id, key });

  return Response.json({ uploadId: upload.id });
}
```

The job queue handles the rest: resize images, run virus scanning, extract text from PDFs. The client gets an immediate response; processing happens asynchronously.

## CORS configuration for S3

Direct browser uploads require CORS configured on the bucket:

```json
[
  {
    "AllowedHeaders": ["Content-Type"],
    "AllowedMethods": ["PUT"],
    "AllowedOrigins": ["https://construxgroup.io", "http://localhost:3000"],
    "ExposeHeaders": [],
    "MaxAgeSeconds": 3000
  }
]
```

Without this, browsers block the direct PUT request with a CORS error.
