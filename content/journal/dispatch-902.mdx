It's particularly well-suited for end-user search interfaces: typo-tolerant by default (2 typos allowed for words over 8 characters), with highlighting, faceting, and geosearch built in. Single binary, REST API, 50ms queries, typos handled. That's the pitch.

## Indexing documents

```bash
# Start Meilisearch
meilisearch --master-key="your-master-key"

# Base URL: http://localhost:7700

# Create an index and add documents
curl -X POST 'http://localhost:7700/indexes/journal/documents' \
  -H 'Content-Type: application/json' \
  -H 'Authorization: Bearer your-master-key' \
  --data '[
    {
      "id": "dispatch-881",
      "title": "gRPC: Protocol Buffers, service definitions, and streaming RPCs",
      "excerpt": "gRPC is Google open-source RPC framework built on HTTP/2 and Protocol Buffers...",
      "tag": "Build Log",
      "date": "2032-03-10",
      "readingTime": 6
    },
    {
      "id": "dispatch-882",
      "title": "ZFS: copy-on-write filesystem with snapshots and send/receive",
      "excerpt": "ZFS is a combined filesystem and volume manager with copy-on-write semantics...",
      "tag": "Build Log",
      "date": "2032-03-13",
      "readingTime": 5
    }
  ]'

# Meilisearch processes indexing asynchronously — returns a task ID
# {
#   "taskUid": 1,
#   "indexUid": "journal",
#   "status": "enqueued"
# }

# Check task status
curl 'http://localhost:7700/tasks/1' -H 'Authorization: Bearer your-master-key'
# { "status": "succeeded", "details": { "receivedDocuments": 2, "indexedDocuments": 2 } }

# Check index stats
curl 'http://localhost:7700/indexes/journal/stats' | jq .
# { "numberOfDocuments": 901, "isIndexing": false }
```

## Search API — queries and filtering

```bash
# Basic search — typo tolerance active by default
curl -X POST 'http://localhost:7700/indexes/journal/search' \
  -H 'Content-Type: application/json' \
  --data '{
    "q": "ebpf linux networking",
    "limit": 5,
    "attributesToHighlight": ["title", "excerpt"],
    "highlightPreTag": "<em>",
    "highlightPostTag": "</em>"
  }'
# Results:
# dispatch-861: "Cilium: eBPF-native networking..." (_rankingScore: 0.9921)
# dispatch-837: "eBPF: kernel tracing..."         (_rankingScore: 0.8814)

# Filter by tag (faceted search)
curl -X POST 'http://localhost:7700/indexes/journal/search' \
  --data '{
    "q": "database",
    "filter": "tag = \"Build Log\"",
    "sort": ["date:desc"],
    "limit": 10
  }'

# Faceted search — get counts per tag
curl -X POST 'http://localhost:7700/indexes/journal/search' \
  --data '{
    "q": "",
    "facets": ["tag"],
    "limit": 0
  }'
# { "facetDistribution": { "tag": { "Build Log": 882, "Strategy": 11, "Product": 5 } } }

# Typo tolerance in action:
curl -X POST 'http://localhost:7700/indexes/journal/search' \
  --data '{"q": "postgress replicaiton"}'
# Still finds: "PostgreSQL streaming replication..." (2 typos tolerated)
```

## Index configuration — ranking and searchable attributes

```bash
# Configure which fields are searchable and their priority
curl -X PATCH 'http://localhost:7700/indexes/journal/settings' \
  --data '{
    "searchableAttributes": ["title", "excerpt", "tag"],
    "displayedAttributes": ["id", "title", "excerpt", "tag", "date", "readingTime"],
    "filterableAttributes": ["tag", "date", "readingTime"],
    "sortableAttributes": ["date", "readingTime"],

    "rankingRules": [
      "words",        // Match more words first
      "typo",         // Fewer typos = higher rank
      "proximity",    // Closer together = higher rank
      "attribute",    // Matches in title > excerpt
      "sort",         // User-specified sort
      "exactness"     // Exact match > prefix match
    ],

    "typoTolerance": {
      "enabled": true,
      "minWordSizeForTypos": {
        "oneTypo": 5,    // Allow 1 typo for words 5+ chars
        "twoTypos": 9    // Allow 2 typos for words 9+ chars
      }
    },

    "synonyms": {
      "k8s": ["kubernetes"],
      "pg": ["postgres", "postgresql"],
      "js": ["javascript", "node"]
    }
  }'
```

## Integrating with Next.js

```typescript
// lib/search.ts — server-side Meilisearch client
import { MeiliSearch } from 'meilisearch'

const client = new MeiliSearch({
  host: process.env.MEILISEARCH_URL!,
  apiKey: process.env.MEILISEARCH_SEARCH_KEY!,   // Read-only search key (safe to expose)
})

export interface SearchResult {
  id: string
  title: string
  excerpt: string
  tag: string
  date: string
  _rankingScore: number
}

export async function searchJournal(query: string, tag?: string): Promise<SearchResult[]> {
  const index = client.index('journal')
  const result = await index.search<SearchResult>(query, {
    limit: 20,
    filter: tag ? `tag = "${tag}"` : undefined,
    attributesToHighlight: ['title', 'excerpt'],
    highlightPreTag: '<mark>',
    highlightPostTag: '</mark>',
  })
  return result.hits
}

// app/api/search/route.ts — search endpoint
export async function GET(request: Request) {
  const { searchParams } = new URL(request.url)
  const q = searchParams.get('q') ?? ''
  const tag = searchParams.get('tag') ?? undefined

  if (q.length < 2) return Response.json([])

  const hits = await searchJournal(q, tag)
  return Response.json(hits)
}
```

Meilisearch vs Elasticsearch: Elasticsearch is the industry standard for large-scale search and analytics — it handles petabytes, complex aggregations, vector search, and runs distributed across many nodes. Meilisearch is optimised for simplicity and end-user search — typo tolerance, fast indexing, and a clean REST API in a single binary that runs on a $10 VPS. The tradeoff is scale: Meilisearch is single-node (no cluster mode beyond read replicas in v1). For a blog with 1,000 posts or a product catalog with 100,000 items, Meilisearch is dramatically simpler to operate. For log analytics at billions of events, Elasticsearch (or ClickHouse) is the right answer.
