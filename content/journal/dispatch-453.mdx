## Sending from a route handler

```ts
// app/api/auth/register/route.ts
import { resend } from '@/lib/email';
import { WelcomeEmail } from '@/emails/WelcomeEmail';

export async function POST(req: Request) {
  const { email, name } = await req.json();

  // Create user in database...
  const confirmToken = generateToken();
  const confirmUrl = `${process.env.NEXT_PUBLIC_SITE_URL}/auth/confirm?token=${confirmToken}`;

  const { data, error } = await resend.emails.send({
    from: 'Construx Group <hello@construxgroup.io>',
    to: email,
    subject: 'Confirm your email',
    react: WelcomeEmail({ name, confirmUrl }),
  });

  if (error) {
    console.error('[email] send failed:', error);
    // Don't fail the registration — email can be resent
  }

  return Response.json({ ok: true });
}
```

## Webhook handling for delivery events

```ts
// app/api/webhooks/resend/route.ts
import { resend } from '@/lib/email';

export async function POST(req: Request) {
  const payload = await req.text();
  const headers = Object.fromEntries(req.headers.entries());

  // Verify signature
  const webhookSecret = process.env.RESEND_WEBHOOK_SECRET;
  if (!webhookSecret) return new Response('Missing webhook secret', { status: 500 });

  try {
    const svix = await import('svix');
    const wh = new svix.Webhook(webhookSecret);
    const event = wh.verify(payload, headers) as { type: string; data: Record<string, unknown> };

    switch (event.type) {
      case 'email.delivered':
        console.log('[email] delivered to', event.data.to);
        break;

      case 'email.bounced':
        // Mark email address as invalid in database
        await markEmailBounced(event.data.to as string);
        break;

      case 'email.complained':
        // Remove from marketing lists
        await removeFromMarketingLists(event.data.to as string);
        break;

      case 'email.opened':
        // Track open rate
        break;
    }

    return Response.json({ ok: true });
  } catch (err) {
    return new Response('Invalid signature', { status: 400 });
  }
}
```

## Batch sending

```ts
// Send to multiple recipients without triggering spam filters
export async function sendBatch(emails: Array<{ to: string; name: string }>) {
  // Resend supports batch sending — up to 100 emails per call
  const batch = emails.map(({ to, name }) => ({
    from: 'Construx Group <hello@construxgroup.io>',
    to,
    subject: 'Your weekly scan summary',
    react: ScanSummaryEmail({ name }),
  }));

  const { data, error } = await resend.batch.send(batch);

  if (error) {
    console.error('[email] batch failed:', error);
    return { sent: 0, error };
  }

  return { sent: data?.data.length ?? 0 };
}
```

## Domain setup

In the Resend dashboard, verify your sending domain with three DNS records:
- SPF record — proves you're authorised to send from the domain
- DKIM record — signs emails with a private key Resend controls
- DMARC record — tells receivers what to do with failing messages

Without these, email goes to spam regardless of content quality. Set them up before sending anything.
