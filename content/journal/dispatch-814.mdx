Built in. Jest-compatible. No setup.

## Basic test

```ts
// orders.test.ts
import { describe, test, expect, beforeEach, afterEach } from 'bun:test';
import { createOrder, getOrder, cancelOrder }            from './orders';
import { db }                                            from '../db/client';

describe('orders', () => {
  beforeEach(async () => {
    await db.execute(sql`TRUNCATE orders CASCADE`);
  });

  test('createOrder inserts and returns the order', async () => {
    const order = await createOrder({
      userId: 'usr_test',
      items:  [{ productId: 'prod_abc', qty: 2 }],
    });

    expect(order.id).toMatch(/^ord_/);
    expect(order.status).toBe('pending');
    expect(order.items).toHaveLength(1);
  });

  test('getOrder throws NOT_FOUND for unknown id', async () => {
    await expect(getOrder('ord_nonexistent')).rejects.toThrow('NOT_FOUND');
  });

  test('cancelOrder transitions to cancelled', async () => {
    const order    = await createOrder({ userId: 'usr_test', items: [] });
    const cancelled = await cancelOrder(order.id);
    expect(cancelled.status).toBe('cancelled');
  });
});
```

## Mocking — `mock` and `spyOn`

```ts
import { test, expect, mock, spyOn } from 'bun:test';
import { sendConfirmationEmail }     from './email';
import * as resend                   from '../lib/resend';

test('sendConfirmationEmail calls resend with correct payload', async () => {
  const spy = spyOn(resend, 'send').mockResolvedValue({ id: 'email_123' });

  await sendConfirmationEmail({ to: 'user@example.com', orderId: 'ord_abc' });

  expect(spy).toHaveBeenCalledWith({
    from:    'orders@construx.io',
    to:      'user@example.com',
    subject: expect.stringContaining('ord_abc'),
    html:    expect.any(String),
  });
});

// Module mock
const mockDb = mock(() => ({
  orders: {
    findMany: mock(() => Promise.resolve([])),
    create:   mock(() => Promise.resolve({ id: 'ord_mock' })),
  },
}));

mock.module('../db/client', () => ({ db: mockDb() }));
```

## Snapshot testing

```ts
import { test, expect } from 'bun:test';
import { renderOrderSummary } from './render';

test('order summary snapshot', () => {
  const html = renderOrderSummary({
    orderId:   'ord_abc',
    amount:    4999,
    items:     [{ name: 'Pro Plan', qty: 1, price: 4999 }],
    createdAt: new Date('2031-08-21T12:00:00Z'),
  });

  expect(html).toMatchSnapshot();
});
```

## Running tests

```bash
# Run all tests
bun test

# Watch mode — re-runs affected tests on change
bun test --watch

# Filter by name
bun test --test-name-pattern 'createOrder'

# Filter by file
bun test orders.test.ts

# Coverage (built-in, no istanbul needed)
bun test --coverage

# Bail on first failure
bun test --bail 1

# CI — parallel, no watch, with coverage
bun test --coverage --reporter=junit > test-results.xml
```

## bun.test.toml — optional config

```toml
[test]
timeout = 10000  # 10s per test
coverage = true
coverageThreshold = { line = 80, function = 80 }
```

## Lifecycle hooks

```ts
import { beforeAll, afterAll, beforeEach, afterEach } from 'bun:test';

// Global setup/teardown — runs once per test file
beforeAll(async () => {
  await db.$connect();
});

afterAll(async () => {
  await db.$disconnect();
});

// Per-test isolation
beforeEach(async () => {
  await db.execute(sql`BEGIN`);
});

afterEach(async () => {
  await db.execute(sql`ROLLBACK`);
});
```

Bun test vs Vitest: Vitest has better ecosystem integration (supports jsdom, Vite plugins, browser mode, Storybook), richer plugin API, and more configuration options. Bun test is faster and has zero setup for Bun-native projects. The Construx default: Bun test for server-side and API code in Bun projects; Vitest for Next.js projects where Vite integration matters.
