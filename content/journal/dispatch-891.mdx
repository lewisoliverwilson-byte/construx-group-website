date: "2032-04-09"
excerpt: "Vault is a secrets management platform that does more than store API keys. Its killer feature is dynamic credentials: instead of issuing a long-lived database password, Vault generates a short-lived credential on demand (e.g., a PostgreSQL user that expires in one hour), then revokes it automatically. The application never stores a secret that can be stolen — it requests one at runtime. Vault also provides a PKI secrets engine (issue X.509 certificates on demand), transit encryption (encrypt/decrypt data without the application holding keys), and OIDC for authentication."
author: "Lewis Wilson"

Long-lived secrets are the problem. Dynamic credentials are the answer.

## Vault setup and authentication methods

```bash
# Start Vault in dev mode (in-memory, auto-unsealed — for testing only)
vault server -dev -dev-root-token-id="root"
export VAULT_ADDR='http://127.0.0.1:8200'
export VAULT_TOKEN='root'

# Production: HA cluster with Raft storage
# /etc/vault.d/vault.hcl:
# ui = true
# storage "raft" {
#   path    = "/opt/vault/data"
#   node_id = "vault-01"
# }
# listener "tcp" {
#   address       = "0.0.0.0:8200"
#   tls_cert_file = "/etc/vault.d/tls.crt"
#   tls_key_file  = "/etc/vault.d/tls.key"
# }
# api_addr = "https://vault.construx.internal:8200"
# cluster_addr = "https://vault-01.construx.internal:8201"

# Enable Kubernetes auth — applications authenticate using their K8s service account
vault auth enable kubernetes
vault write auth/kubernetes/config \
  kubernetes_host="https://kubernetes.default.svc" \
  kubernetes_ca_cert=@/var/run/secrets/kubernetes.io/serviceaccount/ca.crt

# Create a role: map Kubernetes service accounts to Vault policies
vault write auth/kubernetes/role/construx-api \
  bound_service_account_names=construx-api-sa \
  bound_service_account_namespaces=construx \
  policies=construx-api \
  ttl=1h                     # Token valid for 1 hour, then re-authenticate
```

## Dynamic PostgreSQL credentials

```bash
# Enable the database secrets engine
vault secrets enable database

# Configure a PostgreSQL connection (Vault manages the master credentials)
vault write database/config/construx-postgres \
  plugin_name=postgresql-database-plugin \
  allowed_roles="construx-api,construx-worker" \
  connection_url="postgresql://{{username}}:{{password}}@postgres.construx.internal:5432/construx" \
  username="vault_admin" \
  password="$(cat /etc/vault/pg-admin-password)"

# Define a role: SQL statement Vault runs to create credentials
vault write database/roles/construx-api \
  db_name=construx-postgres \
  creation_statements="
    CREATE ROLE \"{{name}}\" WITH LOGIN PASSWORD '{{password}}' VALID UNTIL '{{expiration}}';
    GRANT SELECT, INSERT, UPDATE ON ALL TABLES IN SCHEMA public TO \"{{name}}\";
    GRANT USAGE ON ALL SEQUENCES IN SCHEMA public TO \"{{name}}\";
  " \
  default_ttl="1h" \
  max_ttl="4h"

# Generate a credential on demand
vault read database/creds/construx-api
# Key                Value
# lease_id           database/creds/construx-api/a3f9b2c1-...
# lease_duration     1h
# username           v-kubernetes-construx-QkA9f8
# password           A1B2-c3D4-e5F6-g7H8

# After 1 hour, Vault automatically runs: DROP ROLE "v-kubernetes-construx-QkA9f8"
```

## PKI secrets engine — on-demand TLS certificates

```bash
# Enable PKI secrets engine for internal TLS
vault secrets enable pki
vault secrets tune -max-lease-ttl=87600h pki   # 10 year root CA

# Generate internal root CA
vault write pki/root/generate/internal \
  common_name="Construx Internal CA" \
  ttl=87600h

# Enable intermediate CA (best practice: don't issue leaf certs from root)
vault secrets enable -path=pki_int pki
vault secrets tune -max-lease-ttl=43800h pki_int  # 5 year intermediate

# Generate CSR and sign with root
vault write -format=json pki_int/intermediate/generate/internal \
  common_name="Construx Intermediate CA" | \
  jq -r '.data.csr' > intermediate.csr

vault write -format=json pki/root/sign-intermediate \
  csr=@intermediate.csr format=pem_bundle ttl=43800h | \
  jq -r '.data.certificate' > intermediate_signed.pem

vault write pki_int/intermediate/set-signed certificate=@intermediate_signed.pem

# Define a role for issuing leaf certs
vault write pki_int/roles/construx-internal \
  allowed_domains="construx.internal" \
  allow_subdomains=true \
  max_ttl=720h     # 30-day leaf certs

# Issue a certificate (application calls this at startup)
vault write pki_int/issue/construx-internal \
  common_name="api.construx.internal" \
  alt_names="api-01.construx.internal,api-02.construx.internal" \
  ttl=720h
# Returns: certificate, private_key, issuing_ca, expiration
```

## Vault Agent — sidecar for automatic secret injection

```yaml
# Vault Agent as a Kubernetes init container — injects secrets into the pod
# k8s deployment annotation approach:

annotations:
  vault.hashicorp.com/agent-inject: "true"
  vault.hashicorp.com/role: "construx-api"

  # Inject PostgreSQL credentials into a file
  vault.hashicorp.com/agent-inject-secret-db: "database/creds/construx-api"
  vault.hashicorp.com/agent-inject-template-db: |
    {{- with secret "database/creds/construx-api" -}}
    DATABASE_URL=postgres://{{ .Data.username }}:{{ .Data.password }}@postgres.construx.internal:5432/construx
    {{- end }}

  # Inject API keys from KV store
  vault.hashicorp.com/agent-inject-secret-config: "secret/data/construx-api/production"
  vault.hashicorp.com/agent-inject-template-config: |
    {{- with secret "secret/data/construx-api/production" -}}
    STRIPE_API_KEY={{ .Data.data.stripe_api_key }}
    SENDGRID_API_KEY={{ .Data.data.sendgrid_api_key }}
    {{- end }}

# Files are written to /vault/secrets/ inside the container
# Vault Agent refreshes credentials before they expire (no app restart needed)
```

Vault vs Kubernetes Secrets: K8s Secrets are base64-encoded strings stored in etcd — they're not encrypted by default (unless you configure encryption at rest), anyone with kubectl can read them, and there's no audit log of who accessed what. Vault encrypts everything, logs every access with a full audit trail, and the dynamic credentials model means stolen credentials have a short TTL. The operational overhead is real — Vault is another service to run, HA, unseal, and monitor. But for any production system handling customer data or payment credentials, the audit trail and dynamic credentials are worth it.
