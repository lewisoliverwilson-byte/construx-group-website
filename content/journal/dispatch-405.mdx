Simple components take props and render output. Complex UI components — tabs, select menus, accordions — have internal state and multiple sub-parts. The challenge is giving the user control over composition without exposing every internal detail.

## Compound components

A compound component is a set of related components that share implicit state through context:

```tsx
// components/ui/Tabs.tsx
import { createContext, useContext, useState } from 'react';

type TabsContext = {
  activeTab: string;
  setActiveTab: (id: string) => void;
};

const TabsContext = createContext<TabsContext | null>(null);

function useTabs() {
  const ctx = useContext(TabsContext);
  if (!ctx) throw new Error('Tab components must be used inside <Tabs>');
  return ctx;
}

function Tabs({
  children,
  defaultTab,
}: {
  children: React.ReactNode;
  defaultTab: string;
}) {
  const [activeTab, setActiveTab] = useState(defaultTab);
  return (
    <TabsContext.Provider value={{ activeTab, setActiveTab }}>
      <div>{children}</div>
    </TabsContext.Provider>
  );
}

function TabList({ children }: { children: React.ReactNode }) {
  return (
    <div role="tablist" className="flex gap-1">
      {children}
    </div>
  );
}

function Tab({ id, children }: { id: string; children: React.ReactNode }) {
  const { activeTab, setActiveTab } = useTabs();
  const isActive = activeTab === id;
  return (
    <button
      role="tab"
      aria-selected={isActive}
      onClick={() => setActiveTab(id)}
      className={`px-4 py-2 font-mono text-xs uppercase tracking-wider transition-colors ${
        isActive ? 'text-construx border-b-2 border-construx' : 'text-text-dim hover:text-text-muted'
      }`}
    >
      {children}
    </button>
  );
}

function TabPanel({ id, children }: { id: string; children: React.ReactNode }) {
  const { activeTab } = useTabs();
  if (activeTab !== id) return null;
  return <div role="tabpanel">{children}</div>;
}

// Attach sub-components
Tabs.List = TabList;
Tabs.Tab = Tab;
Tabs.Panel = TabPanel;

export { Tabs };
```

Usage at the call site is clean and composable:

```tsx
<Tabs defaultTab="overview">
  <Tabs.List>
    <Tabs.Tab id="overview">Overview</Tabs.Tab>
    <Tabs.Tab id="analytics">Analytics</Tabs.Tab>
    <Tabs.Tab id="settings">Settings</Tabs.Tab>
  </Tabs.List>
  <Tabs.Panel id="overview"><OverviewContent /></Tabs.Panel>
  <Tabs.Panel id="analytics"><AnalyticsContent /></Tabs.Panel>
  <Tabs.Panel id="settings"><SettingsContent /></Tabs.Panel>
</Tabs>
```

The caller controls the structure. The component controls the state.

## Render props

Render props pass the rendering responsibility to the parent via a function prop. Useful when you want to expose internal state for flexible rendering:

```tsx
// components/ui/Disclosure.tsx
import { useState } from 'react';

type DisclosureProps = {
  children: (props: {
    isOpen: boolean;
    toggle: () => void;
    open: () => void;
    close: () => void;
  }) => React.ReactNode;
  defaultOpen?: boolean;
};

export function Disclosure({ children, defaultOpen = false }: DisclosureProps) {
  const [isOpen, setIsOpen] = useState(defaultOpen);
  return (
    <>
      {children({
        isOpen,
        toggle: () => setIsOpen((v) => !v),
        open: () => setIsOpen(true),
        close: () => setIsOpen(false),
      })}
    </>
  );
}
```

```tsx
// Usage — caller decides exactly how to render based on state
<Disclosure>
  {({ isOpen, toggle }) => (
    <div>
      <button onClick={toggle} className="flex items-center gap-2">
        {isOpen ? 'Hide' : 'Show'} advanced options
        <ChevronDown className={`transition-transform ${isOpen ? 'rotate-180' : ''}`} size={14} />
      </button>
      {isOpen && (
        <div className="mt-4">
          <AdvancedOptions />
        </div>
      )}
    </div>
  )}
</Disclosure>
```

## Controlled and uncontrolled variants

Good components support both patterns:

```tsx
type SelectProps = {
  // Uncontrolled — manages its own state
  defaultValue?: string;
  // Controlled — parent manages state
  value?: string;
  onChange?: (value: string) => void;
  children: React.ReactNode;
};

function Select({ defaultValue, value, onChange, children }: SelectProps) {
  const [internalValue, setInternalValue] = useState(defaultValue ?? '');
  const isControlled = value !== undefined;
  const currentValue = isControlled ? value : internalValue;

  const handleChange = (newValue: string) => {
    if (!isControlled) setInternalValue(newValue);
    onChange?.(newValue);
  };

  return (
    <SelectContext.Provider value={{ value: currentValue, onChange: handleChange }}>
      {children}
    </SelectContext.Provider>
  );
}
```

This mirrors how HTML `<input>` works — controlled (value prop) or uncontrolled (defaultValue prop).

## The polymorphic `as` prop

When a component needs to render as different HTML elements or components:

```tsx
type ButtonProps<T extends React.ElementType = 'button'> = {
  as?: T;
  children: React.ReactNode;
  variant?: 'primary' | 'ghost';
} & React.ComponentPropsWithoutRef<T>;

function Button<T extends React.ElementType = 'button'>({
  as,
  children,
  variant = 'primary',
  className,
  ...props
}: ButtonProps<T>) {
  const Component = as ?? 'button';
  return (
    <Component
      className={`font-mono text-xs uppercase tracking-wider px-4 py-2 ${
        variant === 'primary' ? 'bg-construx text-black' : 'text-text-muted border border-border'
      } ${className ?? ''}`}
      {...props}
    >
      {children}
    </Component>
  );
}
```

```tsx
// Renders as a button
<Button>Submit</Button>

// Renders as a Next.js Link — all Link props are available and typed
<Button as={Link} href="/ventures">View ventures</Button>

// Renders as an anchor
<Button as="a" href="https://example.com" target="_blank">External</Button>
```

The pattern: compound components for complex multi-part UI, render props for flexible state exposure, polymorphic props for reusable interactive elements that need to render as different underlying elements.
