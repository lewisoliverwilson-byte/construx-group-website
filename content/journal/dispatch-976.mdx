excerpt: "Crossplane is a Kubernetes add-on that lets you provision and manage cloud infrastructure — AWS RDS, S3 buckets, GCP CloudSQL, Azure CosmosDB — using Kubernetes Custom Resource Definitions and the kubectl apply workflow. The mental model: every cloud resource is a Kubernetes resource. You define what you want in YAML, Crossplane reconciles the actual cloud state to match. This enables GitOps for infrastructure: the same ArgoCD workflow that manages your Deployments can manage your RDS instances. Crossplane Compositions let you define opinionated abstractions (a `PostgresDatabase` CRD that provisions RDS + security groups + parameter groups + backups with one manifest), hiding provider-specific details from application teams."

Cloud resources as Kubernetes CRDs. `kubectl apply` provisions RDS. GitOps for infrastructure.

## Installation and provider setup

```bash
# Install Crossplane via Helm
helm repo add crossplane-stable https://charts.crossplane.io/stable
helm install crossplane crossplane-stable/crossplane \
  --namespace crossplane-system \
  --create-namespace \
  --version 1.18.0

# Verify installation
kubectl get pods -n crossplane-system
# NAME                                       READY   STATUS    AGE
# crossplane-7d4b89c9f-xk2m9                1/1     Running   2m
# crossplane-rbac-manager-5f8c4b7d9-n8r4p   1/1     Running   2m

# Install the AWS provider
kubectl apply -f - <<EOF
apiVersion: pkg.crossplane.io/v1
kind: Provider
metadata:
  name: upbound-provider-aws
spec:
  package: xpkg.upbound.io/upbound/provider-aws-rds:v1.14.0
  packagePullPolicy: IfNotPresent
  revisionActivationPolicy: Automatic
EOF

# Wait for provider to be healthy
kubectl get providers
# NAME                     INSTALLED   HEALTHY   PACKAGE
# upbound-provider-aws     True        True      upbound/provider-aws-rds:v1.14.0

# Configure provider credentials (using IRSA or static credentials)
kubectl apply -f - <<EOF
apiVersion: aws.upbound.io/v1beta1
kind: ProviderConfig
metadata:
  name: construx-aws
spec:
  credentials:
    source: IRSA   # use IAM Roles for Service Accounts
EOF
```

## Provisioning AWS resources

```yaml
# Provision an RDS PostgreSQL instance directly
apiVersion: rds.aws.upbound.io/v1beta1
kind: Instance
metadata:
  name: construx-prod-postgres
  namespace: construx-prod
spec:
  forProvider:
    region: eu-west-1
    engine: postgres
    engineVersion: "16.3"
    instanceClass: db.t4g.medium
    allocatedStorage: 100
    storageType: gp3
    iops: 3000
    dbName: construx_prod
    username: construx
    multiAz: true
    backupRetentionPeriod: 7
    deletionProtection: true
    storageEncrypted: true
    kmsKeyIdSelector:
      matchLabels:
        app: construx-prod
    vpcSecurityGroupIdSelector:
      matchLabels:
        role: database
    dbSubnetGroupNameSelector:
      matchLabels:
        env: production

    # Crossplane manages password via a Secret
    passwordSecretRef:
      name: construx-db-password
      namespace: construx-prod
      key: password

  # Where to write the connection details (endpoint, port, etc.)
  writeConnectionSecretsToRef:
    name: construx-db-connection
    namespace: construx-prod

  # What provider config to use for credentials
  providerConfigRef:
    name: construx-aws
```

```bash
# Apply the RDS instance manifest
kubectl apply -f rds-instance.yaml
# instance.rds.aws.upbound.io/construx-prod-postgres created

# Watch provisioning progress
kubectl get instance construx-prod-postgres -w
# NAME                       READY   SYNCED   EXTERNAL-NAME          AGE
# construx-prod-postgres     False   True     construx-prod-postgres  45s
# construx-prod-postgres     True    True     construx-prod-postgres  8m34s

# Read the connection details (written to a Secret by Crossplane)
kubectl get secret construx-db-connection -n construx-prod -o yaml
# data:
#   endpoint: Y29uc3RydXgtcHJvZC1...   (base64)
#   port:     NTQzMg==
#   username: Y29uc3RydXg=

# Delete the manifest → Crossplane deletes the RDS instance
# (unless deletionProtection: true — then it blocks deletion)
kubectl delete instance construx-prod-postgres
```

## Compositions — opinionated abstractions

```yaml
# CompositeResourceDefinition: define a new CRD (PostgresDatabase)
# that application teams use without knowing provider details
apiVersion: apiextensions.crossplane.io/v1
kind: CompositeResourceDefinition
metadata:
  name: xpostgresdatabases.construx.io
spec:
  group: construx.io
  names:
    kind: XPostgresDatabase
    plural: xpostgresdatabases
  claimNames:
    kind: PostgresDatabase        # what app teams actually create
    plural: postgresdatabases
  versions:
    - name: v1alpha1
      served: true
      referenceable: true
      schema:
        openAPIV3Schema:
          type: object
          properties:
            spec:
              type: object
              properties:
                env:
                  type: string
                  enum: [dev, staging, production]
                size:
                  type: string
                  enum: [small, medium, large]
                  default: medium
              required: [env]

---
# Composition: implement the XPostgresDatabase using AWS RDS
apiVersion: apiextensions.crossplane.io/v1
kind: Composition
metadata:
  name: postgresdatabase-aws
  labels:
    provider: aws
spec:
  compositeTypeRef:
    apiVersion: construx.io/v1alpha1
    kind: XPostgresDatabase

  resources:
    - name: rds-instance
      base:
        apiVersion: rds.aws.upbound.io/v1beta1
        kind: Instance
        spec:
          forProvider:
            region: eu-west-1
            engine: postgres
            engineVersion: "16.3"
            multiAz: true
            storageEncrypted: true
            backupRetentionPeriod: 7
            deletionProtection: true
      patches:
        # Map 'size' input to instanceClass
        - type: CombineFromComposite
          combine:
            variables:
              - fromFieldPath: spec.size
            strategy: string
            string:
              fmt: |
                {{- if eq . "small" -}}db.t4g.small
                {{- else if eq . "large" -}}db.r8g.xlarge
                {{- else -}}db.t4g.medium{{- end -}}
          toFieldPath: spec.forProvider.instanceClass
```

```yaml
# Application team: create a PostgresDatabase (no AWS knowledge required)
apiVersion: construx.io/v1alpha1
kind: PostgresDatabase
metadata:
  name: api-database
  namespace: construx-prod
spec:
  env: production
  size: medium
  # Crossplane Composition handles all the AWS-specific details:
  # - Instance class selection
  # - Multi-AZ configuration
  # - Security groups
  # - Subnet groups
  # - Backup policies
  # - Encryption
```

## GitOps with ArgoCD

```yaml
# ArgoCD Application: manage infrastructure with the same GitOps workflow as apps
apiVersion: argoproj.io/v1alpha1
kind: Application
metadata:
  name: construx-infrastructure
  namespace: argocd
spec:
  project: default
  source:
    repoURL: https://github.com/construx/infrastructure
    targetRevision: master
    path: crossplane/
  destination:
    server: https://kubernetes.default.svc
    namespace: crossplane-system
  syncPolicy:
    automated:
      prune: false             # don't auto-delete infrastructure resources
      selfHeal: true
    syncOptions:
      - CreateNamespace=true
      - RespectIgnoreDifferences=true
    retry:
      limit: 5
      backoff:
        duration: 5s
        factor: 2
        maxDuration: 3m
```

```bash
# Infrastructure drift detection: Crossplane continuously reconciles
# If someone manually changes an RDS parameter group via the AWS console:
# → Crossplane detects the drift within 60 seconds
# → Crossplane restores the configuration to match the manifest
# → Logs a warning event

kubectl get events -n construx-prod --field-selector reason=SyncFailed
# LAST SEEN   REASON     OBJECT                           MESSAGE
# 2m          SyncFailed  Instance/construx-prod-postgres  actual instanceClass=db.t4g.large differs from desired db.t4g.medium, reconciling

# Observe: list all managed resources (across all providers)
kubectl get managed
# NAME                                    SYNCED  READY  EXTERNAL-NAME
# construx-prod-postgres.rds.aws...       True    True   construx-prod-postgres
# construx-logs.s3.aws...                True    True   construx-logs-eu-west-1
```

Crossplane vs Terraform vs Pulumi: Terraform and Pulumi are provisioning tools — you run them to apply state, and the state file records what they manage. Crossplane is a continuous reconciler — it watches your Kubernetes resources and continuously drives cloud state to match them, which means it catches manual changes (drift) automatically. The tradeoff: Crossplane requires a running Kubernetes cluster to operate (the control plane lives in-cluster), and debugging a failed reconciliation is harder than reading a Terraform plan output. The value: GitOps for infrastructure with the same tooling (kubectl, ArgoCD) as application deployment — no separate Terraform pipelines, no state files to manage.
