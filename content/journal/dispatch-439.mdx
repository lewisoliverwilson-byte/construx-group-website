Generics solve the problem of writing code once that works for many types. Without generics, you either duplicate code for each type, or use `any` and lose type safety. Generics give you the third option: parameterize the type itself.

## Basic generics

```ts
// Without generics — only works for strings
function firstString(arr: string[]): string | undefined {
  return arr[0];
}

// With generics — works for any type, retains type information
function first<T>(arr: T[]): T | undefined {
  return arr[0];
}

const str = first(['a', 'b', 'c']);  // string | undefined
const num = first([1, 2, 3]);        // number | undefined
const post = first(posts);           // Post | undefined
```

TypeScript infers `T` from the argument — you rarely need to specify it explicitly.

## Generic functions

```ts
// Identity function
function identity<T>(value: T): T {
  return value;
}

// Map with type inference
function mapArray<T, U>(arr: T[], fn: (item: T) => U): U[] {
  return arr.map(fn);
}

const lengths = mapArray(['hello', 'world'], (s) => s.length);  // number[]
const dates = mapArray(['2028-01-01', '2028-06-15'], (s) => new Date(s));  // Date[]

// Pair — different types for each position
function pair<A, B>(a: A, b: B): [A, B] {
  return [a, b];
}

const result = pair('hello', 42);  // [string, number]
```

## Constraints

Use `extends` to constrain what types `T` can be:

```ts
// T must have a length property
function longest<T extends { length: number }>(a: T, b: T): T {
  return a.length >= b.length ? a : b;
}

longest('hello', 'hi');           // string
longest([1, 2, 3], [1, 2]);      // number[]
longest({ length: 5 }, { length: 3 });  // { length: number }
// longest(1, 2);  // Error: number doesn't have length

// T must be a key of U
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}

const user = { id: '1', name: 'Alice', age: 30 };
const name = getProperty(user, 'name');  // string
const age = getProperty(user, 'age');    // number
// getProperty(user, 'email');  // Error: not a valid key
```

## Generic interfaces and types

```ts
// Generic response wrapper
interface ApiResponse<T> {
  data: T;
  status: number;
  message: string;
}

type PostResponse = ApiResponse<Post>;
type PostListResponse = ApiResponse<Post[]>;
type PaginatedResponse<T> = ApiResponse<{
  items: T[];
  total: number;
  page: number;
}>;

// Generic repository pattern
interface Repository<T, ID = string> {
  findById(id: ID): Promise<T | null>;
  findAll(filter?: Partial<T>): Promise<T[]>;
  create(data: Omit<T, 'id' | 'createdAt' | 'updatedAt'>): Promise<T>;
  update(id: ID, data: Partial<T>): Promise<T>;
  delete(id: ID): Promise<void>;
}

class UserRepository implements Repository<User> {
  async findById(id: string): Promise<User | null> { /* ... */ }
  // ...
}
```

## Conditional types

Types that branch based on another type:

```ts
// Unwrap a Promise type
type Awaited<T> = T extends Promise<infer U> ? U : T;

type A = Awaited<Promise<string>>;   // string
type B = Awaited<number>;            // number (not a Promise, returns as-is)

// Non-nullable
type NonNullable<T> = T extends null | undefined ? never : T;

type C = NonNullable<string | null | undefined>;  // string

// Array element type
type ElementType<T> = T extends (infer U)[] ? U : never;

type D = ElementType<Post[]>;   // Post
type E = ElementType<string[]>; // string
```

## Infer

`infer` extracts a type from a pattern:

```ts
// Extract return type of a function
type ReturnType<T extends (...args: unknown[]) => unknown> =
  T extends (...args: unknown[]) => infer R ? R : never;

type F = ReturnType<() => string>;             // string
type G = ReturnType<(x: number) => boolean>;  // boolean

// Extract the resolved value from a Promise
type PromiseValue<T> = T extends Promise<infer V> ? V : never;

async function getUser(): Promise<User> { /* ... */ }
type H = PromiseValue<ReturnType<typeof getUser>>;  // User
```

## Mapped types

Transform every property of a type:

```ts
// Make all properties optional
type Partial<T> = { [K in keyof T]?: T[K] };

// Make all properties required
type Required<T> = { [K in keyof T]-?: T[K] };

// Make all properties readonly
type Readonly<T> = { readonly [K in keyof T]: T[K] };

// Pick specific keys
type Pick<T, K extends keyof T> = { [P in K]: T[P] };

// Transform values
type Stringify<T> = { [K in keyof T]: string };

// Custom: make all string properties uppercase (requires template literals)
type Capitalize<T> = {
  [K in keyof T as K extends string ? Capitalize<K> : K]: T[K]
};
```

## Generic utility functions

```ts
// Type-safe object groupBy
function groupBy<T, K extends string>(
  items: T[],
  getKey: (item: T) => K,
): Record<K, T[]> {
  const result = {} as Record<K, T[]>;
  for (const item of items) {
    const key = getKey(item);
    (result[key] ??= []).push(item);
  }
  return result;
}

const byTag = groupBy(posts, (p) => p.tag);
// Record<'Build Log' | 'Strategy' | ..., Post[]>

// Type-safe pick
function pick<T, K extends keyof T>(obj: T, keys: K[]): Pick<T, K> {
  return Object.fromEntries(keys.map((k) => [k, obj[k]])) as Pick<T, K>;
}

const minimal = pick(user, ['id', 'name']);
// { id: string; name: string }
```

Generics are the mechanism TypeScript uses for its entire standard library. Learning to read and write them is what separates writing TypeScript from just writing annotated JavaScript.
