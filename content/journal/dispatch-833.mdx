excerpt: "OAuth 2.0 is a delegation protocol. A user authorises your app to act on their behalf with a third-party service. The authorization code flow is the correct flow for server-side apps: the user logs in with the provider, your server exchanges the authorization code for tokens, and stores the refresh token. PKCE (Proof Key for Code Exchange) prevents authorization code interception attacks and is required for public clients."

The user authorises. Your server gets tokens. Never put client secrets in the browser.

## The flow in four steps

```
1. Redirect user → provider authorization endpoint
2. User authenticates and grants access
3. Provider redirects back with authorization code
4. Server exchanges code → access token + refresh token
```

## Generate PKCE verifier and challenge

```ts
// lib/pkce.ts
import { createHash } from 'crypto';

export function generateCodeVerifier(): string {
  const bytes = crypto.getRandomValues(new Uint8Array(32));
  return Buffer.from(bytes)
    .toString('base64url')
    .substring(0, 128);
}

export async function generateCodeChallenge(verifier: string): Promise<string> {
  const digest = createHash('sha256').update(verifier).digest();
  return Buffer.from(digest).toString('base64url');
}

// Generate CSRF state token
export function generateState(): string {
  return Buffer.from(crypto.getRandomValues(new Uint8Array(16))).toString('hex');
}
```

## Step 1 — build the authorization URL

```ts
// app/api/auth/login/route.ts
import { generateCodeVerifier, generateCodeChallenge, generateState } from '@/lib/pkce';

export async function GET(req: Request) {
  const verifier   = generateCodeVerifier();
  const challenge  = await generateCodeChallenge(verifier);
  const state      = generateState();

  // Store verifier and state in a short-lived cookie
  const params = new URLSearchParams({
    client_id:             process.env.GOOGLE_CLIENT_ID!,
    redirect_uri:          `${process.env.NEXT_PUBLIC_URL}/api/auth/callback`,
    response_type:         'code',
    scope:                 'openid email profile',
    state,
    code_challenge:        challenge,
    code_challenge_method: 'S256',
  });

  const response = Response.redirect(
    `https://accounts.google.com/o/oauth2/v2/auth?${params}`,
    302,
  );

  response.headers.set('Set-Cookie',
    `pkce_verifier=${verifier}; HttpOnly; Secure; SameSite=Lax; Max-Age=600; Path=/`
  );
  response.headers.append('Set-Cookie',
    `oauth_state=${state}; HttpOnly; Secure; SameSite=Lax; Max-Age=600; Path=/`
  );

  return response;
}
```

## Step 3 + 4 — handle the callback

```ts
// app/api/auth/callback/route.ts
export async function GET(req: Request) {
  const url    = new URL(req.url);
  const code   = url.searchParams.get('code');
  const state  = url.searchParams.get('state');

  const cookies     = parseCookies(req.headers.get('Cookie') ?? '');
  const verifier    = cookies.pkce_verifier;
  const savedState  = cookies.oauth_state;

  // CSRF check
  if (!code || !state || state !== savedState || !verifier) {
    return Response.redirect('/login?error=invalid_state', 302);
  }

  // Exchange code for tokens
  const tokenRes = await fetch('https://oauth2.googleapis.com/token', {
    method:  'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body:    new URLSearchParams({
      code,
      client_id:     process.env.GOOGLE_CLIENT_ID!,
      client_secret: process.env.GOOGLE_CLIENT_SECRET!,
      redirect_uri:  `${process.env.NEXT_PUBLIC_URL}/api/auth/callback`,
      grant_type:    'authorization_code',
      code_verifier: verifier,
    }),
  });

  const tokens = await tokenRes.json() as {
    access_token:  string;
    id_token:      string;
    refresh_token: string;
    expires_in:    number;
  };

  // Decode id_token (JWT) — no verification needed since it came directly from Google
  const payload = JSON.parse(
    Buffer.from(tokens.id_token.split('.')[1], 'base64url').toString()
  );

  // Upsert user in database
  const user = await db.users.upsert({
    where:  { googleId: payload.sub },
    create: { googleId: payload.sub, email: payload.email, name: payload.name },
    update: { email: payload.email, name: payload.name },
  });

  // Create a session
  const session = await createSession(user.id, tokens.refresh_token);

  return Response.redirect('/dashboard', {
    headers: { 'Set-Cookie': sessionCookie(session.id) },
  });
}
```

## Refreshing the access token

```ts
// lib/auth.ts
export async function refreshAccessToken(refreshToken: string): Promise<string> {
  const res = await fetch('https://oauth2.googleapis.com/token', {
    method:  'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body:    new URLSearchParams({
      grant_type:    'refresh_token',
      refresh_token: refreshToken,
      client_id:     process.env.GOOGLE_CLIENT_ID!,
      client_secret: process.env.GOOGLE_CLIENT_SECRET!,
    }),
  });

  if (!res.ok) throw new Error('Token refresh failed');
  const { access_token, expires_in } = await res.json();
  return access_token;
}
```

OAuth 2.0 vs JWT auth: JWT auth issues tokens your server signs and verifies — no external provider involved. OAuth 2.0 delegates authentication to a trusted provider (Google, GitHub). Use OAuth when you want to avoid managing passwords and benefit from the provider's 2FA and account recovery. Use JWT for internal service-to-service authentication where no third party is involved.
