Internal CLI tools compress workflows that would otherwise require opening three browser tabs. Scoutr's scrape-trigger script, the Construx deployment helper, the journal dispatch generator — all CLIs. They're easy to write badly and easy to write well. The pattern below is what works in production.

## Minimal setup

```bash
mkdir my-cli && cd my-cli
npm init -y
npm install commander chalk ora
npm install -D typescript tsx @types/node
```

```json
// package.json
{
  "name": "my-cli",
  "version": "1.0.0",
  "bin": {
    "my-cli": "./dist/cli.js"
  },
  "scripts": {
    "build": "tsc",
    "dev":   "tsx src/cli.ts"
  }
}
```

```json
// tsconfig.json
{
  "compilerOptions": {
    "target":         "ES2022",
    "module":         "commonjs",
    "outDir":         "./dist",
    "rootDir":        "./src",
    "strict":         true,
    "esModuleInterop": true
  }
}
```

## Commander.js structure

```ts
// src/cli.ts
import { Command } from 'commander';
import { deployCommand } from './commands/deploy';
import { statusCommand } from './commands/status';
import { logsCommand } from './commands/logs';
import { readFileSync } from 'fs';
import { join } from 'path';

const pkg = JSON.parse(
  readFileSync(join(__dirname, '../package.json'), 'utf-8')
);

const program = new Command();

program
  .name('construx')
  .description('Construx Group deployment and operations CLI')
  .version(pkg.version);

// Register subcommands
program.addCommand(deployCommand);
program.addCommand(statusCommand);
program.addCommand(logsCommand);

program.parse(process.argv);
```

## Subcommand with options

```ts
// src/commands/deploy.ts
import { Command } from 'commander';
import chalk from 'chalk';
import ora from 'ora';

export const deployCommand = new Command('deploy')
  .description('Deploy a venture to production')
  .argument('<venture>', 'Venture to deploy (scoutr, marqet, hyve)')
  .option('-e, --env <env>',     'Target environment', 'production')
  .option('-b, --branch <name>', 'Branch to deploy',   'master')
  .option('--dry-run',           'Print what would happen without deploying')
  .action(async (venture: string, options: { env: string; branch: string; dryRun?: boolean }) => {
    const VENTURES = ['scoutr', 'marqet', 'hyve'];

    if (!VENTURES.includes(venture)) {
      console.error(chalk.red(`Unknown venture: ${venture}`));
      console.error(`Available: ${VENTURES.join(', ')}`);
      process.exit(1);
    }

    console.log(
      `${chalk.cyan('→')} Deploying ${chalk.bold(venture)} to ${chalk.yellow(options.env)} from ${chalk.dim(options.branch)}`
    );

    if (options.dryRun) {
      console.log(chalk.dim('  [dry-run] No changes will be made'));
      return;
    }

    const spinner = ora('Triggering deployment...').start();

    try {
      const result = await triggerDeploy({ venture, env: options.env, branch: options.branch });
      spinner.succeed(`Deployment triggered: ${chalk.green(result.id)}`);
      console.log(`  ${chalk.dim('Track at:')} ${result.url}`);
    } catch (err) {
      spinner.fail('Deployment failed');
      console.error(chalk.red(String(err)));
      process.exit(1);
    }
  });
```

## Interactive prompts

For commands that need user input:

```ts
npm install @inquirer/prompts
```

```ts
import { input, select, confirm } from '@inquirer/prompts';

export const initCommand = new Command('init')
  .description('Initialise a new venture')
  .action(async () => {
    const name = await input({
      message: 'Venture name:',
      validate: (v) => v.length > 0 || 'Required',
    });

    const stack = await select({
      message: 'Primary stack:',
      choices: [
        { name: 'Next.js + PostgreSQL', value: 'nextjs-pg' },
        { name: 'Next.js + Turso',     value: 'nextjs-turso' },
        { name: 'Remix + PostgreSQL',  value: 'remix-pg' },
      ],
    });

    const confirmed = await confirm({
      message: `Create venture "${name}" with ${stack}?`,
    });

    if (!confirmed) {
      console.log(chalk.dim('Aborted'));
      return;
    }

    await scaffoldVenture({ name, stack });
  });
```

## Config file pattern

CLIs often need persistent config (auth tokens, default environment):

```ts
// src/config.ts
import { join } from 'path';
import { homedir } from 'os';
import { existsSync, readFileSync, writeFileSync, mkdirSync } from 'fs';

const CONFIG_DIR  = join(homedir(), '.construx');
const CONFIG_FILE = join(CONFIG_DIR, 'config.json');

interface Config {
  apiToken?: string;
  defaultEnv?: string;
  region?: string;
}

export function readConfig(): Config {
  if (!existsSync(CONFIG_FILE)) return {};
  return JSON.parse(readFileSync(CONFIG_FILE, 'utf-8'));
}

export function writeConfig(config: Config): void {
  mkdirSync(CONFIG_DIR, { recursive: true });
  writeFileSync(CONFIG_FILE, JSON.stringify(config, null, 2));
}

// Login command
export const loginCommand = new Command('login')
  .description('Authenticate with the Construx API')
  .action(async () => {
    const token = await input({ message: 'API token:' });
    writeConfig({ ...readConfig(), apiToken: token });
    console.log(chalk.green('✓ Authenticated'));
  });
```

## Publishing

```bash
# Build first
npm run build

# Add shebang to entry point output
# #!/usr/bin/env node
# — add to src/cli.ts top:
```

```ts
#!/usr/bin/env node
// src/cli.ts (first line — the shebang is passed through to dist/cli.js)
```

```bash
# Make executable
chmod +x dist/cli.js

# Test locally
npm link
my-cli --help

# Publish to npm
npm publish
```

For private internal CLIs, publish to a private npm registry or distribute as a tarball:

```bash
npm pack
# Share the .tgz — install with: npm install -g ./my-cli-1.0.0.tgz
```

## Error handling conventions

```ts
// Always exit with non-zero on error — lets shell scripts detect failures
process.exitCode = 0;

try {
  await runCommand();
} catch (err) {
  if (err instanceof CliError) {
    console.error(chalk.red(`Error: ${err.message}`));
    if (err.hint) console.error(chalk.dim(`  Hint: ${err.hint}`));
    process.exitCode = err.code ?? 1;
  } else {
    console.error(chalk.red('Unexpected error:'));
    console.error(err);
    process.exitCode = 1;
  }
}

class CliError extends Error {
  constructor(
    message: string,
    public readonly code: number = 1,
    public readonly hint?: string,
  ) {
    super(message);
  }
}
```

Setting `process.exitCode` rather than calling `process.exit()` lets any pending I/O flush (outstanding `console.log` calls, file writes) before the process terminates.
