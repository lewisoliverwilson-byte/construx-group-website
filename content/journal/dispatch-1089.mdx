excerpt: "While auto-instrumentation (Pixie's eBPF approach or the OTel Java agent) captures framework-level telemetry automatically, manual OpenTelemetry SDK instrumentation lets you create spans for specific operations, add custom attributes, record business metrics, and propagate context across async boundaries. The OTel SDK provides Tracer (for creating spans), Meter (for recording metrics — counters, histograms, gauges), and Logger (for structured log emission with automatic trace correlation). Context propagation is OTel's key mechanism: when a request enters a service, the incoming trace context (trace ID, span ID) is extracted from HTTP headers (W3C TraceContext format); when the service makes outbound calls, the current context is injected into the outgoing headers. This creates a continuous trace chain across service boundaries without any special framework support. The OTel SDK is language-neutral: Go, Java, Python, Rust, .NET, JavaScript all implement the same spec. Telemetry is exported via OTLP (gRPC or HTTP) to any compatible backend — Jaeger, Tempo, Honeycomb, Datadog, or the OTel Collector."

## Go SDK: tracing, metrics, and context propagation

```go
package main

import (
    "context"
    "log"
    "net/http"
    "time"

    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/attribute"
    "go.opentelemetry.io/otel/exporters/otlp/otlptrace/otlptracegrpc"
    "go.opentelemetry.io/otel/exporters/otlp/otlpmetric/otlpmetricgrpc"
    "go.opentelemetry.io/otel/metric"
    "go.opentelemetry.io/otel/propagation"
    "go.opentelemetry.io/otel/sdk/resource"
    sdkmetric "go.opentelemetry.io/otel/sdk/metric"
    sdktrace "go.opentelemetry.io/otel/sdk/trace"
    semconv "go.opentelemetry.io/otel/semconv/v1.26.0"
)

// initOTel: initialise SDK with OTLP exporters pointing at Alloy/Collector
func initOTel(ctx context.Context) (shutdown func(context.Context) error) {
    res, _ := resource.New(ctx,
        resource.WithAttributes(
            semconv.ServiceName("construx-api"),
            semconv.ServiceVersion("v1.42.0"),
            semconv.DeploymentEnvironment("production"),
        ),
    )

    // Traces → OTLP gRPC → Alloy → Tempo
    traceExporter, _ := otlptracegrpc.New(ctx,
        otlptracegrpc.WithEndpoint("alloy.monitoring.svc.cluster.local:4317"),
        otlptracegrpc.WithInsecure(),
    )
    tp := sdktrace.NewTracerProvider(
        sdktrace.WithResource(res),
        sdktrace.WithBatcher(traceExporter),
        sdktrace.WithSampler(sdktrace.ParentBased(
            sdktrace.TraceIDRatioBased(0.1),   // sample 10% of root spans
        )),
    )
    otel.SetTracerProvider(tp)
    otel.SetTextMapPropagator(propagation.NewCompositeTextMapPropagator(
        propagation.TraceContext{},   // W3C traceparent header
        propagation.Baggage{},
    ))

    // Metrics → OTLP gRPC → Alloy → Mimir
    metricExporter, _ := otlpmetricgrpc.New(ctx,
        otlpmetricgrpc.WithEndpoint("alloy.monitoring.svc.cluster.local:4317"),
        otlpmetricgrpc.WithInsecure(),
    )
    mp := sdkmetric.NewMeterProvider(
        sdkmetric.WithResource(res),
        sdkmetric.WithReader(sdkmetric.NewPeriodicReader(metricExporter,
            sdkmetric.WithInterval(30*time.Second),
        )),
    )
    otel.SetMeterProvider(mp)

    return func(ctx context.Context) error {
        tp.Shutdown(ctx)
        mp.Shutdown(ctx)
        return nil
    }
}

// Handler: manual span creation with attributes
func handleCreateOrder(w http.ResponseWriter, r *http.Request) {
    tracer := otel.Tracer("construx-api")
    meter  := otel.Meter("construx-api")

    // Extract incoming trace context from request headers
    ctx := otel.GetTextMapPropagator().Extract(r.Context(),
        propagation.HeaderCarrier(r.Header),
    )

    // Create a span for this operation
    ctx, span := tracer.Start(ctx, "http.create_order",
        trace.WithAttributes(
            attribute.String("tenant.id", r.Header.Get("X-Tenant-ID")),
            attribute.String("http.method", r.Method),
            attribute.String("http.route",  "/api/v1/orders"),
        ),
    )
    defer span.End()

    // Record a business metric: orders created counter
    ordersCounter, _ := meter.Int64Counter("orders.created",
        metric.WithDescription("Total orders created"),
        metric.WithUnit("{order}"),
    )
    ordersCounter.Add(ctx, 1,
        metric.WithAttributes(
            attribute.String("tenant.id", r.Header.Get("X-Tenant-ID")),
        ),
    )

    // Nested span: database write
    ctx, dbSpan := tracer.Start(ctx, "db.insert_order")
    if err := insertOrder(ctx, r); err != nil {
        dbSpan.RecordError(err)
        dbSpan.SetStatus(codes.Error, err.Error())
        http.Error(w, "internal error", 500)
        dbSpan.End()
        span.End()
        return
    }
    dbSpan.End()

    // When making outbound HTTP calls: inject context into headers
    req, _ := http.NewRequestWithContext(ctx, "POST", "http://payment-service/charge", nil)
    otel.GetTextMapPropagator().Inject(ctx, propagation.HeaderCarrier(req.Header))
    // req.Header["Traceparent"] = "00-4bf92f3577b34da6-a3ce929d0e0e4736-01"
    http.DefaultClient.Do(req)
}
```

```python
# Python SDK: equivalent instrumentation for a FastAPI service

from opentelemetry import trace, metrics
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter
from opentelemetry.sdk.metrics import MeterProvider
from opentelemetry.sdk.metrics.export import PeriodicExportingMetricReader
from opentelemetry.exporter.otlp.proto.grpc.metric_exporter import OTLPMetricExporter

# Initialise OTel SDK
tracer_provider = TracerProvider()
tracer_provider.add_span_processor(
    BatchSpanProcessor(OTLPSpanExporter(endpoint="http://alloy:4317"))
)
trace.set_tracer_provider(tracer_provider)

meter_provider = MeterProvider(
    metric_readers=[PeriodicExportingMetricReader(
        OTLPMetricExporter(endpoint="http://alloy:4317"),
        export_interval_millis=30_000,
    )]
)
metrics.set_meter_provider(meter_provider)

tracer = trace.get_tracer("construx-payments")
meter  = metrics.get_meter("construx-payments")

# Record a histogram for payment processing duration
payment_duration = meter.create_histogram(
    "payment.processing.duration",
    unit="ms",
    description="Time to process a payment",
)

def process_payment(payment_id: str, amount: float, tenant_id: str):
    with tracer.start_as_current_span("payment.process") as span:
        span.set_attribute("payment.id",       payment_id)
        span.set_attribute("payment.amount",   amount)
        span.set_attribute("tenant.id",        tenant_id)

        start = time.monotonic()
        try:
            result = stripe_client.charge(amount)
            span.set_attribute("payment.status", "succeeded")
        except stripe.CardError as e:
            span.record_exception(e)
            span.set_status(StatusCode.ERROR, str(e))
            span.set_attribute("payment.status", "declined")
        finally:
            duration_ms = (time.monotonic() - start) * 1000
            payment_duration.record(duration_ms, {"tenant.id": tenant_id})
```

OTel manual instrumentation vs auto-instrumentation vs Pixie vs Datadog APM: Auto-instrumentation (OTel Java agent, Python instrumentor) intercepts framework calls (HTTP handlers, database drivers) automatically — zero code changes but limited to what the instrumentor knows about. Pixie (dispatch-1084) captures at the kernel level — also zero code changes, even better coverage for protocols, but 24-hour data window and no business-level attributes (you can't add tenant_id to a Pixie trace). Manual OTel SDK instrumentation is the only approach that lets you add business-domain attributes (tenant_id, order_id, payment_amount) to spans, create spans for custom operations (a specific algorithm, an internal queue drain), and record business metrics (orders created, payment decline rate). The production pattern is all three in combination: Pixie for instant baseline RED metrics, auto-instrumentation for framework-level spans (HTTP, DB, gRPC), and manual SDK instrumentation for the business operations that matter for SLO tracking and incident investigation.
