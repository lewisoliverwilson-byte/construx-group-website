CSS transitions work for simple state changes. When you need enter/exit animations, layout shifts, or anything orchestrated, CSS becomes a fight. Framer Motion treats animation as a natural extension of React state: you describe what a component looks like in different states, and the library handles the transition.

## Setup

```bash
npm install framer-motion
```

## Fade and slide on enter

```tsx
'use client';

import { motion } from 'framer-motion';

// Fade in from below on mount
export function FadeIn({ children, delay = 0 }: { children: React.ReactNode; delay?: number }) {
  return (
    <motion.div
      initial={{ opacity: 0, y: 16 }}
      animate={{ opacity: 1, y: 0 }}
      transition={{ duration: 0.4, delay, ease: [0.25, 0.1, 0.25, 1] }}
    >
      {children}
    </motion.div>
  );
}

// Staggered children
export function StaggerList({ items }: { items: string[] }) {
  return (
    <motion.ul
      initial="hidden"
      animate="visible"
      variants={{
        visible: {
          transition: { staggerChildren: 0.08 },
        },
      }}
    >
      {items.map((item) => (
        <motion.li
          key={item}
          variants={{
            hidden: { opacity: 0, x: -12 },
            visible: { opacity: 1, x: 0 },
          }}
          transition={{ duration: 0.3 }}
        >
          {item}
        </motion.li>
      ))}
    </motion.ul>
  );
}
```

## Enter/exit with AnimatePresence

```tsx
'use client';

import { AnimatePresence, motion } from 'framer-motion';
import { useState } from 'react';

export function Modal({ children }: { children: React.ReactNode }) {
  const [open, setOpen] = useState(false);

  return (
    <>
      <button onClick={() => setOpen(true)}>Open</button>

      <AnimatePresence>
        {open && (
          <>
            {/* Backdrop */}
            <motion.div
              key="backdrop"
              className="fixed inset-0 bg-black/60 z-40"
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              exit={{ opacity: 0 }}
              onClick={() => setOpen(false)}
            />

            {/* Modal panel */}
            <motion.div
              key="modal"
              className="fixed top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 z-50 w-full max-w-md bg-neutral-900 rounded-lg p-6"
              initial={{ opacity: 0, scale: 0.95, y: -8 }}
              animate={{ opacity: 1, scale: 1, y: 0 }}
              exit={{ opacity: 0, scale: 0.95, y: 8 }}
              transition={{ duration: 0.2, ease: [0.36, 0.66, 0.04, 1] }}
            >
              {children}
            </motion.div>
          </>
        )}
      </AnimatePresence>
    </>
  );
}
```

## Layout animations

```tsx
'use client';

import { motion, LayoutGroup } from 'framer-motion';
import { useState } from 'react';

const TABS = ['Overview', 'Analytics', 'Settings'];

export function TabPanel() {
  const [active, setActive] = useState('Overview');

  return (
    <LayoutGroup>
      <div className="flex gap-1 relative">
        {TABS.map((tab) => (
          <button
            key={tab}
            onClick={() => setActive(tab)}
            className="relative px-4 py-2 text-sm font-medium z-10"
            style={{ color: active === tab ? '#000' : 'rgba(240,239,255,0.6)' }}
          >
            {active === tab && (
              <motion.div
                layoutId="tab-indicator"
                className="absolute inset-0 bg-construx rounded"
                transition={{ type: 'spring', stiffness: 400, damping: 30 }}
              />
            )}
            <span className="relative z-10">{tab}</span>
          </button>
        ))}
      </div>
    </LayoutGroup>
  );
}
```

The `layoutId` prop connects elements across renders — when `active` changes, Framer Motion automatically animates the indicator between positions.

## Scroll-triggered reveal

```tsx
'use client';

import { motion, useInView } from 'framer-motion';
import { useRef } from 'react';

export function RevealOnScroll({ children }: { children: React.ReactNode }) {
  const ref = useRef<HTMLDivElement>(null);
  const inView = useInView(ref, { once: true, margin: '-80px' });

  return (
    <motion.div
      ref={ref}
      initial={{ opacity: 0, y: 32 }}
      animate={inView ? { opacity: 1, y: 0 } : {}}
      transition={{ duration: 0.5, ease: [0.25, 0.1, 0.25, 1] }}
    >
      {children}
    </motion.div>
  );
}
```

## Gesture responses

```tsx
'use client';

import { motion } from 'framer-motion';

export function InteractiveCard({ title, description }: { title: string; description: string }) {
  return (
    <motion.div
      className="p-6 rounded-lg bg-neutral-900 border border-white/8 cursor-pointer"
      whileHover={{ y: -4, borderColor: 'rgba(249,115,22,0.3)', boxShadow: '0 12px 32px rgba(0,0,0,0.4)' }}
      whileTap={{ scale: 0.98 }}
      transition={{ type: 'spring', stiffness: 400, damping: 30 }}
    >
      <h3 className="text-base font-semibold mb-2">{title}</h3>
      <p className="text-sm text-text-muted">{description}</p>
    </motion.div>
  );
}
```

## Drag

```tsx
'use client';

import { motion, useMotionValue, useTransform } from 'framer-motion';

export function DraggableSlider({ onValue }: { onValue: (v: number) => void }) {
  const x = useMotionValue(0);
  const width = 300;

  const percentage = useTransform(x, [0, width], [0, 100]);

  return (
    <div className="relative h-6 w-[300px] bg-white/10 rounded-full overflow-hidden">
      {/* Track fill */}
      <motion.div
        className="absolute left-0 top-0 bottom-0 bg-construx/40 rounded-full"
        style={{ width: useTransform(x, [0, width], ['0%', '100%']) }}
      />

      {/* Handle */}
      <motion.div
        className="absolute top-1/2 -translate-y-1/2 w-5 h-5 bg-construx rounded-full cursor-grab active:cursor-grabbing"
        style={{ x }}
        drag="x"
        dragConstraints={{ left: 0, right: width }}
        dragElastic={0}
        dragMomentum={false}
        onDrag={() => onValue(Math.round(percentage.get()))}
      />
    </div>
  );
}
```

## Scroll-linked progress

```tsx
'use client';

import { motion, useScroll, useTransform } from 'framer-motion';

export function ReadingProgressBar() {
  const { scrollYProgress } = useScroll();

  return (
    <motion.div
      className="fixed top-0 left-0 right-0 h-0.5 bg-construx z-50 origin-left"
      style={{ scaleX: scrollYProgress }}
    />
  );
}

// Parallax effect
export function ParallaxHero({ children }: { children: React.ReactNode }) {
  const { scrollY } = useScroll();
  const y = useTransform(scrollY, [0, 600], [0, -120]);

  return (
    <motion.div style={{ y }}>
      {children}
    </motion.div>
  );
}
```

## Performance notes

Framer Motion uses the Web Animations API when possible to keep animations off the main thread. For GPU-composited properties (`transform`, `opacity`), it runs at 60fps even on busy pages. For other CSS properties (`height`, `background`, `border`), it causes layout reflow on every frame — avoid animating these.

The `will-change: transform` trick is unnecessary — Framer Motion handles compositing hints internally.

Use `useReducedMotion` to respect user accessibility preferences:

```tsx
import { useReducedMotion } from 'framer-motion';

function AnimatedCard() {
  const shouldReduceMotion = useReducedMotion();

  return (
    <motion.div
      initial={{ opacity: 0, y: shouldReduceMotion ? 0 : 20 }}
      animate={{ opacity: 1, y: 0 }}
      transition={{ duration: shouldReduceMotion ? 0 : 0.4 }}
    >
      ...
    </motion.div>
  );
}
```

Framer Motion's API surface is large but the 20% you use for 80% of cases is: `initial`, `animate`, `exit`, `transition`, `whileHover`, `whileTap`, `AnimatePresence`, and `layoutId`. Learn those first.
