Terraform's model: you describe what you want, Terraform computes a diff against current state, shows you the plan, and applies the changes. No manual clicking in the AWS console, no drift between environments, no forgotten resources.

## Provider setup

```hcl
# main.tf
terraform {
  required_version = ">= 1.9"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }

  backend "s3" {
    bucket         = "construx-terraform-state"
    key            = "prod/terraform.tfstate"
    region         = "eu-west-2"
    dynamodb_table = "terraform-locks"
    encrypt        = true
  }
}

provider "aws" {
  region = var.aws_region
}
```

## Variables and outputs

```hcl
# variables.tf
variable "aws_region" {
  type    = string
  default = "eu-west-2"
}

variable "environment" {
  type    = string
  validation {
    condition     = contains(["dev", "staging", "prod"], var.environment)
    error_message = "Must be dev, staging, or prod."
  }
}

variable "db_password" {
  type      = string
  sensitive = true
}

# outputs.tf
output "api_url" {
  value = "https://${aws_cloudfront_distribution.api.domain_name}"
}

output "db_endpoint" {
  value     = aws_db_instance.main.endpoint
  sensitive = true
}
```

## VPC and networking

```hcl
# network.tf
resource "aws_vpc" "main" {
  cidr_block           = "10.0.0.0/16"
  enable_dns_hostnames = true

  tags = { Name = "${var.environment}-vpc" }
}

resource "aws_subnet" "private" {
  count             = 2
  vpc_id            = aws_vpc.main.id
  cidr_block        = cidrsubnet(aws_vpc.main.cidr_block, 8, count.index)
  availability_zone = data.aws_availability_zones.available.names[count.index]

  tags = { Name = "${var.environment}-private-${count.index}" }
}

resource "aws_subnet" "public" {
  count                   = 2
  vpc_id                  = aws_vpc.main.id
  cidr_block              = cidrsubnet(aws_vpc.main.cidr_block, 8, count.index + 10)
  availability_zone       = data.aws_availability_zones.available.names[count.index]
  map_public_ip_on_launch = true

  tags = { Name = "${var.environment}-public-${count.index}" }
}

data "aws_availability_zones" "available" { state = "available" }
```

## RDS PostgreSQL

```hcl
# database.tf
resource "aws_db_instance" "main" {
  identifier        = "${var.environment}-construx-db"
  engine            = "postgres"
  engine_version    = "16.4"
  instance_class    = "db.t4g.medium"
  allocated_storage = 20
  storage_encrypted = true

  db_name  = "construx"
  username = "construx_admin"
  password = var.db_password

  vpc_security_group_ids = [aws_security_group.rds.id]
  db_subnet_group_name   = aws_db_subnet_group.main.name

  backup_retention_period = 7
  deletion_protection     = var.environment == "prod"
  skip_final_snapshot     = var.environment != "prod"

  tags = { Environment = var.environment }
}

resource "aws_db_subnet_group" "main" {
  name       = "${var.environment}-db-subnet-group"
  subnet_ids = aws_subnet.private[*].id
}
```

## ECS Fargate service

```hcl
# ecs.tf
resource "aws_ecs_cluster" "main" {
  name = "${var.environment}-construx"
}

resource "aws_ecs_task_definition" "api" {
  family                   = "${var.environment}-construx-api"
  cpu                      = 512
  memory                   = 1024
  network_mode             = "awsvpc"
  requires_compatibilities = ["FARGATE"]
  execution_role_arn       = aws_iam_role.ecs_execution.arn

  container_definitions = jsonencode([{
    name  = "api"
    image = "${aws_ecr_repository.api.repository_url}:latest"

    portMappings = [{ containerPort = 3000, protocol = "tcp" }]

    environment = [
      { name = "NODE_ENV",    value = "production" },
      { name = "PORT",        value = "3000" },
    ]

    secrets = [
      { name = "DATABASE_URL", valueFrom = aws_ssm_parameter.db_url.arn },
    ]

    logConfiguration = {
      logDriver = "awslogs"
      options = {
        awslogs-group  = "/ecs/${var.environment}-construx-api"
        awslogs-region = var.aws_region
      }
    }
  }])
}

resource "aws_ecs_service" "api" {
  name            = "${var.environment}-construx-api"
  cluster         = aws_ecs_cluster.main.id
  task_definition = aws_ecs_task_definition.api.arn
  desired_count   = var.environment == "prod" ? 2 : 1
  launch_type     = "FARGATE"

  network_configuration {
    subnets         = aws_subnet.private[*].id
    security_groups = [aws_security_group.ecs.id]
  }

  load_balancer {
    target_group_arn = aws_lb_target_group.api.arn
    container_name   = "api"
    container_port   = 3000
  }
}
```

## The plan/apply workflow

```bash
# Initialise (downloads providers, configures backend)
terraform init

# Format and validate
terraform fmt -recursive
terraform validate

# Show what would change
terraform plan -var="environment=prod" -var="db_password=$DB_PASS"

# Apply changes
terraform apply -var="environment=prod" -var="db_password=$DB_PASS"

# Destroy infrastructure
terraform destroy -var="environment=dev"

# Show current state
terraform show
terraform state list
terraform state show aws_db_instance.main
```

## Modules for reuse

```hcl
# modules/rds/main.tf — reusable RDS module
variable "identifier" { type = string }
variable "password"   { type = string; sensitive = true }
variable "subnet_ids" { type = list(string) }

resource "aws_db_instance" "this" {
  identifier = var.identifier
  # ...
}

output "endpoint" { value = aws_db_instance.this.endpoint }
```

```hcl
# Root: use the module
module "prod_db" {
  source = "./modules/rds"

  identifier = "prod-construx"
  password   = var.db_password
  subnet_ids = aws_subnet.private[*].id
}

output "db_url" {
  value = "postgresql://${module.prod_db.endpoint}/construx"
}
```

Modules encapsulate resource groups. Use them for anything you'll deploy more than once — VPCs, RDS instances, ECS services.
