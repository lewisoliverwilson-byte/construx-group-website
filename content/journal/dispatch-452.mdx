Semantic search requires vector embeddings — numerical representations of text where similar meanings cluster together in high-dimensional space. Traditionally you'd push these to Pinecone, Weaviate, or Qdrant. pgvector puts the same capability inside Postgres.

## Setup

```sql
-- Enable the extension
CREATE EXTENSION IF NOT EXISTS vector;

-- Add a vector column (1536 dimensions for text-embedding-3-small)
ALTER TABLE posts ADD COLUMN embedding vector(1536);

-- Create an HNSW index for approximate nearest-neighbor search
CREATE INDEX ON posts USING hnsw (embedding vector_cosine_ops)
WITH (m = 16, ef_construction = 64);
```

HNSW (Hierarchical Navigable Small World) is the recommended index type for pgvector. It trades a small accuracy reduction for dramatically faster search — queries return in milliseconds at scale.

## Drizzle schema

```ts
// lib/db/schema.ts
import { pgTable, text, uuid, integer } from 'drizzle-orm/pg-core';
import { customType } from 'drizzle-orm/pg-core';

const vector = customType<{ data: number[]; driverData: string }>({
  dataType(config) {
    return `vector(${config?.length ?? 1536})`;
  },
  toDriver(value: number[]): string {
    return JSON.stringify(value);
  },
  fromDriver(value: string): number[] {
    return typeof value === 'string' ? JSON.parse(value) : value;
  },
});

export const posts = pgTable('posts', {
  id: uuid('id').primaryKey().defaultRandom(),
  title: text('title').notNull(),
  excerpt: text('excerpt').notNull(),
  content: text('content').notNull(),
  embedding: vector('embedding', { length: 1536 }),
  tokenCount: integer('token_count'),
});
```

## Generating embeddings with Claude

```ts
// lib/embeddings.ts
import Anthropic from '@anthropic-ai/sdk';

const client = new Anthropic();

export async function embed(text: string): Promise<number[]> {
  const response = await client.beta.messages.create({
    model: 'claude-opus-4-8',
    max_tokens: 1,  // We only want the embedding, not a completion
    messages: [{ role: 'user', content: text }],
    // Note: Use text-embedding-3-small from OpenAI for pure embeddings at scale
    // Claude's native embedding endpoint is in beta
  });

  // For production: use OpenAI's text-embedding-3-small (1536 dims, cheap)
  throw new Error('Use OpenAI embeddings endpoint or Voyage AI');
}

// In practice with OpenAI:
import OpenAI from 'openai';

const openai = new OpenAI();

export async function embedText(text: string): Promise<number[]> {
  const response = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: text.slice(0, 8192),  // Token limit
  });
  return response.data[0].embedding;
}

export async function embedBatch(texts: string[]): Promise<number[][]> {
  const response = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: texts.map((t) => t.slice(0, 8192)),
  });
  return response.data.map((d) => d.embedding);
}
```

## Storing embeddings

```ts
// lib/indexing.ts
import { db } from './db';
import { posts } from './db/schema';
import { embedText } from './embeddings';
import { eq } from 'drizzle-orm';

export async function indexPost(postId: string) {
  const post = await db.query.posts.findFirst({
    where: eq(posts.id, postId),
  });
  if (!post) throw new Error(`Post ${postId} not found`);

  // Concatenate fields for better embedding quality
  const text = `${post.title}\n\n${post.excerpt}\n\n${post.content}`;
  const embedding = await embedText(text);

  await db
    .update(posts)
    .set({ embedding })
    .where(eq(posts.id, postId));
}

// Bulk index all posts without embeddings
export async function indexAllPosts() {
  const unindexed = await db.query.posts.findMany({
    where: isNull(posts.embedding),
  });

  // Process in batches to stay within rate limits
  const BATCH_SIZE = 20;
  for (let i = 0; i < unindexed.length; i += BATCH_SIZE) {
    const batch = unindexed.slice(i, i + BATCH_SIZE);
    const texts = batch.map((p) => `${p.title}\n\n${p.excerpt}`);
    const embeddings = await embedBatch(texts);

    await Promise.all(
      batch.map((post, j) =>
        db.update(posts).set({ embedding: embeddings[j] }).where(eq(posts.id, post.id))
      )
    );

    console.log(`Indexed ${i + batch.length}/${unindexed.length}`);
  }
}
```

## Similarity search

```ts
// lib/search.ts
import { db } from './db';
import { sql } from 'drizzle-orm';
import { embedText } from './embeddings';

interface SearchResult {
  id: string;
  title: string;
  excerpt: string;
  similarity: number;
}

export async function semanticSearch(
  query: string,
  limit = 10,
  threshold = 0.7,
): Promise<SearchResult[]> {
  const queryEmbedding = await embedText(query);
  const embeddingStr = JSON.stringify(queryEmbedding);

  const results = await db.execute(sql`
    SELECT
      id,
      title,
      excerpt,
      1 - (embedding <=> ${embeddingStr}::vector) AS similarity
    FROM posts
    WHERE embedding IS NOT NULL
      AND 1 - (embedding <=> ${embeddingStr}::vector) > ${threshold}
    ORDER BY embedding <=> ${embeddingStr}::vector
    LIMIT ${limit}
  `);

  return results.rows as SearchResult[];
}
```

The `<=>` operator is cosine distance (0 = identical, 2 = opposite). `1 - distance` gives similarity (1 = identical, -1 = opposite).

## Hybrid search: combine FTS and vector

```ts
export async function hybridSearch(query: string, limit = 10): Promise<SearchResult[]> {
  const queryEmbedding = await embedText(query);
  const embeddingStr = JSON.stringify(queryEmbedding);

  // Reciprocal Rank Fusion (RRF) combines rankings from both methods
  const results = await db.execute(sql`
    WITH semantic AS (
      SELECT id, RANK() OVER (ORDER BY embedding <=> ${embeddingStr}::vector) AS rank
      FROM posts
      WHERE embedding IS NOT NULL
      ORDER BY embedding <=> ${embeddingStr}::vector
      LIMIT 50
    ),
    fulltext AS (
      SELECT id, RANK() OVER (ORDER BY ts_rank(search_vector, query) DESC) AS rank
      FROM posts, plainto_tsquery('english', ${query}) query
      WHERE search_vector @@ query
      LIMIT 50
    )
    SELECT
      p.id,
      p.title,
      p.excerpt,
      COALESCE(1.0 / (60 + s.rank), 0) + COALESCE(1.0 / (60 + f.rank), 0) AS score
    FROM posts p
    LEFT JOIN semantic s ON p.id = s.id
    LEFT JOIN fulltext f ON p.id = f.id
    WHERE s.id IS NOT NULL OR f.id IS NOT NULL
    ORDER BY score DESC
    LIMIT ${limit}
  `);

  return results.rows as SearchResult[];
}
```

Hybrid search consistently outperforms either method alone. Semantic search catches conceptually similar results that don't share keywords. Full-text search catches exact matches that might have low cosine similarity.

## Re-ranking with Claude

For high-stakes search (customer-facing, paid product), a two-stage approach: first retrieve top-N candidates, then rerank with an LLM:

```ts
import Anthropic from '@anthropic-ai/sdk';

const client = new Anthropic();

export async function rerankResults(
  query: string,
  candidates: SearchResult[],
  topK = 5,
): Promise<SearchResult[]> {
  const response = await client.messages.create({
    model: 'claude-haiku-4-5-20251001',
    max_tokens: 256,
    messages: [
      {
        role: 'user',
        content: `Query: "${query}"

Candidates (ranked 0 to ${candidates.length - 1}):
${candidates.map((c, i) => `${i}. ${c.title}: ${c.excerpt.slice(0, 100)}`).join('\n')}

Return the indices of the top ${topK} most relevant results as a JSON array, most relevant first.
Only return the array, no explanation.`,
      },
    ],
  });

  const text = response.content[0].type === 'text' ? response.content[0].text : '[]';
  const indices = JSON.parse(text) as number[];
  return indices.slice(0, topK).map((i) => candidates[i]).filter(Boolean);
}
```

The full pipeline: embeddings in Postgres → HNSW index → hybrid query → LLM reranking. No external vector database, no additional infra, no data synchronisation complexity.
