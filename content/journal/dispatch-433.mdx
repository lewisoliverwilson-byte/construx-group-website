GraphQL schema-first development has a sync problem: your SDL file and your TypeScript types live in separate worlds. Pothos eliminates that by deriving the schema from TypeScript code. The schema is always in sync because there's only one source of truth.

## Setup

```bash
npm install @pothos/core graphql graphql-yoga
```

```ts
// lib/schema/builder.ts
import SchemaBuilder from '@pothos/core';

const builder = new SchemaBuilder<{
  Context: {
    userId: string | null;
    db: DB;
  };
}>({});

export default builder;
```

## Defining types

```ts
// lib/schema/types/post.ts
import builder from '../builder';
import { Post as DBPost } from '@/lib/db/schema';

builder.objectType('Post', {
  description: 'A blog post',
  fields: (t) => ({
    id: t.exposeID('id'),
    title: t.exposeString('title'),
    excerpt: t.exposeString('excerpt'),
    publishedAt: t.expose('publishedAt', { type: 'DateTime' }),
    author: t.field({
      type: 'Author',
      resolve: (post, _, ctx) => ctx.db.getAuthor(post.authorId),
    }),
    wordCount: t.int({
      resolve: (post) => post.content.split(/\s+/).length,
    }),
  }),
});
```

```ts
// lib/schema/types/author.ts
builder.objectType('Author', {
  fields: (t) => ({
    id: t.exposeID('id'),
    name: t.exposeString('name'),
    bio: t.exposeString('bio', { nullable: true }),
    posts: t.field({
      type: ['Post'],
      resolve: (author, _, ctx) => ctx.db.getPostsByAuthor(author.id),
    }),
  }),
});
```

## Queries

```ts
// lib/schema/resolvers/query.ts
builder.queryType({
  fields: (t) => ({
    post: t.field({
      type: 'Post',
      nullable: true,
      args: {
        slug: t.arg.string({ required: true }),
      },
      resolve: (_, { slug }, ctx) => ctx.db.getPostBySlug(slug),
    }),

    posts: t.field({
      type: ['Post'],
      args: {
        limit: t.arg.int({ defaultValue: 20 }),
        offset: t.arg.int({ defaultValue: 0 }),
      },
      resolve: (_, { limit, offset }, ctx) =>
        ctx.db.getPosts({ limit: limit ?? 20, offset: offset ?? 0 }),
    }),
  }),
});
```

## Mutations

```ts
builder.mutationType({
  fields: (t) => ({
    createPost: t.field({
      type: 'Post',
      args: {
        input: t.arg({ type: CreatePostInput, required: true }),
      },
      resolve: async (_, { input }, ctx) => {
        if (!ctx.userId) throw new Error('Unauthorized');
        return ctx.db.createPost({ ...input, authorId: ctx.userId });
      },
    }),
  }),
});

const CreatePostInput = builder.inputType('CreatePostInput', {
  fields: (t) => ({
    title: t.string({ required: true }),
    content: t.string({ required: true }),
    tags: t.stringList({ defaultValue: [] }),
  }),
});
```

## Relay pagination

Pothos has a Relay plugin for cursor-based pagination:

```bash
npm install @pothos/plugin-relay
```

```ts
import RelayPlugin from '@pothos/plugin-relay';

const builder = new SchemaBuilder<{
  Context: { userId: string | null; db: DB };
}>({
  plugins: [RelayPlugin],
  relay: {},
});

builder.node('Post', {
  id: { resolve: (post) => post.id },
  fields: (t) => ({
    title: t.exposeString('title'),
    // ... other fields
  }),
  loadOne: (id, ctx) => ctx.db.getPostById(id),
  loadMany: (ids, ctx) => ctx.db.getPostsByIds(ids),
});

builder.queryFields((t) => ({
  posts: t.connection({
    type: 'Post',
    resolve: async (_, args, ctx) => {
      const { first, after } = args;
      return ctx.db.getPostsConnection({ first: first ?? 20, after });
    },
  }),
}));
```

## Serving with GraphQL Yoga

```ts
// app/api/graphql/route.ts
import { createYoga } from 'graphql-yoga';
import { schema } from '@/lib/schema';
import { createContext } from '@/lib/schema/context';

const yoga = createYoga({
  schema,
  context: createContext,
  graphqlEndpoint: '/api/graphql',
});

export { yoga as GET, yoga as POST };
```

```ts
// lib/schema/context.ts
import { NextRequest } from 'next/server';
import { getServerSession } from 'next-auth';
import { db } from '@/lib/db';

export async function createContext({ request }: { request: NextRequest }) {
  const session = await getServerSession();
  return {
    userId: session?.user?.id ?? null,
    db,
  };
}
```

## Schema export

```ts
// lib/schema/index.ts
import './types/post';
import './types/author';
import './resolvers/query';
import './resolvers/mutation';

export const schema = builder.toSchema();
```

The pattern: Pothos keeps TypeScript and GraphQL in lockstep by making TypeScript the source of truth. No SDL files, no codegen pipeline, no drift. The schema emerges from the types.
