AWS infrastructure as real TypeScript. CloudFormation generated, not written.

## Stack definition

```ts
// lib/construx-stack.ts
import * as cdk            from 'aws-cdk-lib';
import * as s3             from 'aws-cdk-lib/aws-s3';
import * as cloudfront     from 'aws-cdk-lib/aws-cloudfront';
import * as origins        from 'aws-cdk-lib/aws-cloudfront-origins';
import * as iam            from 'aws-cdk-lib/aws-iam';
import * as route53        from 'aws-cdk-lib/aws-route53';
import * as targets        from 'aws-cdk-lib/aws-route53-targets';
import * as acm            from 'aws-cdk-lib/aws-certificatemanager';
import { Construct }       from 'constructs';

export class ConstruxStack extends cdk.Stack {
  constructor(scope: Construct, id: string, props?: cdk.StackProps) {
    super(scope, id, props);

    // S3 bucket for static assets
    const assetsBucket = new s3.Bucket(this, 'Assets', {
      bucketName:         `construx-assets-${this.account}`,
      blockPublicAccess:  s3.BlockPublicAccess.BLOCK_ALL,
      versioned:          true,
      lifecycleRules: [{
        expiration:    cdk.Duration.days(365),
        prefix:        'temp/',
      }],
    });

    // ACM certificate — must be in us-east-1 for CloudFront
    const certificate = acm.Certificate.fromCertificateArn(
      this, 'Certificate',
      process.env.ACM_CERT_ARN!,
    );

    // CloudFront distribution
    const distribution = new cloudfront.Distribution(this, 'CDN', {
      defaultBehavior: {
        origin:         new origins.S3Origin(assetsBucket),
        viewerProtocol: cloudfront.ViewerProtocol.REDIRECT_TO_HTTPS,
        cachePolicy:    cloudfront.CachePolicy.CACHING_OPTIMIZED,
      },
      domainNames:   ['assets.construxgroup.io'],
      certificate,
      priceClass:    cloudfront.PriceClass.PRICE_CLASS_100,
    });

    // Route 53 alias record
    const zone = route53.HostedZone.fromLookup(this, 'Zone', {
      domainName: 'construxgroup.io',
    });

    new route53.ARecord(this, 'AssetsDnsRecord', {
      zone,
      recordName: 'assets',
      target:     route53.RecordTarget.fromAlias(
        new targets.CloudFrontTarget(distribution),
      ),
    });

    // Output the distribution URL
    new cdk.CfnOutput(this, 'DistributionUrl', {
      value:       `https://${distribution.distributionDomainName}`,
      description: 'CloudFront distribution URL',
    });
  }
}
```

## App entry point

```ts
// bin/construx.ts
import * as cdk from 'aws-cdk-lib';
import { ConstruxStack } from '../lib/construx-stack';

const app = new cdk.App();

new ConstruxStack(app, 'ConstruxProd', {
  env: {
    account: process.env.CDK_DEFAULT_ACCOUNT,
    region:  'eu-west-2',
  },
  tags: {
    Project:     'construx-website',
    Environment: 'production',
    ManagedBy:   'cdk',
  },
});
```

## Deploy

```bash
# Synthesize — generate CloudFormation template
cdk synth

# Diff — what will change
cdk diff ConstruxProd

# Deploy
cdk deploy ConstruxProd

# Destroy (careful!)
cdk destroy ConstruxProd
```

## Custom constructs — reusable building blocks

```ts
// lib/constructs/secure-bucket.ts
import * as s3   from 'aws-cdk-lib/aws-s3';
import * as kms  from 'aws-cdk-lib/aws-kms';
import { Construct } from 'constructs';

interface SecureBucketProps {
  bucketName: string;
  versioned?:  boolean;
}

export class SecureBucket extends Construct {
  public readonly bucket: s3.Bucket;

  constructor(scope: Construct, id: string, props: SecureBucketProps) {
    super(scope, id);

    const key = new kms.Key(this, 'Key', { enableKeyRotation: true });

    this.bucket = new s3.Bucket(this, 'Bucket', {
      bucketName:         props.bucketName,
      encryptionKey:      key,
      blockPublicAccess:  s3.BlockPublicAccess.BLOCK_ALL,
      versioned:          props.versioned ?? true,
      enforceSSL:         true,
    });
  }
}

// Usage — same interface everywhere
const assets  = new SecureBucket(this, 'Assets', { bucketName: 'construx-assets' });
const backups  = new SecureBucket(this, 'Backups', { bucketName: 'construx-backups' });
```

## GitHub Actions OIDC + CDK

```yaml
jobs:
  deploy:
    permissions:
      id-token: write
      contents: read

    steps:
      - uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::123456789:role/github-cdk-deploy
          aws-region:     eu-west-2

      - run: bun install
      - run: bunx cdk deploy --require-approval never ConstruxProd
```

CDK vs Terraform vs Pulumi: Terraform has the largest ecosystem and works across cloud providers. CDK is AWS-only but generates CloudFormation (auditable by AWS tooling). Pulumi supports multiple clouds with real programming languages. The Construx default: CDK for AWS-only infrastructure; Terraform when multi-cloud or when existing Terraform state is in use.
