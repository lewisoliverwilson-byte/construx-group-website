Controlled inputs in React re-render the component on every keystroke. For a form with 20 fields and 100 renders/second, that is 2000 re-renders per second while the user types. React Hook Form avoids this by registering refs to DOM elements and reading values only when needed.

## Install

```bash
npm install react-hook-form
npm install @hookform/resolvers zod
```

## Basic form with Zod validation

```tsx
'use client';
import { useForm }              from 'react-hook-form';
import { zodResolver }          from '@hookform/resolvers/zod';
import { z }                    from 'zod';

const PostSchema = z.object({
  title:   z.string().min(1, 'Title is required').max(200),
  content: z.string().min(10, 'Content must be at least 10 characters'),
  tags:    z.array(z.string()).max(10).default([]),
  slug:    z.string().regex(/^[a-z0-9-]+$/, 'Slug must be lowercase with hyphens only'),
});

type PostForm = z.infer<typeof PostSchema>;

function NewPostForm() {
  const { register, handleSubmit, formState: { errors, isSubmitting } } = useForm<PostForm>({
    resolver:      zodResolver(PostSchema),
    defaultValues: { tags: [] },
  });

  const onSubmit = async (data: PostForm) => {
    await fetch('/api/posts', { method: 'POST', body: JSON.stringify(data) });
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <div>
        <input {...register('title')} placeholder="Title" />
        {errors.title && <p>{errors.title.message}</p>}
      </div>

      <div>
        <textarea {...register('content')} placeholder="Content" />
        {errors.content && <p>{errors.content.message}</p>}
      </div>

      <div>
        <input {...register('slug')} placeholder="slug-here" />
        {errors.slug && <p>{errors.slug.message}</p>}
      </div>

      <button type="submit" disabled={isSubmitting}>
        {isSubmitting ? 'Publishing…' : 'Publish'}
      </button>
    </form>
  );
}
```

## Controller: for controlled components (shadcn, etc.)

```tsx
import { useForm, Controller } from 'react-hook-form';
import { Select }              from '@/components/ui/select';
import { Switch }              from '@/components/ui/switch';

function SettingsForm() {
  const { control, handleSubmit } = useForm({
    defaultValues: { role: 'viewer', notifications: true },
  });

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      {/* Controller wraps components that don't support ref forwarding */}
      <Controller
        name="role"
        control={control}
        render={({ field }) => (
          <Select onValueChange={field.onChange} defaultValue={field.value}>
            <SelectItem value="admin">Admin</SelectItem>
            <SelectItem value="editor">Editor</SelectItem>
            <SelectItem value="viewer">Viewer</SelectItem>
          </Select>
        )}
      />

      <Controller
        name="notifications"
        control={control}
        render={({ field }) => (
          <Switch
            checked={field.value}
            onCheckedChange={field.onChange}
          />
        )}
      />
    </form>
  );
}
```

## Watch, setValue, and field arrays

```tsx
import { useForm, useFieldArray, useWatch } from 'react-hook-form';

function DynamicForm() {
  const { control, register, watch, setValue } = useForm({
    defaultValues: { items: [{ name: '', quantity: 1 }] },
  });

  const { fields, append, remove } = useFieldArray({ control, name: 'items' });

  // Watch a specific field
  const total = useWatch({ control, name: 'items' })
    .reduce((sum, item) => sum + Number(item.quantity || 0), 0);

  return (
    <form>
      {fields.map((field, i) => (
        <div key={field.id}>
          <input {...register(`items.${i}.name`)} placeholder="Item name" />
          <input {...register(`items.${i}.quantity`, { valueAsNumber: true })} type="number" />
          <button type="button" onClick={() => remove(i)}>Remove</button>
        </div>
      ))}
      <button type="button" onClick={() => append({ name: '', quantity: 1 })}>Add item</button>
      <p>Total: {total}</p>
    </form>
  );
}
```

## Server Actions integration (React 19)

```tsx
'use client';
import { useActionState } from 'react';
import { useForm }         from 'react-hook-form';
import { zodResolver }     from '@hookform/resolvers/zod';

function ServerForm() {
  const [state, formAction, isPending] = useActionState(createPost, null);
  const { register, handleSubmit, formState: { errors } } = useForm<PostForm>({
    resolver: zodResolver(PostSchema),
  });

  return (
    <form action={formAction} onSubmit={handleSubmit(() => {})}>
      <input {...register('title')} />
      {errors.title && <p>{errors.title.message}</p>}
      {state?.error?.title && <p>{state.error.title[0]}</p>}
      <button disabled={isPending}>Submit</button>
    </form>
  );
}
```

## React Hook Form vs Formik vs Zod alone

React Hook Form: fastest, uncontrolled by default, best UX for large forms, Zod integration. Formik: controlled, slower on large forms, Yup/Zod integration, more verbose. Zod alone: validation only, no form state management. Use React Hook Form for any form with more than 3-4 fields where re-render performance is visible. For simple forms, controlled `useState` is fine.
