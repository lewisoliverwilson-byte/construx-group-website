Kernel observability with no patches, no modules, production-safe overhead.

## bpftrace one-liners

```bash
# Trace all open() syscalls with filename and latency
bpftrace -e '
  tracepoint:syscalls:sys_enter_openat { @start[tid] = nsecs; @file[tid] = str(args->filename); }
  tracepoint:syscalls:sys_exit_openat  {
    $lat = (nsecs - @start[tid]) / 1000;
    printf("%-8d %-20s %d µs\n", pid, @file[tid], $lat);
    delete(@start[tid]); delete(@file[tid]);
  }
'

# Top processes by read() syscall count
bpftrace -e 'tracepoint:syscalls:sys_enter_read { @reads[comm] = count(); } interval:s:5 { print(@reads); clear(@reads); }'

# File I/O latency histogram
bpftrace -e '
  tracepoint:syscalls:sys_enter_read { @start[tid] = nsecs; }
  tracepoint:syscalls:sys_exit_read  {
    @usecs = hist((nsecs - @start[tid]) / 1000);
    delete(@start[tid]);
  }
  interval:s:10 { print(@usecs); exit(); }
'
# Output:
# @usecs:
# [0]             1240 |@@@@@@@@@@@@@@@@                          |
# [1]             3891 |@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@|

# [4, 8)           847 |@@@@@@@@@                                 |
# [8, 16)          211 |@@                                        |
# [16, 32)          43 |                                          |
# [32, 64)          12 |                                          |
```

## BCC Python — trace slow disk I/O

```python
#!/usr/bin/env python3
# trace-slow-io.py — alert on block I/O > 10ms
from bcc import BPF

bpf_text = """
#include <uapi/linux/ptrace.h>
#include <linux/blkdev.h>

BPF_HASH(start, struct request *);

void trace_start(struct pt_regs *ctx, struct request *req) {
    u64 ts = bpf_ktime_get_ns();
    start.update(&req, &ts);
}

void trace_done(struct pt_regs *ctx, struct request *req) {
    u64 *tsp = start.lookup(&req);
    if (tsp == 0) return;
    u64 delta_ms = (bpf_ktime_get_ns() - *tsp) / 1000000;
    if (delta_ms > 10) {
        bpf_trace_printk("slow I/O: %llu ms\\n", delta_ms);
    }
    start.delete(&req);
}
"""

b = BPF(text=bpf_text)
b.attach_kprobe(event="blk_start_request",  fn_name="trace_start")
b.attach_kprobe(event="blk_account_io_done", fn_name="trace_done")

print("Tracing slow I/O > 10ms... Ctrl-C to stop")
b.trace_print()
```

## Profile CPU by stack trace

```bash
# Sample all CPU stacks at 99Hz for 10 seconds — FlameGraph input
bpftrace -e '
  profile:hz:99 { @[kstack, ustack, comm] = count(); }
  interval:s:10 { exit(); }
' | flamegraph.pl > /tmp/cpu-profile.svg

# On-CPU time per function — find hot paths
bpftrace -e '
  profile:hz:99 /pid == 1234/ {
    @[func] = count();
  }
  interval:s:5 {
    print(@);
    clear(@);
  }
'
```

## Trace TCP connections

```bash
# New TCP connections with remote address and port
bpftrace -e '
  kprobe:tcp_connect {
    $sk = (struct sock *)arg0;
    printf("%-6d %-16s → %s:%d\n",
      pid, comm,
      ntop(AF_INET, $sk->__sk_common.skc_daddr),
      $sk->__sk_common.skc_dport >> 8
    );
  }
'

# TCP retransmits — useful for diagnosing packet loss
bpftrace -e 'kprobe:tcp_retransmit_skb { @retransmits[comm, pid] = count(); }'
```

## uprobe — trace userspace functions

```bash
# Trace Go HTTP handler latency in a running binary
bpftrace -e '
  uprobe:/usr/local/bin/api:net/http.(*ServeMux).ServeHTTP { @start[tid] = nsecs; }
  uretprobe:/usr/local/bin/api:net/http.(*ServeMux).ServeHTTP {
    $lat = (nsecs - @start[tid]) / 1000;
    @latency_us = hist($lat);
    delete(@start[tid]);
  }
  interval:s:10 { print(@latency_us); exit(); }
'
```

eBPF vs strace: strace intercepts every syscall with `ptrace`, adding 2-10x overhead per syscall — not suitable for production profiling. eBPF programs run in-kernel with JIT-compiled bytecode verified by the kernel's verifier, adding ~100ns overhead per event rather than microseconds. eBPF can safely trace production systems at full traffic load.
