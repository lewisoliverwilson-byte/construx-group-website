Background jobs are everywhere in production apps: sending emails, processing uploads, running AI inference, syncing external APIs. You don't want any of that in a request handler. BullMQ gives you a queue, a worker, and a scheduler — with persistence via Redis so jobs survive restarts.

## Setup

```bash
npm install bullmq ioredis
```

```ts
// lib/queue/connection.ts
import { Redis } from 'ioredis';

export const connection = new Redis({
  host: process.env.REDIS_HOST ?? 'localhost',
  port: parseInt(process.env.REDIS_PORT ?? '6379'),
  maxRetriesPerRequest: null,  // Required by BullMQ
});
```

## Defining queues

```ts
// lib/queue/queues.ts
import { Queue } from 'bullmq';
import { connection } from './connection';

export const scanQueue = new Queue('scoutr-scan', {
  connection,
  defaultJobOptions: {
    attempts: 3,
    backoff: { type: 'exponential', delay: 2000 },
    removeOnComplete: { count: 100 },
    removeOnFail: { count: 50 },
  },
});

export const emailQueue = new Queue('notifications', {
  connection,
  defaultJobOptions: {
    attempts: 5,
    backoff: { type: 'exponential', delay: 1000 },
    removeOnComplete: true,
  },
});

export const embeddingQueue = new Queue('embeddings', {
  connection,
  defaultJobOptions: {
    attempts: 2,
    removeOnComplete: { count: 200 },
    removeOnFail: { count: 100 },
  },
});
```

## Adding jobs

```ts
// From a Next.js route handler
import { scanQueue } from '@/lib/queue/queues';

export async function POST(req: Request) {
  const { userId, retailerUrl, productIds } = await req.json();

  // Add to queue — returns immediately
  const job = await scanQueue.add(
    'deep-scan',
    { userId, retailerUrl, productIds },
    {
      jobId: `scan-${userId}-${Date.now()}`,  // Deduplication
      priority: 1,  // Lower number = higher priority
      delay: 0,
    },
  );

  return Response.json({ jobId: job.id, status: 'queued' });
}
```

## Workers

```ts
// workers/scan-worker.ts
import { Worker, Job } from 'bullmq';
import { connection } from '@/lib/queue/connection';
import { db } from '@/lib/db';
import { runScan } from '@/lib/scanner';

interface ScanJobData {
  userId: string;
  retailerUrl: string;
  productIds: string[];
}

const worker = new Worker<ScanJobData>(
  'scoutr-scan',
  async (job: Job<ScanJobData>) => {
    const { userId, retailerUrl, productIds } = job.data;

    // Report progress — visible in the BullMQ dashboard
    await job.updateProgress(10);

    const results = await runScan({ retailerUrl, productIds });

    await job.updateProgress(80);

    // Persist results
    await db.insert(scanResults).values(
      results.map((r) => ({ ...r, userId, scannedAt: new Date() })),
    );

    await job.updateProgress(100);

    return { count: results.length, userId };
  },
  {
    connection,
    concurrency: 4,  // 4 parallel scans
    limiter: {
      max: 10,     // Max 10 jobs
      duration: 60000,  // Per minute
    },
  },
);

worker.on('completed', (job, result) => {
  console.log(`[scan] job ${job.id} completed — ${result.count} results`);
});

worker.on('failed', (job, error) => {
  console.error(`[scan] job ${job?.id} failed:`, error.message);
});
```

## Scheduled jobs with QueueScheduler

```ts
// workers/scheduler.ts
import { Queue } from 'bullmq';
import { connection } from '@/lib/queue/connection';

const scanQueue = new Queue('scoutr-scan', { connection });

// Repeat every hour
await scanQueue.add(
  'hourly-scan',
  { userId: 'system', retailerUrl: 'all', productIds: [] },
  {
    repeat: { pattern: '0 * * * *' },  // Cron syntax
    jobId: 'recurring-hourly-scan',
  },
);

// Delayed job — runs in 30 minutes
await scanQueue.add(
  'delayed-rescan',
  { userId: 'user_123', retailerUrl: 'amazon.co.uk', productIds: ['B001', 'B002'] },
  { delay: 30 * 60 * 1000 },
);
```

## Job events and webhooks

```ts
// Listening for job completion across the cluster
import { QueueEvents } from 'bullmq';
import { connection } from '@/lib/queue/connection';

const queueEvents = new QueueEvents('scoutr-scan', { connection });

queueEvents.on('completed', async ({ jobId, returnvalue }) => {
  const result = JSON.parse(returnvalue);

  // Notify user via WebSocket
  await notifyUser(result.userId, {
    type: 'scan-complete',
    count: result.count,
  });
});

queueEvents.on('failed', async ({ jobId, failedReason }) => {
  console.error(`[event] job ${jobId} failed: ${failedReason}`);
});
```

## Checking job status from a route handler

```ts
// app/api/scan/status/[jobId]/route.ts
import { scanQueue } from '@/lib/queue/queues';

export async function GET(req: Request, { params }: { params: { jobId: string } }) {
  const job = await scanQueue.getJob(params.jobId);

  if (!job) {
    return Response.json({ error: 'Job not found' }, { status: 404 });
  }

  const state = await job.getState();
  const progress = job.progress;

  return Response.json({
    id: job.id,
    state,
    progress,
    data: job.data,
    result: job.returnvalue,
    failedReason: job.failedReason,
    attemptsMade: job.attemptsMade,
    timestamp: job.timestamp,
  });
}
```

## Bull Board (dashboard)

```ts
// app/api/admin/queues/[[...slug]]/route.ts
import { createBullBoard } from '@bull-board/api';
import { BullMQAdapter } from '@bull-board/api/bullMQAdapter';
import { NextAdapter } from '@bull-board/nextjs';
import { scanQueue, emailQueue, embeddingQueue } from '@/lib/queue/queues';

const { hapiPlugin, router } = createBullBoard({
  queues: [
    new BullMQAdapter(scanQueue),
    new BullMQAdapter(emailQueue),
    new BullMQAdapter(embeddingQueue),
  ],
  serverAdapter: new NextAdapter(),
});

export const { GET, POST } = router;
```

The dashboard gives you job counts by state, retry controls, and a view into job data and error messages. Invaluable in production when something goes wrong.

## Deployment on Railway

Workers run as a separate Railway service from the Next.js app. The same Redis instance is shared. The worker process is long-lived — it keeps a connection to Redis and processes jobs as they arrive. In the Railway service config:

```
Start command: node dist/workers/index.js
```

One worker service per queue type is enough for most apps. Scale horizontally by increasing `concurrency` before adding more worker processes.
