Cloudflare Workers development used to require deploying to see results. Wrangler brings the runtime local.

## Install and init

```bash
npm install -D wrangler

# Create a new Worker project
npm create cloudflare@latest my-worker -- --type worker

# Or scaffold a Hono Worker
npm create cloudflare@latest my-api -- --template hono
```

## wrangler.toml

```toml
name               = "construx-api"
main               = "src/index.ts"
compatibility_date = "2031-01-01"
compatibility_flags = ["nodejs_compat"]

[vars]
ENVIRONMENT = "development"

[[kv_namespaces]]
binding  = "CACHE"
id       = "abc123def456"

[[d1_databases]]
binding      = "DB"
database_name = "construx"
database_id  = "xyz789"

[[r2_buckets]]
binding     = "ASSETS"
bucket_name = "construx-assets"

[durable_objects]
bindings = [
  { name = "RATE_LIMITER", class_name = "RateLimiter" }
]

[[migrations]]
tag  = "v1"
new_classes = ["RateLimiter"]
```

## Local development

```bash
# Start local dev server with hot reload
wrangler dev

# Start on a specific port
wrangler dev --port 8788

# Use remote bindings (real KV/D1) instead of local simulation
wrangler dev --remote
```

Wrangler starts a local HTTP server on port 8787 by default. KV, D1, R2, and Queues are simulated locally using Miniflare. The Workers runtime is exact — no surprises when you deploy.

## D1 (SQLite at the edge)

```bash
# Create a database
wrangler d1 create construx

# Run a migration
wrangler d1 execute construx --file=migrations/0001_init.sql

# Run an ad-hoc query locally
wrangler d1 execute construx --local --command="SELECT * FROM posts LIMIT 5"

# Run against production
wrangler d1 execute construx --command="SELECT COUNT(*) FROM posts"
```

## KV operations

```bash
# List keys
wrangler kv key list --namespace-id=abc123

# Get a value
wrangler kv key get --namespace-id=abc123 "session:lewis"

# Put a value
wrangler kv key put --namespace-id=abc123 "session:lewis" '{"user":"lewis"}'

# Delete a key
wrangler kv key delete --namespace-id=abc123 "session:lewis"
```

## Deploy

```bash
# Deploy to production
wrangler deploy

# Deploy to a specific environment
wrangler deploy --env staging

# Deploy and tail logs
wrangler deploy && wrangler tail
```

## Tail logs

```bash
# Stream logs from a deployed Worker
wrangler tail

# Filter by status
wrangler tail --status error

# Filter by sampling rate
wrangler tail --sampling-rate 0.1  # 10% of requests
```

`wrangler tail` streams structured JSON logs from your deployed Worker in real time. Each log entry includes the request URL, status, duration, and any `console.log` output.

## Environments

```toml
# wrangler.toml
[env.staging]
name = "construx-api-staging"
vars = { ENVIRONMENT = "staging" }

[[env.staging.d1_databases]]
binding      = "DB"
database_id  = "staging-db-id"

[env.production]
name = "construx-api"
vars = { ENVIRONMENT = "production" }
```

```bash
wrangler deploy --env staging
wrangler deploy --env production
```
