Ink renders React components to the terminal. Components are the same mental model as web React — you compose them, pass props, manage state with hooks — but instead of rendering to the DOM, they render to stdout. The output updates in place as state changes.

## Setup

```bash
npm init
npm install ink react
npm install -D typescript @types/node @types/react
```

```tsx
// src/index.tsx
import React from 'react';
import { render, Text, Box } from 'ink';

function App() {
  return (
    <Box flexDirection="column">
      <Text color="green">construx CLI</Text>
      <Text dimColor>v1.0.0</Text>
    </Box>
  );
}

render(<App />);
```

## Progress bar with live state

```tsx
import React, { useState, useEffect } from 'react';
import { render, Text, Box }          from 'ink';

interface ProgressBarProps {
  value: number;  // 0–100
  width?: number;
}

function ProgressBar({ value, width = 40 }: ProgressBarProps) {
  const filled = Math.round((value / 100) * width);
  const empty  = width - filled;
  return (
    <Text>
      <Text color="green">{'█'.repeat(filled)}</Text>
      <Text dimColor>{'░'.repeat(empty)}</Text>
      <Text> {value}%</Text>
    </Text>
  );
}

function BuildProgress() {
  const [progress, setProgress] = useState(0);
  const [done,     setDone]     = useState(false);

  useEffect(() => {
    if (progress >= 100) {
      setDone(true);
      return;
    }
    const id = setTimeout(() => setProgress((p) => Math.min(100, p + Math.random() * 8)), 100);
    return () => clearTimeout(id);
  }, [progress]);

  return (
    <Box flexDirection="column" gap={1}>
      <Text color="cyan" bold>Building construx-web...</Text>
      <ProgressBar value={progress} />
      {done && <Text color="green">✓ Build complete in 3.2s</Text>}
    </Box>
  );
}

render(<BuildProgress />);
```

## Spinner for async operations

```tsx
import { Spinner } from '@inkjs/ui';

function Fetching({ label }: { label: string }) {
  return (
    <Box gap={1}>
      <Spinner />
      <Text>{label}</Text>
    </Box>
  );
}
```

Or a manual spinner:

```tsx
const FRAMES = ['⠋', '⠙', '⠹', '⠸', '⠼', '⠴', '⠦', '⠧', '⠇', '⠏'];

function Spinner({ label }: { label: string }) {
  const [frame, setFrame] = useState(0);

  useEffect(() => {
    const id = setInterval(() => setFrame((f) => (f + 1) % FRAMES.length), 80);
    return () => clearInterval(id);
  }, []);

  return (
    <Text>
      <Text color="cyan">{FRAMES[frame]} </Text>
      {label}
    </Text>
  );
}
```

## Multi-step wizard

```tsx
import React, { useState } from 'react';
import { render, Text, Box, useInput } from 'ink';

type Step = 'name' | 'region' | 'confirm' | 'done';

function DeployWizard() {
  const [step,    setStep]    = useState<Step>('name');
  const [name,    setName]    = useState('');
  const [region,  setRegion]  = useState('eu-west-2');
  const [input,   setInput]   = useState('');

  useInput((inputChar, key) => {
    if (key.return) {
      if (step === 'name') {
        setName(input);
        setInput('');
        setStep('region');
      } else if (step === 'region') {
        setRegion(input || 'eu-west-2');
        setInput('');
        setStep('confirm');
      } else if (step === 'confirm') {
        if (input.toLowerCase() === 'y') {
          setStep('done');
        }
      }
    } else if (key.backspace || key.delete) {
      setInput((i) => i.slice(0, -1));
    } else if (inputChar) {
      setInput((i) => i + inputChar);
    }
  });

  return (
    <Box flexDirection="column" gap={1}>
      <Text color="green" bold>Construx Deploy Wizard</Text>

      {step === 'name' && (
        <Box gap={1}>
          <Text color="cyan">App name:</Text>
          <Text>{input}<Text color="cyan">▌</Text></Text>
        </Box>
      )}

      {step === 'region' && (
        <Box flexDirection="column">
          <Text color="green">✓ App: <Text bold>{name}</Text></Text>
          <Box gap={1}>
            <Text color="cyan">Region (eu-west-2):</Text>
            <Text>{input}<Text color="cyan">▌</Text></Text>
          </Box>
        </Box>
      )}

      {step === 'confirm' && (
        <Box flexDirection="column">
          <Text color="green">✓ App: <Text bold>{name}</Text></Text>
          <Text color="green">✓ Region: <Text bold>{region}</Text></Text>
          <Box gap={1}>
            <Text color="yellow">Deploy now? (y/n):</Text>
            <Text>{input}<Text color="yellow">▌</Text></Text>
          </Box>
        </Box>
      )}

      {step === 'done' && (
        <Text color="green">✓ Deploying {name} to {region}...</Text>
      )}
    </Box>
  );
}

render(<DeployWizard />);
```

## Packaging as a CLI

```json
// package.json
{
  "name": "@construx/cli",
  "bin": {
    "construx": "./dist/index.js"
  },
  "scripts": {
    "build": "tsx --build",
    "prepublishOnly": "npm run build"
  }
}
```

Add `#!/usr/bin/env node` at the top of the compiled entry point. `npm link` for local development — `construx` becomes a global command pointing to your source.

Ink is the right choice for CLIs that need live-updating output: deploy progress, streaming build logs, interactive prompts, dashboard views. For simple one-shot scripts that print and exit, plain `console.log` is simpler. Use Ink when the terminal is a rendering surface, not just a print buffer.
