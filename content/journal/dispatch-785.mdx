One file. Every developer gets the same environment. Every time.

## flake.nix for a Node.js project

```nix
{
  description = "Construx website dev environment";

  inputs = {
    nixpkgs.url     = "github:NixOS/nixpkgs/nixos-unstable";
    flake-utils.url = "github:numtide/flake-utils";
  };

  outputs = { self, nixpkgs, flake-utils }:
    flake-utils.lib.eachDefaultSystem (system:
      let
        pkgs = nixpkgs.legacyPackages.${system};
      in {
        devShells.default = pkgs.mkShell {
          buildInputs = with pkgs; [
            nodejs_22      # exact version pinned by nixpkgs revision
            bun
            postgresql_16
            redis
            openssl
            pkg-config
            turso-cli
            biome
            just           # task runner
            curl
            jq
          ];

          shellHook = ''
            echo "✓ node $(node --version)"
            echo "✓ bun $(bun --version)"
            export DATABASE_URL="postgresql://localhost:5432/construx"
            export REDIS_URL="redis://localhost:6379"
            export NODE_ENV="development"
          '';
        };
      });
}
```

```bash
# Enter the dev shell — installs everything from Nix store
nix develop

# Or use direnv so it activates automatically on cd
echo "use flake" > .envrc
direnv allow
```

## Lock file — pin exact versions

```bash
# flake.lock is committed — pins the exact nixpkgs commit
# Everyone gets bit-for-bit identical packages
cat flake.lock | jq '.nodes.nixpkgs.locked'
# {
#   "lastModified": 1731678037,
#   "narHash": "sha256-Gxxx...",
#   "owner": "NixOS",
#   "repo": "nixpkgs",
#   "rev": "7bd8..."
# }

# Update to latest nixpkgs
nix flake update

# Update a single input
nix flake lock --update-input nixpkgs
```

## Multiple shells — dev vs CI

```nix
devShells = {
  default = pkgs.mkShell {
    buildInputs = with pkgs; [
      nodejs_22 bun postgresql_16 redis
    ];
    shellHook = ''export NODE_ENV="development"'';
  };

  ci = pkgs.mkShell {
    buildInputs = with pkgs; [
      nodejs_22 bun
    ];
    shellHook = ''export NODE_ENV="test"'';
  };
};
```

```bash
# CI: use the minimal shell
nix develop .#ci --command bun test
```

## CI integration

```yaml
# .github/workflows/ci.yml
jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: DeterminateSystems/nix-installer-action@v14
      - uses: DeterminateSystems/magic-nix-cache-action@v8

      # Caches Nix store — subsequent runs are fast
      - run: nix develop .#ci --command bun install
      - run: nix develop .#ci --command bun test
```

Nix vs asdf vs mise: asdf/mise manage versions per tool but don't handle native dependencies (openssl, pkg-config) or environment variables. Nix handles the whole environment declaratively. The tradeoff: Nix has a steeper learning curve and a slow first install (subsequent installs pull from cache). The Construx default: Nix flakes for projects with native dependencies or complex environment requirements; mise for simple Node.js-only projects.
