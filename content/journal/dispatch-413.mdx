React Query solves a specific problem: UI state that comes from a server. The naive solution — fetch in useEffect, manage loading/error/data state yourself — breaks down as soon as you have multiple components that need the same data, want stale-while-revalidate behavior, or need to invalidate after mutations.

## Setup

```bash
npm install @tanstack/react-query @tanstack/react-query-devtools
```

```tsx
// app/providers.tsx
'use client';

import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { ReactQueryDevtools } from '@tanstack/react-query-devtools';
import { useState } from 'react';

export function Providers({ children }: { children: React.ReactNode }) {
  const [queryClient] = useState(
    () =>
      new QueryClient({
        defaultOptions: {
          queries: {
            staleTime: 60 * 1000,         // data is fresh for 1 minute
            gcTime: 5 * 60 * 1000,        // keep in cache for 5 minutes
            retry: 2,
            refetchOnWindowFocus: true,
          },
        },
      }),
  );

  return (
    <QueryClientProvider client={queryClient}>
      {children}
      <ReactQueryDevtools initialIsOpen={false} />
    </QueryClientProvider>
  );
}
```

```tsx
// app/layout.tsx
import { Providers } from './providers';

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <body>
        <Providers>{children}</Providers>
      </body>
    </html>
  );
}
```

## Queries

```ts
// lib/api.ts — typed API functions
export async function getUser(id: string) {
  const res = await fetch(`/api/users/${id}`);
  if (!res.ok) throw new Error(`Failed to fetch user: ${res.statusText}`);
  return res.json() as Promise<User>;
}

export async function getPosts(filters: { status?: 'draft' | 'published' }) {
  const params = new URLSearchParams(filters as Record<string, string>);
  const res = await fetch(`/api/posts?${params}`);
  if (!res.ok) throw new Error('Failed to fetch posts');
  return res.json() as Promise<Post[]>;
}
```

```tsx
// components/UserProfile.tsx
import { useQuery } from '@tanstack/react-query';
import { getUser } from '@/lib/api';

function UserProfile({ userId }: { userId: string }) {
  const { data: user, isPending, error } = useQuery({
    queryKey: ['users', userId],  // cache key — must be unique per data shape
    queryFn: () => getUser(userId),
    staleTime: 5 * 60 * 1000,   // override default: user profiles are stale after 5min
  });

  if (isPending) return <Skeleton />;
  if (error) return <ErrorState error={error} />;

  return <div>{user.name}</div>;
}
```

The `queryKey` array is the cache key. `['users', userId]` and `['users', 'abc-123']` are different entries. Change any element and the query refetches.

## Query key factories

As your app grows, manage keys centrally:

```ts
// lib/query-keys.ts
export const queryKeys = {
  users: {
    all: ['users'] as const,
    lists: () => [...queryKeys.users.all, 'list'] as const,
    list: (filters: object) => [...queryKeys.users.lists(), filters] as const,
    details: () => [...queryKeys.users.all, 'detail'] as const,
    detail: (id: string) => [...queryKeys.users.details(), id] as const,
  },
  posts: {
    all: ['posts'] as const,
    lists: () => [...queryKeys.posts.all, 'list'] as const,
    list: (filters: object) => [...queryKeys.posts.lists(), filters] as const,
    detail: (id: string) => [...queryKeys.posts.all, 'detail', id] as const,
  },
} as const;

// Usage
const { data } = useQuery({
  queryKey: queryKeys.users.detail(userId),
  queryFn: () => getUser(userId),
});
```

## Mutations with cache invalidation

```tsx
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { queryKeys } from '@/lib/query-keys';

function DeletePostButton({ postId }: { postId: string }) {
  const queryClient = useQueryClient();

  const deleteMutation = useMutation({
    mutationFn: async (id: string) => {
      const res = await fetch(`/api/posts/${id}`, { method: 'DELETE' });
      if (!res.ok) throw new Error('Delete failed');
    },
    onSuccess: () => {
      // Invalidate all post list queries — they'll refetch
      queryClient.invalidateQueries({ queryKey: queryKeys.posts.lists() });
    },
    onError: (error) => {
      toast.error(error.message);
    },
  });

  return (
    <button
      onClick={() => deleteMutation.mutate(postId)}
      disabled={deleteMutation.isPending}
    >
      {deleteMutation.isPending ? 'Deleting...' : 'Delete'}
    </button>
  );
}
```

## Optimistic updates

```tsx
const updateTitleMutation = useMutation({
  mutationFn: (data: { id: string; title: string }) =>
    fetch(`/api/posts/${data.id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ title: data.title }),
    }).then((r) => r.json()),

  onMutate: async (data) => {
    // Cancel in-flight queries so they don't overwrite optimistic update
    await queryClient.cancelQueries({ queryKey: queryKeys.posts.detail(data.id) });

    // Snapshot previous value for rollback
    const previous = queryClient.getQueryData(queryKeys.posts.detail(data.id));

    // Apply optimistic update
    queryClient.setQueryData(queryKeys.posts.detail(data.id), (old: Post) => ({
      ...old,
      title: data.title,
    }));

    return { previous };
  },

  onError: (err, data, context) => {
    // Rollback on error
    queryClient.setQueryData(queryKeys.posts.detail(data.id), context?.previous);
    toast.error('Failed to update title');
  },

  onSettled: (data, err, variables) => {
    // Always sync with server after mutation
    queryClient.invalidateQueries({ queryKey: queryKeys.posts.detail(variables.id) });
  },
});
```

## Prefetching

Load data before the user needs it:

```tsx
// Prefetch on hover — data is ready when they click
function PostLink({ post }: { post: Post }) {
  const queryClient = useQueryClient();

  const prefetch = () => {
    queryClient.prefetchQuery({
      queryKey: queryKeys.posts.detail(post.id),
      queryFn: () => getPost(post.id),
      staleTime: 30 * 1000,
    });
  };

  return (
    <Link href={`/posts/${post.id}`} onMouseEnter={prefetch}>
      {post.title}
    </Link>
  );
}
```

## Infinite queries

```tsx
const { data, fetchNextPage, hasNextPage, isFetchingNextPage } = useInfiniteQuery({
  queryKey: queryKeys.posts.lists(),
  queryFn: ({ pageParam }) =>
    fetch(`/api/posts?cursor=${pageParam}&limit=20`).then((r) => r.json()),
  initialPageParam: undefined,
  getNextPageParam: (lastPage) => lastPage.nextCursor ?? undefined,
});

const posts = data?.pages.flatMap((p) => p.items) ?? [];
```

The pattern: query key factories for cache management, mutations with invalidation for data consistency, optimistic updates for snappy UI, prefetch on hover for perceived performance. React Query eliminates almost all manual loading state management.
