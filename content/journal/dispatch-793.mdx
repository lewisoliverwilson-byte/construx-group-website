## Matrix builds

```yaml
jobs:
  test:
    strategy:
      fail-fast: false
      matrix:
        node:    [20, 22]
        os:      [ubuntu-latest, macos-latest]
        exclude:
          - node: 20
            os:   macos-latest  # Only test Node 22 on macOS
    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: ${{ matrix.node }} }
      - run: bun install && bun test
```

## Reusable workflows

```yaml
# .github/workflows/_deploy.yml — reusable
on:
  workflow_call:
    inputs:
      environment:
        required: true
        type:     string
    secrets:
      AWS_ROLE_ARN:
        required: true

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment: ${{ inputs.environment }}
    steps:
      - uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_ROLE_ARN }}
          aws-region:     eu-west-2

      - run: aws amplify start-job --app-id $APP_ID --branch-name main --job-type RELEASE

# .github/workflows/deploy-staging.yml — calls the reusable workflow
jobs:
  deploy:
    uses: ./.github/workflows/_deploy.yml
    with:
      environment: staging
    secrets:
      AWS_ROLE_ARN: ${{ secrets.STAGING_AWS_ROLE_ARN }}
```

## OIDC — no long-lived credentials

```yaml
# No AWS_ACCESS_KEY_ID or AWS_SECRET_ACCESS_KEY stored in GitHub
permissions:
  id-token: write  # Required for OIDC
  contents: read

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume:    arn:aws:iam::123456789012:role/github-actions-deploy
          role-session-name: github-actions
          aws-region:        eu-west-2
      # AWS CLI now works — token is short-lived, no secrets to rotate
      - run: aws s3 sync dist/ s3://construx-assets/
```

In AWS: create an IAM OIDC provider for `token.actions.githubusercontent.com`, then a role with a trust policy limited to your specific repo and branch.

## Artifact sharing between jobs

```yaml
jobs:
  build:
    runs-on: ubuntu-latest
    outputs:
      artifact-name: ${{ steps.build.outputs.artifact-name }}
    steps:
      - run: bun run build
        id: build
      - uses: actions/upload-artifact@v4
        with:
          name: dist-${{ github.sha }}
          path: dist/
          retention-days: 3

  deploy:
    needs: build
    runs-on: ubuntu-latest
    steps:
      - uses: actions/download-artifact@v4
        with:
          name: dist-${{ github.sha }}
          path: dist/
      - run: aws s3 sync dist/ s3://construx-assets/
```

## Caching — deterministic cache keys

```yaml
- uses: actions/cache@v4
  with:
    path: ~/.bun/install/cache
    key:  bun-${{ runner.os }}-${{ hashFiles('bun.lockb') }}
    restore-keys: |
      bun-${{ runner.os }}-

- uses: actions/cache@v4
  with:
    path: .next/cache
    key:  next-${{ runner.os }}-${{ hashFiles('bun.lockb') }}-${{ hashFiles('**/*.ts', '**/*.tsx') }}
    restore-keys: |
      next-${{ runner.os }}-${{ hashFiles('bun.lockb') }}-
      next-${{ runner.os }}-
```

## Concurrency — cancel superseded runs

```yaml
concurrency:
  group:            ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

# Result: pushing 3 commits quickly cancels the first two deployments
# Only the latest run proceeds to completion
```

## Environments with required reviewers

```yaml
# In GitHub Settings → Environments → production:
#   - Required reviewers: [lewis]
#   - Wait timer: 5 minutes
#   - Protected branches: main

jobs:
  deploy-prod:
    environment: production  # Pauses here until a reviewer approves
    runs-on: ubuntu-latest
    steps:
      - run: echo "Deploying to production..."
```

The Construx defaults: OIDC instead of long-lived credentials always; reusable workflows for anything shared across 3+ repos; concurrency groups on all deploy workflows; environment protection on production with required reviewer.
