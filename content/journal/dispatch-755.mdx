Log storage that doesn't cost a fortune. Index labels, not content — query what you label.

## Install with Docker Compose

```yaml
# docker-compose.yml
services:
  loki:
    image: grafana/loki:3.1.0
    ports: ['3100:3100']
    volumes:
      - loki-data:/loki
      - ./loki-config.yaml:/etc/loki/config.yaml
    command: -config.file=/etc/loki/config.yaml

  promtail:
    image: grafana/promtail:3.1.0
    volumes:
      - /var/log:/var/log:ro
      - ./promtail-config.yaml:/etc/promtail/config.yaml
    command: -config.file=/etc/promtail/config.yaml

  grafana:
    image: grafana/grafana:11.0.0
    ports: ['3000:3000']
    environment:
      GF_AUTH_ANONYMOUS_ENABLED: 'true'
      GF_AUTH_ANONYMOUS_ORG_ROLE: Admin

volumes:
  loki-data:
```

## Promtail config — ship container logs

```yaml
# promtail-config.yaml
server:
  http_listen_port: 9080

clients:
  - url: http://loki:3100/loki/api/v1/push

scrape_configs:
  - job_name: containers
    docker_sd_configs:
      - host: unix:///var/run/docker.sock
        refresh_interval: 5s
    relabel_configs:
      - source_labels: [__meta_docker_container_name]
        target_label: container
      - source_labels: [__meta_docker_container_label_com_docker_compose_service]
        target_label: service
      - target_label: env
        replacement: production
```

## LogQL — label filtering then content search

```logql
# All logs from construx-api in production
{env="prod", service="construx-api"}

# Only errors
{env="prod", service="construx-api"} |= "level=error"

# JSON parsing — extract structured fields
{env="prod", service="construx-api"}
  | json
  | level = "error"
  | line_format "{{.ts}} {{.msg}}"

# Rate of errors per minute
rate({env="prod"} |= "level=error" [1m])

# Top 10 slowest requests (duration field from JSON logs)
{env="prod", service="construx-api"}
  | json
  | duration > 500
  | line_format "{{.method}} {{.path}} {{.duration}}ms"
```

## Push logs from Node.js

```ts
// src/lib/logger.ts
import pino   from 'pino';
import pinoLoki from 'pino-loki';

export const logger = pino(
  { level: 'info' },
  pinoLoki({
    host:    'http://localhost:3100',
    labels: { env: 'prod', service: 'construx-api' },
    batching: true,
    interval: 5,
  }),
);

// Use throughout the app
logger.info({ userId: 'usr_abc', route: '/api/v1/workspace' }, 'Request processed');
logger.error({ err: error, userId }, 'Unhandled error');
```

## Query via logcli

```bash
# Install
brew install logcli

# Tail live logs
logcli query '{env="prod"}' --tail --addr=https://loki.construxgroup.io

# Export last hour to file
logcli query '{env="prod", service="construx-api"} |= "error"' \
  --since=1h \
  --output=jsonl \
  > errors.jsonl
```

## Loki vs Elasticsearch vs CloudWatch Logs

Loki: label-indexed, low storage cost, integrates with Grafana, LogQL is simpler than Lucene. Elasticsearch: full-text index, rich query DSL, high ingestion cost, better for complex log analysis. CloudWatch Logs: zero infra if already on AWS, per-GB pricing, slower queries. The Construx default: Loki + Grafana + Prometheus + Tempo for a complete self-hosted observability stack on Kubernetes.
