excerpt: "OpenTelemetry (OTel) is the CNCF standard for instrumenting applications to emit traces, metrics, and logs in a vendor-neutral format. You instrument your code once with the OTel SDK, then route signals to any backend: Jaeger for traces, Prometheus for metrics, Loki for logs. The OTel Collector acts as an agent: it receives signals from your applications, processes them (batch, sample, enrich), and exports to one or more backends. OTel replaces a fragmented landscape of vendor-specific SDKs (Datadog agent, New Relic SDK, Jaeger client) with a single instrumentation layer."

Instrument once. Route anywhere. Vendor lock-in is a configuration choice.

## Instrumenting a Go service

```go
// cmd/api/main.go — OTel SDK setup
package main

import (
    "context"
    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/exporters/otlp/otlptrace/otlptracehttp"
    "go.opentelemetry.io/otel/sdk/resource"
    sdktrace "go.opentelemetry.io/otel/sdk/trace"
    semconv "go.opentelemetry.io/otel/semconv/v1.26.0"
)

func initTracer(ctx context.Context) (func(), error) {
    // OTLP exporter: send traces to OTel Collector over HTTP
    exporter, err := otlptracehttp.New(ctx,
        otlptracehttp.WithEndpoint("otel-collector:4318"),
        otlptracehttp.WithInsecure(),
    )
    if err != nil {
        return nil, err
    }

    // Resource: attributes attached to every span from this service
    res := resource.NewWithAttributes(
        semconv.SchemaURL,
        semconv.ServiceName("construx-api"),
        semconv.ServiceVersion("v2.4.1"),
        semconv.DeploymentEnvironment("production"),
    )

    tp := sdktrace.NewTracerProvider(
        sdktrace.WithBatcher(exporter),           // Batch spans before sending
        sdktrace.WithResource(res),
        sdktrace.WithSampler(sdktrace.TraceIDRatioBased(0.1)), // 10% sampling
    )
    otel.SetTracerProvider(tp)

    return func() { tp.Shutdown(ctx) }, nil
}
```

## Creating spans and propagating context

```go
// internal/order/server.go
package order

import (
    "context"
    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/attribute"
    "go.opentelemetry.io/otel/codes"
)

var tracer = otel.Tracer("construx-api/internal/order")

func (s *Server) CreateOrder(ctx context.Context, req *pb.CreateOrderRequest) (*pb.Order, error) {
    // Start a span — it becomes the parent of all child spans in this context
    ctx, span := tracer.Start(ctx, "CreateOrder")
    defer span.End()

    // Add structured attributes to the span
    span.SetAttributes(
        attribute.String("customer.id",  req.CustomerId),
        attribute.Int("order.item_count", len(req.Items)),
    )

    order, err := s.db.InsertOrder(ctx, req)  // ctx propagates the trace
    if err != nil {
        span.RecordError(err)
        span.SetStatus(codes.Error, err.Error())
        return nil, err
    }

    span.SetAttributes(
        attribute.String("order.id", order.ID),
        attribute.Int64("order.total_cents", order.TotalCents),
    )
    return order, nil
}

// Database layer — child span automatically linked via context
func (r *OrderRepo) InsertOrder(ctx context.Context, req *pb.CreateOrderRequest) (*Order, error) {
    ctx, span := tracer.Start(ctx, "db.InsertOrder")
    defer span.End()

    span.SetAttributes(attribute.String("db.system", "postgresql"))
    // pgx automatically adds db.statement attribute when using otelpgx
    return r.pool.QueryRow(ctx, insertOrderSQL, ...).Scan(...)
}
```

## OTel Collector configuration

```yaml
# otel-collector-config.yaml
receivers:
  otlp:
    protocols:
      grpc:
        endpoint: "0.0.0.0:4317"    # gRPC receiver
      http:
        endpoint: "0.0.0.0:4318"    # HTTP receiver

processors:
  batch:
    timeout: 5s                     # Batch for 5s before sending
    send_batch_size: 1024           # Or when 1024 spans accumulate

  resource:
    attributes:
      - action: insert
        key: deployment.environment
        value: production

  probabilistic_sampler:
    sampling_percentage: 10         # Keep 10% of traces at collector level

  memory_limiter:
    limit_mib: 512                  # OOM protection

exporters:
  otlp/jaeger:
    endpoint: "jaeger:4317"         # Send traces to Jaeger
    tls:
      insecure: true

  prometheus:
    endpoint: "0.0.0.0:8889"        # Expose metrics for Prometheus scrape

  loki:
    endpoint: "http://loki:3100/loki/api/v1/push"

service:
  pipelines:
    traces:
      receivers:  [otlp]
      processors: [memory_limiter, resource, probabilistic_sampler, batch]
      exporters:  [otlp/jaeger]

    metrics:
      receivers:  [otlp]
      processors: [batch]
      exporters:  [prometheus]

    logs:
      receivers:  [otlp]
      processors: [batch]
      exporters:  [loki]
```

## Metrics with OTel SDK

```go
// Histogram: measure latency distributions
package order

import (
    "go.opentelemetry.io/otel/metric"
    "go.opentelemetry.io/otel"
)

var (
    meter         = otel.Meter("construx-api/internal/order")
    orderDuration metric.Float64Histogram
    orderCounter  metric.Int64Counter
)

func init() {
    orderDuration, _ = meter.Float64Histogram(
        "order.create.duration",
        metric.WithDescription("Time to create an order"),
        metric.WithUnit("ms"),
    )
    orderCounter, _ = meter.Int64Counter(
        "order.create.count",
        metric.WithDescription("Total orders created"),
    )
}

func (s *Server) CreateOrder(ctx context.Context, req *pb.CreateOrderRequest) (*pb.Order, error) {
    start := time.Now()
    defer func() {
        ms := float64(time.Since(start).Milliseconds())
        orderDuration.Record(ctx, ms, metric.WithAttributes(
            attribute.String("status", "success"),
        ))
        orderCounter.Add(ctx, 1)
    }()
    // ...
}
```

OTel vs Prometheus-only: Prometheus is pull-based and metrics-only — you can't correlate a slow request metric to the specific database query that caused it. OTel gives you the full picture: a trace shows the exact path of a slow request (which service, which database call, which query), and the trace ID links to correlated logs. The tradeoff is operational complexity: the OTel Collector is another service to run and configure. For small deployments, Prometheus + structured logs gets you 80% of the value with 20% of the setup. At ten services, the cross-service trace correlation makes OTel's overhead worthwhile.
