One schema. Types, validation, and API docs from a single source.

## Define schemas with Zod

```ts
// lib/schemas.ts
import { z } from 'zod';

export const OrderStatus = z.enum(['pending', 'processing', 'shipped', 'delivered', 'cancelled']);

export const CreateOrderSchema = z.object({
  customerId:  z.string().uuid(),
  lineItems:   z.array(z.object({
    sku:      z.string().min(1).max(64),
    quantity: z.number().int().positive().max(1000),
  })).min(1).max(50),
  shippingAddr: z.object({
    line1:    z.string().min(1).max(200),
    city:     z.string().min(1).max(100),
    postcode: z.string().regex(/^[A-Z]{1,2}\d[A-Z\d]? \d[A-Z]{2}$/i),
    country:  z.literal('GB'),
  }),
  metadata: z.record(z.string()).optional(),
});

export const OrderSchema = z.object({
  id:          z.string().uuid(),
  customerId:  z.string().uuid(),
  status:      OrderStatus,
  totalCents:  z.number().int().nonnegative(),
  createdAt:   z.string().datetime(),
  lineItems:   z.array(z.object({
    sku:         z.string(),
    quantity:    z.number().int().positive(),
    priceCents:  z.number().int().nonnegative(),
  })),
});

export type CreateOrder = z.infer<typeof CreateOrderSchema>;
export type Order       = z.infer<typeof OrderSchema>;
```

## Generate OpenAPI spec from Zod

```ts
// lib/openapi.ts
import { OpenApiGeneratorV31, OpenAPIRegistry } from '@asteasolutions/zod-to-openapi';
import { extendZodWithOpenApi } from '@asteasolutions/zod-to-openapi';
import { z } from 'zod';
import { CreateOrderSchema, OrderSchema } from './schemas';

extendZodWithOpenApi(z);

const registry = new OpenAPIRegistry();

registry.registerPath({
  method:   'post',
  path:     '/orders',
  summary:  'Create a new order',
  tags:     ['Orders'],
  request: {
    body: {
      content: { 'application/json': { schema: CreateOrderSchema } },
    },
  },
  responses: {
    201: {
      description: 'Order created',
      content: { 'application/json': { schema: OrderSchema } },
    },
    422: {
      description: 'Validation error',
      content: { 'application/json': { schema: z.object({ errors: z.array(z.object({
        field:   z.string(),
        message: z.string(),
      })) }) } },
    },
  },
});

const generator = new OpenApiGeneratorV31(registry.definitions);
export const openApiSpec = generator.generateDocument({
  openapi: '3.1.0',
  info: { title: 'Construx API', version: '1.4.2' },
  servers: [{ url: 'https://api.construxgroup.io' }],
});
```

## Validate requests in API handlers

```ts
// app/api/orders/route.ts
import { CreateOrderSchema } from '@/lib/schemas';
import { ZodError } from 'zod';

export async function POST(req: Request) {
  let body: unknown;
  try {
    body = await req.json();
  } catch {
    return Response.json({ error: 'Invalid JSON' }, { status: 400 });
  }

  const parsed = CreateOrderSchema.safeParse(body);
  if (!parsed.success) {
    const errors = parsed.error.errors.map((e) => ({
      field:   e.path.join('.'),
      message: e.message,
    }));
    return Response.json({ errors }, { status: 422 });
  }

  // parsed.data is fully typed as CreateOrder
  const order = await createOrder(parsed.data);
  return Response.json(order, { status: 201 });
}
```

## Serve the OpenAPI spec

```ts
// app/api/openapi.json/route.ts
import { openApiSpec } from '@/lib/openapi';

export function GET() {
  return Response.json(openApiSpec);
}

// Scalar UI — serve interactive API docs
// app/api/docs/route.ts

  const html = `
    <!doctype html>
    <html>
      <head><title>Construx API Docs</title></head>

        <script id="api-reference" data-url="/api/openapi.json"></script>

      </body>
    </html>

  return new Response(html, { headers: { 'Content-Type': 'text/html' } });
}
```

## Generate TypeScript client from spec

```bash
# Generate a typed client from the OpenAPI spec
npx openapi-typescript https://api.construxgroup.io/api/openapi.json -o lib/api-types.ts

# Use with openapi-fetch (type-safe fetch wrapper)
import createClient from 'openapi-fetch';
import type { paths } from '@/lib/api-types';

const client = createClient<paths>({ baseUrl: 'https://api.construxgroup.io' });

// TypeScript knows the request body type and response type exactly
const { data, error } = await client.POST('/orders', {
  body: {
    customerId:   'acc-001',
    lineItems:    [{ sku: 'CX-BOLT-M8', quantity: 4 }],
    shippingAddr: { line1: '1 London Wall', city: 'London', postcode: 'EC2Y 5AB', country: 'GB' },
  },
});
```

zod-to-openapi vs manual spec: writing OpenAPI YAML by hand means three copies of truth: the spec, the TypeScript types, and the runtime validation. They drift. zod-to-openapi collapses them to one: the Zod schema IS the spec and IS the type. The tradeoff is that generated specs can have more verbose output than hand-crafted ones — `$defs` get reused but the structure follows Zod's internal representation. For most APIs, the consistency is worth the verbosity.
