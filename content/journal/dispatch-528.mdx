Internationalisation in Next.js App Router involves two distinct problems: routing (URLs like `/en/about` vs `/fr/about`) and translations (which string to show for each locale). next-intl handles both.

## Installation

```bash
npm install next-intl
```

## File structure

```
├── messages/
│   ├── en.json
│   └── fr.json
├── middleware.ts
├── i18n.ts
└── app/
    └── [locale]/
        ├── layout.tsx
        └── page.tsx
```

## i18n configuration

```ts
// i18n.ts
import { getRequestConfig } from 'next-intl/server';

export default getRequestConfig(async ({ locale }) => ({
  messages: (await import(`./messages/${locale}.json`)).default,
}));
```

## Middleware for locale routing

```ts
// middleware.ts
import createMiddleware from 'next-intl/middleware';

export default createMiddleware({
  locales:       ['en', 'fr'],
  defaultLocale: 'en',
  localePrefix:  'as-needed',  // /about instead of /en/about for default locale
});

export const config = {
  matcher: ['/((?!api|_next|_vercel|.*\\..*).*)'],
};
```

`localePrefix: 'as-needed'` keeps URLs clean for the default locale.

## App Router layout

```tsx
// app/[locale]/layout.tsx
import { NextIntlClientProvider } from 'next-intl';
import { getMessages }            from 'next-intl/server';

export default async function LocaleLayout({
  children,
  params: { locale },
}: {
  children: React.ReactNode;
  params:   { locale: string };
}) {
  const messages = await getMessages();

  return (
    <html lang={locale}>
      <body>
        <NextIntlClientProvider messages={messages}>
          {children}
        </NextIntlClientProvider>
      </body>
    </html>
  );
}
```

## Using translations in Server Components

```tsx
// app/[locale]/page.tsx
import { useTranslations } from 'next-intl';

export default function HomePage() {
  const t = useTranslations('home.hero');

  return (
    <section>
      <h1>{t('title')}</h1>
      <p>{t('subtitle')}</p>
      <a href="/journal">{t('cta')}</a>
    </section>
  );
}
```

## Client Components

```tsx
'use client';

import { useTranslations } from 'next-intl';

export function ReadingTime({ minutes }: { minutes: number }) {
  const t = useTranslations('journal');
  return <span>{t('readingTime', { minutes })}</span>;
}
```

ICU message format handles plurals: `{minutes, plural, one {# min read} other {# min read}}`.

## Type-safe messages

Add a global type declaration for autocomplete on translation keys:

```ts
// global.d.ts
import en from './messages/en.json';

type Messages = typeof en;

declare global {
  interface IntlMessages extends Messages {}
}
```

Now `useTranslations('home.hero')` gives autocomplete on the available keys and catches typos at compile time.

## Locale switcher

```tsx
'use client';

import { useLocale }   from 'next-intl';
import { usePathname, useRouter } from 'next/navigation';

export function LocaleSwitcher() {
  const locale   = useLocale();
  const pathname = usePathname();
  const router   = useRouter();

  function switchLocale(newLocale: string) {
    // Replace the current locale prefix in the path
    const newPath = pathname.replace(`/${locale}`, `/${newLocale}`);
    router.replace(newPath);
  }

  return (
    <select value={locale} onChange={(e) => switchLocale(e.target.value)}>
      <option value="en">EN</option>
      <option value="fr">FR</option>
    </select>
  );
}
```

## When to add i18n

Adding i18n mid-project is more work than starting with it — all the string literals need extracting. For international products, set up the routing and message file structure from the start, even if you only have one locale initially. Adding a second locale later is then just a translation file, not a refactor.
