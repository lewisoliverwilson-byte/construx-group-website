You write a function that makes HTTP requests, define VU counts and ramp schedules in code, set thresholds that fail the test automatically, and run it in CI. The output is a structured summary with percentile latencies, error rates, and pass/fail per threshold. Load tests as code. Thresholds fail CI. No GUI required.

## Basic test

```js
// scripts/load-test.js
import http   from 'k6/http';
import { check, sleep } from 'k6';

export const options = {
  stages: [
    { duration: '30s',  target: 100 },  // ramp up
    { duration: '3m',   target: 100 },  // stay at 100 VUs
    { duration: '30s',  target: 0   },  // ramp down
  ],
  thresholds: {
    http_req_duration: ['p95<500'],      // 95th percentile under 500ms
    http_req_failed:   ['rate<0.01'],    // Error rate under 1%
  },
};

export default function () {
  const res = http.get('https://api.construxgroup.io/health');

  check(res, {
    'status is 200':     (r) => r.status === 200,
    'response time OK':  (r) => r.timings.duration < 500,
  });

  sleep(1);
}
```

```bash
k6 run scripts/load-test.js
```

## Authenticated requests

```js
import http   from 'k6/http';
import { check } from 'k6';

const BASE_URL = __ENV.BASE_URL || 'https://api.construxgroup.io';

// Runs once per VU — get a token
export function setup() {
  const res = http.post(`${BASE_URL}/auth/token`, JSON.stringify({
    email:    __ENV.TEST_EMAIL,
    password: __ENV.TEST_PASSWORD,
  }), { headers: { 'Content-Type': 'application/json' } });

  return { token: res.json('access_token') };
}

export default function ({ token }) {
  const headers = {
    Authorization:  `Bearer ${token}`,
    'Content-Type': 'application/json',
  };

  // Create a resource
  const create = http.post(`${BASE_URL}/projects`, JSON.stringify({ name: `Load test ${Date.now()}` }), { headers });
  check(create, { 'project created': (r) => r.status === 201 });

  const id = create.json('id');

  // Read it back
  const read = http.get(`${BASE_URL}/projects/${id}`, { headers });
  check(read, { 'project found': (r) => r.status === 200 });

  // Delete
  http.del(`${BASE_URL}/projects/${id}`, null, { headers });
}
```

export const options = {
  scenarios: {
    // Constant arrival rate — 200 req/s regardless of response time
    api_constant: {
      executor:         'constant-arrival-rate',
      rate:             200,

    // Browser-like users — new users every 2s for 3 minutes
    web_ramping: {
      executor:       'ramping-arrival-rate',
      startRate:      10,

        { duration: '2m', target: 100 },

      ],

    },
  },

## Custom metrics

```js
import { Trend, Counter, Rate } from 'k6/metrics';

const checkoutDuration = new Trend('checkout_duration');
const checkoutErrors   = new Counter('checkout_errors');
const checkoutRate     = new Rate('checkout_success_rate');

export default function () {
  const start = Date.now();
  const res   = http.post('/checkout', payload, { headers });
  const ok    = res.status === 200;

  checkoutDuration.add(Date.now() - start);
  checkoutRate.add(ok);
  if (!ok) checkoutErrors.add(1);
}

// Threshold on custom metric
export const options = {
  thresholds: {
    checkout_duration:     ['p95<2000'],
    checkout_success_rate: ['rate>0.99'],
  },
};
```

## CI integration

```yaml
# .github/workflows/load-test.yml
on:
  schedule:
    - cron: '0 4 * * *'  # Nightly at 4am

jobs:
  load-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: grafana/setup-k6-action@v1

      - run: |
          k6 run \
            --env BASE_URL=https://api.construxgroup.io \
            --env TEST_EMAIL=${{ secrets.LOAD_TEST_EMAIL }} \
            --env TEST_PASSWORD=${{ secrets.LOAD_TEST_PASSWORD }} \
            --out json=results.json \
            scripts/load-test.js

      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: k6-results
          path: results.json
```

k6 vs Locust vs JMeter: Locust uses Python, which makes it familiar for backend devs but slower per VU. JMeter is XML-based and heavy. k6 scripts are JavaScript, run efficiently, and integrate cleanly with CI. The Construx default: k6 for all load tests; run nightly against staging, and before major releases against production-like environments.
