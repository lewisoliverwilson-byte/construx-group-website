The gap between unit tests and reality is wide. You can have 100% unit test coverage and still ship a broken login flow. Playwright closes this gap by running a real browser against your actual application.

## Setup

```bash
npm install -D @playwright/test
npx playwright install chromium  # CI: only install what you need
```

```ts
// playwright.config.ts
import { defineConfig, devices } from '@playwright/test';

export default defineConfig({
  testDir: './e2e',
  timeout: 30_000,
  expect: { timeout: 5_000 },
  fullyParallel: true,
  retries: process.env.CI ? 2 : 0,
  workers: process.env.CI ? 2 : undefined,
  reporter: 'html',

  use: {
    baseURL: 'http://localhost:3000',
    trace: 'on-first-retry',
    screenshot: 'only-on-failure',
  },

  projects: [
    { name: 'chromium', use: { ...devices['Desktop Chrome'] } },
    // Add mobile only if you have mobile-specific flows to test
    { name: 'mobile-chrome', use: { ...devices['Pixel 7'] } },
  ],

  // Start dev server before tests
  webServer: {
    command: 'npm run dev',
    url: 'http://localhost:3000',
    reuseExistingServer: !process.env.CI,
    stdout: 'ignore',
    stderr: 'pipe',
  },
});
```

## Writing tests with good locators

```ts
// e2e/auth.spec.ts
import { test, expect } from '@playwright/test';

test.describe('authentication', () => {
  test('sign in with valid credentials', async ({ page }) => {
    await page.goto('/sign-in');

    // Good locators: semantic, resilient to class changes
    await page.getByLabel('Email address').fill('test@construxgroup.io');
    await page.getByLabel('Password').fill('correct-password');
    await page.getByRole('button', { name: 'Sign in' }).click();

    // Wait for navigation
    await expect(page).toHaveURL('/dashboard');
    await expect(page.getByRole('heading', { name: 'Dashboard' })).toBeVisible();
  });

  test('shows error for invalid credentials', async ({ page }) => {
    await page.goto('/sign-in');

    await page.getByLabel('Email address').fill('test@construxgroup.io');
    await page.getByLabel('Password').fill('wrong-password');
    await page.getByRole('button', { name: 'Sign in' }).click();

    await expect(page.getByText('Invalid credentials')).toBeVisible();
    await expect(page).toHaveURL('/sign-in');  // Still on sign-in page
  });

  test('redirects to sign-in from protected route', async ({ page }) => {
    await page.goto('/dashboard');
    await expect(page).toHaveURL(/sign-in\?next=\/dashboard/);
  });
});
```

## Authentication setup (avoid logging in every test)

```ts
// e2e/fixtures.ts
import { test as base, type Page } from '@playwright/test';

async function signIn(page: Page) {
  await page.goto('/sign-in');
  await page.getByLabel('Email address').fill('test@construxgroup.io');
  await page.getByLabel('Password').fill('test-password');
  await page.getByRole('button', { name: 'Sign in' }).click();
  await page.waitForURL('/dashboard');
}

// Save auth state to disk — reuse across tests
export const test = base.extend({
  page: async ({ page }, use) => {
    await signIn(page);
    await use(page);
  },
});

export { expect } from '@playwright/test';
```

```ts
// playwright.config.ts — run setup before tests
export default defineConfig({
  // ...
  globalSetup: './e2e/global-setup.ts',
});
```

```ts
// e2e/global-setup.ts
import { chromium } from '@playwright/test';
import path from 'path';

export default async function globalSetup() {
  const browser = await chromium.launch();
  const page = await browser.newPage();

  await page.goto('http://localhost:3000/sign-in');
  await page.getByLabel('Email address').fill('test@construxgroup.io');
  await page.getByLabel('Password').fill('test-password');
  await page.getByRole('button', { name: 'Sign in' }).click();
  await page.waitForURL('/dashboard');

  // Save authentication state
  await page.context().storageState({ path: 'e2e/.auth/user.json' });
  await browser.close();
}
```

```ts
// playwright.config.ts — use saved auth state for authenticated tests
projects: [
  // Setup project that runs first
  { name: 'setup', testMatch: /.*\.setup\.ts/ },
  {
    name: 'authenticated',
    use: {
      ...devices['Desktop Chrome'],
      storageState: 'e2e/.auth/user.json',
    },
    dependencies: ['setup'],
  },
],
```

## API mocking for deterministic tests

```ts
// e2e/scan.spec.ts
import { test, expect } from './fixtures';

test('shows scan results', async ({ page }) => {
  // Mock the scan API so tests don't depend on external services
  await page.route('/api/scan/results', async (route) => {
    await route.fulfill({
      status: 200,
      contentType: 'application/json',
      body: JSON.stringify({
        results: [
          { productId: 'B001', title: 'Test Product', priceDelta: 15.2, alertLevel: 'high' },
          { productId: 'B002', title: 'Another Product', priceDelta: 4.8, alertLevel: 'low' },
        ],
      }),
    });
  });

  await page.goto('/dashboard/scan');
  await expect(page.getByText('Test Product')).toBeVisible();
  await expect(page.getByText('+15.2%')).toBeVisible();
});
```

## Component testing

```ts
// e2e/components/ProductCard.spec.ts
import { test, expect } from '@playwright/experimental-ct-react';
import ProductCard from '@/components/ProductCard';

test('renders product title and price', async ({ mount }) => {
  const component = await mount(
    <ProductCard
      title="Test Product"
      price={29.99}
      currency="GBP"
      inStock={true}
    />
  );

  await expect(component.getByText('Test Product')).toBeVisible();
  await expect(component.getByText('£29.99')).toBeVisible();
  await expect(component.getByText('In stock')).toBeVisible();
});

test('shows out of stock state', async ({ mount }) => {
  const component = await mount(
    <ProductCard title="Test Product" price={29.99} currency="GBP" inStock={false} />
  );

  await expect(component.getByText('Out of stock')).toBeVisible();
  await expect(component.getByRole('button', { name: 'Add to cart' })).toBeDisabled();
});
```

## CI configuration

```yaml
# .github/workflows/e2e.yml
name: E2E Tests

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '22' }

      - name: Install dependencies
        run: npm ci

      - name: Install Playwright browsers
        run: npx playwright install --with-deps chromium

      - name: Run E2E tests
        run: npx playwright test
        env:
          DATABASE_URL: ${{ secrets.TEST_DATABASE_URL }}
          JWT_SECRET: ${{ secrets.JWT_SECRET }}

      - name: Upload test report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report
          path: playwright-report/
          retention-days: 7
```

Run E2E tests against a test database seeded with known state, not production. The `storageState` approach means login runs once per CI run, not per test — keeping the suite fast.
