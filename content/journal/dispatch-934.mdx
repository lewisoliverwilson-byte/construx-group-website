excerpt: "GraphQL is a query language for APIs where the client specifies exactly what data it needs in a single request. A GraphQL schema defines types, queries, mutations, and subscriptions; resolvers provide the data. The N+1 problem — where fetching a list of items triggers N+1 database queries — is solved by DataLoader, which batches and deduplicates requests in a single event loop tick. Subscriptions deliver real-time updates via WebSocket. GraphQL is the right choice when clients need flexible queries over a rich type system, or when multiple clients (web, mobile, CLI) have different data requirements against the same backend."

Client declares what it needs. Server returns exactly that. DataLoader solves N+1.

## Schema definition and resolvers

```graphql
# schema.graphql

type Order {
  id: ID!
  status: OrderStatus!
  totalCents: Int!
  customer: Customer!          # Nested type — DataLoader needed here
  items: [OrderItem!]!
  createdAt: DateTime!
}

type Customer {
  id: ID!
  email: String!
  name: String!
  orders(limit: Int = 10, offset: Int = 0): [Order!]!
}

type OrderItem {
  sku: String!
  quantity: Int!
  priceCents: Int!
}

enum OrderStatus {
  PENDING
  SHIPPED
  CANCELLED
}

type Query {
  order(id: ID!): Order
  orders(status: OrderStatus, limit: Int = 20): [Order!]!
  customer(id: ID!): Customer
}

type Mutation {
  createOrder(input: CreateOrderInput!): Order!
  updateOrderStatus(id: ID!, status: OrderStatus!): Order!
}

input CreateOrderInput {
  customerId: ID!
  items: [OrderItemInput!]!
}

input OrderItemInput {
  sku: String!
  quantity: Int!
}

type Subscription {
  orderStatusChanged(orderId: ID!): Order!
}
```

## Go resolvers with gqlgen

```go
// graph/resolver.go — generated by gqlgen, implement interfaces here

type Resolver struct {
    db          *database.DB
    orderLoader *dataloader.OrderLoader
}

// Query resolver
func (r *queryResolver) Order(ctx context.Context, id string) (*model.Order, error) {
    return r.db.GetOrder(ctx, id)
}

func (r *queryResolver) Orders(ctx context.Context, status *model.OrderStatus, limit *int) ([]*model.Order, error) {
    l := 20
    if limit != nil { l = *limit }

    var s *string
    if status != nil {
        str := string(*status)
        s = &str
    }
    return r.db.ListOrders(ctx, s, l)
}

// Order.Customer resolver — this is where N+1 would happen without DataLoader
func (r *orderResolver) Customer(ctx context.Context, obj *model.Order) (*model.Customer, error) {
    // DataLoader batches all Customer lookups in a single event loop tick
    return loaders.For(ctx).CustomerLoader.Load(obj.CustomerID)
}

// Mutation resolver
func (r *mutationResolver) CreateOrder(ctx context.Context, input model.CreateOrderInput) (*model.Order, error) {
    order, err := r.db.CreateOrder(ctx, database.CreateOrderParams{
        CustomerID: input.CustomerID,
        Items:      convertItems(input.Items),
    })
    if err != nil {
        return nil, err
    }

    // Publish to subscribers
    r.subscriptions.Publish(order.ID, order)
    return order, nil
}
```

## DataLoader — solving the N+1 problem

```go
// loaders/customer.go — DataLoader batches Customer queries

type CustomerLoader struct {
    // Max batch size and wait time
    loader *dataloader.Loader[string, *model.Customer]
}

func NewCustomerLoader(db *database.DB) *CustomerLoader {
    return &CustomerLoader{
        loader: dataloader.NewBatchedLoader(
            func(ctx context.Context, keys []string) []*dataloader.Result[*model.Customer] {
                // Called once per batch (one event loop tick) with ALL keys
                customers, err := db.GetCustomersByIDs(ctx, keys)
                if err != nil {
                    // Return error for all keys
                    results := make([]*dataloader.Result[*model.Customer], len(keys))
                    for i := range keys {
                        results[i] = &dataloader.Result[*model.Customer]{Error: err}
                    }
                    return results
                }

                // Map customers by ID for O(1) lookup
                customerMap := make(map[string]*model.Customer, len(customers))
                for _, c := range customers {
                    customerMap[c.ID] = c
                }

                // Return results in the same order as input keys
                results := make([]*dataloader.Result[*model.Customer], len(keys))
                for i, key := range keys {
                    results[i] = &dataloader.Result[*model.Customer]{Data: customerMap[key]}
                }
                return results
            },
            dataloader.WithBatchCapacity[string, *model.Customer](100),
            dataloader.WithWait[string, *model.Customer](1*time.Millisecond),
        ),
    }
}

// Without DataLoader — fetching 10 orders → 10+1 = 11 DB queries:
// SELECT * FROM orders LIMIT 10;
// SELECT * FROM customers WHERE id = 'cust-001';
// SELECT * FROM customers WHERE id = 'cust-002';
// ... × 10

// With DataLoader — fetching 10 orders → 2 DB queries:
// SELECT * FROM orders LIMIT 10;
// SELECT * FROM customers WHERE id IN ('cust-001', 'cust-002', ..., 'cust-010');
```

## Subscriptions (WebSocket)

```go
// Subscription resolver — real-time order status updates

func (r *subscriptionResolver) OrderStatusChanged(ctx context.Context, orderID string) (<-chan *model.Order, error) {
    ch := make(chan *model.Order, 1)

    // Subscribe to the pub/sub channel for this order
    unsubscribe := r.subscriptions.Subscribe(orderID, func(order *model.Order) {
        select {
        case ch <- order:
        default:   // Drop if channel is full (client not reading fast enough)
        }
    })

    // Clean up when the client disconnects
    go func() {
        <-ctx.Done()
        unsubscribe()
        close(ch)
    }()

    return ch, nil
}

// Client subscription query (GraphQL over WebSocket)
// subscription {
//   orderStatusChanged(orderId: "a3f9b2c1") {
//     id
//     status
//     totalCents
//   }
// }
```

## Client-side queries and fragments

```typescript
// TypeScript client with Apollo Client

// Query — client specifies exactly the fields it needs (no over/under-fetching)
const GET_ORDER = gql`
  query GetOrder($id: ID!) {
    order(id: $id) {
      id
      status
      totalCents
      customer {
        name
        email
      }
      items {
        sku
        quantity
        priceCents
      }
    }
  }
`;

const { data, loading, error } = useQuery(GET_ORDER, { variables: { id: orderId } });

// Fragment: reusable field selection (DRY)
const ORDER_FIELDS = gql`
  fragment OrderFields on Order {
    id
    status
    totalCents
    createdAt
  }
`;

// Mutation
const UPDATE_STATUS = gql`
  mutation UpdateOrderStatus($id: ID!, $status: OrderStatus!) {
    updateOrderStatus(id: $id, status: $status) {
      ...OrderFields
    }
  }
  ${ORDER_FIELDS}
`;

// Subscription (WebSocket transport)
const ORDER_SUBSCRIPTION = gql`
  subscription OrderStatusChanged($orderId: ID!) {
    orderStatusChanged(orderId: $orderId) {
      id
      status
    }
  }
`;

useSubscription(ORDER_SUBSCRIPTION, {
  variables: { orderId },
  onData: ({ data }) => {
    console.log('Order updated:', data.data.orderStatusChanged);
  },
});
```

GraphQL vs REST vs gRPC: REST is the default — simple, cacheable (HTTP semantics apply), and widely understood. The problem: versioning (v1/v2), over-fetching (GET /orders returns 50 fields, you need 3), and multiple round trips for related data (GET /orders, then GET /customers/:id). GraphQL solves over/under-fetching — the client requests exactly what it needs, and related data is fetched in one request. The trade-off: HTTP caching doesn't apply to POST queries, introspection adds attack surface, and complex schemas require tooling. gRPC is for service-to-service communication with strict contracts (proto schemas) and high throughput. The practical rule: use REST for public APIs with caching requirements, GraphQL for clients with variable field requirements (web + mobile + third-party), and gRPC for internal service communication where performance and schema evolution matter.
