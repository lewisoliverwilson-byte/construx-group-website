excerpt: "Temporal is a durable execution platform for long-running business processes. A Temporal Workflow is a function that runs to completion even across process restarts, crashes, or infrastructure failures — it never needs to manage retry logic, state persistence, or failure recovery itself. Temporal replays the workflow from its event history when a worker restarts, transparently continuing from where it left off. Activities are the side-effecting operations in a workflow (API calls, database writes, email sends) — each Activity is automatically retried with configurable backoff if it fails, and the workflow state is preserved across Activity retries. Signals deliver asynchronous external events to a running workflow (e.g., a payment confirmation, a user approval). Queries allow reading workflow state without affecting it. Child Workflows decompose complex orchestration into reusable sub-workflows. The result is that business process logic — order fulfillment, user onboarding, subscription billing, multi-step approvals — can be written as simple sequential code without explicit error handling, because Temporal's execution guarantee means the code will run to completion regardless of infrastructure failures. At Construx, Temporal orchestrates the subscription billing cycle: charge the card, if declined retry in 3 hours, if still declined send email, pause subscription after 7 days — all as straightforward Go code with zero retry boilerplate."

## Temporal workflow and activity implementation

```go
package billing

import (
    "time"

    "go.temporal.io/sdk/workflow"
    "go.temporal.io/sdk/activity"
)

// SubscriptionBillingWorkflow: durable billing cycle — runs to completion guaranteed
// Temporal replays this function from event history on worker restart
func SubscriptionBillingWorkflow(ctx workflow.Context, input BillingInput) error {
    logger := workflow.GetLogger(ctx)
    options := workflow.ActivityOptions{
        StartToCloseTimeout: 30 * time.Second,
        RetryPolicy: &temporal.RetryPolicy{
            InitialInterval:    5 * time.Second,
            BackoffCoefficient: 2.0,
            MaximumInterval:    5 * time.Minute,
            MaximumAttempts:    3,
        },
    }
    ctx = workflow.WithActivityOptions(ctx, options)

    // Attempt to charge the card
    var chargeResult ChargeResult
    err := workflow.ExecuteActivity(ctx, ChargeCardActivity, ChargeInput{
        TenantID:  input.TenantID,
        Amount:    input.Amount,
        CardToken: input.CardToken,
    }).Get(ctx, &chargeResult)

    if err != nil || chargeResult.Status == "declined" {
        // Card declined: wait 3 hours, try again
        logger.Info("charge declined, scheduling retry", "tenantID", input.TenantID)

        workflow.ExecuteActivity(ctx, SendEmailActivity, EmailInput{
            To:      input.BillingEmail,
            Subject: "Payment failed — we'll retry in 3 hours",
        }).Get(ctx, nil)

        // Sleep durably — if the worker restarts during this sleep, it resumes correctly
        workflow.Sleep(ctx, 3*time.Hour)

        // Retry the charge once more
        err = workflow.ExecuteActivity(ctx, ChargeCardActivity, ChargeInput{
            TenantID:  input.TenantID,
            Amount:    input.Amount,
            CardToken: input.CardToken,
        }).Get(ctx, &chargeResult)

        if err != nil || chargeResult.Status == "declined" {
            // Still failing: pause subscription after 7 days if no Signal arrives
            selector := workflow.NewSelector(ctx)
            timerFuture := workflow.NewTimer(ctx, 7*24*time.Hour)
            selector.AddFuture(timerFuture, func(f workflow.Future) {})

            // Listen for a payment-updated Signal (user updates card details)
            paymentUpdatedCh := workflow.GetSignalChannel(ctx, "payment-updated")
            selector.AddReceive(paymentUpdatedCh, func(c workflow.ReceiveChannel, more bool) {
                var updatedCard string
                c.Receive(ctx, &updatedCard)
                input.CardToken = updatedCard
            })

            selector.Select(ctx)   // wait for either the timer or the signal

            if timerFuture.IsReady() {
                // 7-day timer fired without a signal — pause subscription
                return workflow.ExecuteActivity(ctx, PauseSubscriptionActivity, input.TenantID).Get(ctx, nil)
            }
            // Signal received — retry charge with updated card (loop continues)
        }
    }

    // Charge succeeded — record and schedule next billing cycle
    workflow.ExecuteActivity(ctx, RecordPaymentActivity, chargeResult).Get(ctx, nil)
    return workflow.ExecuteActivity(ctx, ScheduleNextBillingActivity, input).Get(ctx, nil)
}

// ChargeCardActivity: actual stripe API call — side-effecting, retried automatically
func ChargeCardActivity(ctx context.Context, input ChargeInput) (ChargeResult, error) {
    logger := activity.GetLogger(ctx)
    activity.RecordHeartbeat(ctx, "charging stripe")   // heartbeat for long-running activity detection

    result, err := stripeClient.Charge(input.CardToken, input.Amount)
    if err != nil {
        logger.Error("stripe charge failed", "error", err)
        return ChargeResult{}, err   // Temporal retries this activity automatically
    }
    return ChargeResult{Status: result.Status, ChargeID: result.ID}, nil
}
```

```go
// Start a workflow from application code
func StartBillingCycle(client client.Client, input BillingInput) {
    options := client.StartWorkflowOptions{
        ID:        fmt.Sprintf("billing-%s-%s", input.TenantID, input.BillingPeriod),
        TaskQueue: "billing-task-queue",
        // Idempotency: if same workflow ID started twice, second call returns existing workflow
    }
    we, _ := client.ExecuteWorkflow(context.Background(), options, billing.SubscriptionBillingWorkflow, input)
    log.Printf("started workflow: %s/%s", we.GetID(), we.GetRunID())
}

// Send a Signal to a running workflow (user updated their card)
func HandleCardUpdate(client client.Client, tenantID, cardToken string) {
    client.SignalWorkflow(
        context.Background(),
        fmt.Sprintf("billing-%s-2034-01", tenantID),   // workflow ID
        "",   // run ID empty = latest run
        "payment-updated",
        cardToken,
    )
}

// Query workflow state (read-only — does not advance execution)
func GetBillingStatus(client client.Client, tenantID string) (string, error) {
    resp, _ := client.QueryWorkflow(
        context.Background(),
        fmt.Sprintf("billing-%s-2034-01", tenantID),
        "",
        "billing-status",   // query type registered in workflow
    )
    var status string
    resp.Get(&status)
    return status, nil
}
```

Temporal vs Airflow vs AWS Step Functions vs Celery vs manual retry logic: Manual retry logic (try/except + database state table) is fragile — a process crash between writing "retrying" to the database and actually retrying creates inconsistent state, and the retry logic becomes the dominant complexity in the code. Celery provides task queues with basic retry — correct for independent stateless task execution, but multi-step workflows with branching, waits, and signals require Celery beat + custom state machines that are difficult to reason about. AWS Step Functions is the managed alternative — state machine JSON/YAML definition, no code execution on the Step Functions side, visual workflow editor, but verbose state machine syntax and AWS-specific. Airflow is the dominant workflow scheduler for data engineering — excellent for recurring batch DAGs, but not designed for long-running business processes (days or weeks), inter-workflow signalling, or customer-initiated events (Signals). Temporal's key advantage is the durable execution model: workflows are just code, with no state machine syntax or explicit retry loops. The developer writes sequential Go/Java/TypeScript code; Temporal handles persistence, replay, and fault tolerance transparently. For complex business processes that span hours, days, or weeks with external events and branching paths, Temporal is the production choice.
