Rate limiting is one of those things you don't think about until something breaks. Either you get hammered by a scraper and your database falls over, or you set limits too tight and block real users. Here's the approach that avoids both.

## The sliding window algorithm

Fixed windows have an edge case: a user can send N requests at 23:59 and N more at 00:00, doubling the allowed rate at the boundary. Sliding windows don't have this problem.

```ts
// lib/rate-limit.ts
import { Redis } from '@upstash/redis';

const redis = new Redis({
  url: process.env.UPSTASH_REDIS_REST_URL!,
  token: process.env.UPSTASH_REDIS_REST_TOKEN!,
});

type RateLimitResult = {
  success: boolean;
  remaining: number;
  reset: number;  // Unix timestamp
};

export async function rateLimit(
  key: string,
  limit: number,
  windowMs: number,
): Promise<RateLimitResult> {
  const now = Date.now();
  const windowStart = now - windowMs;

  const pipeline = redis.pipeline();
  pipeline.zremrangebyscore(key, 0, windowStart);          // remove old requests
  pipeline.zadd(key, { score: now, member: now.toString() });
  pipeline.zcard(key);                                      // count in window
  pipeline.pexpire(key, windowMs);                          // auto-cleanup

  const results = await pipeline.exec();
  const count = results[2] as number;

  return {
    success: count <= limit,
    remaining: Math.max(0, limit - count),
    reset: now + windowMs,
  };
}
```

## Middleware approach

For broad rate limiting across all API routes:

```ts
// middleware.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { rateLimit } from '@/lib/rate-limit';

export async function middleware(req: NextRequest) {
  if (!req.nextUrl.pathname.startsWith('/api/')) {
    return NextResponse.next();
  }

  // Use IP for unauthenticated, user ID for authenticated
  const ip = req.ip ?? req.headers.get('x-forwarded-for') ?? 'unknown';
  const userId = req.cookies.get('session')?.value;
  const identifier = userId ?? ip;

  const key = `rate-limit:${req.nextUrl.pathname}:${identifier}`;
  const result = await rateLimit(key, 100, 60_000);  // 100 req/min

  if (!result.success) {
    return new NextResponse('Too Many Requests', {
      status: 429,
      headers: {
        'Retry-After': String(Math.ceil((result.reset - Date.now()) / 1000)),
        'X-RateLimit-Limit': '100',
        'X-RateLimit-Remaining': '0',
        'X-RateLimit-Reset': String(result.reset),
      },
    });
  }

  const response = NextResponse.next();
  response.headers.set('X-RateLimit-Limit', '100');
  response.headers.set('X-RateLimit-Remaining', String(result.remaining));
  response.headers.set('X-RateLimit-Reset', String(result.reset));
  return response;
}
```

## Per-route limits

Different endpoints have different sensitivity. An LLM endpoint at 10 req/min is very different from a static list at 1000 req/min:

```ts
// lib/rate-limit-config.ts
type RouteConfig = {
  limit: number;
  windowMs: number;
};

const routeConfig: Record<string, RouteConfig> = {
  '/api/ai/generate':    { limit: 10,   windowMs: 60_000 },
  '/api/ai/stream':      { limit: 5,    windowMs: 60_000 },
  '/api/posts':          { limit: 200,  windowMs: 60_000 },
  '/api/auth/login':     { limit: 10,   windowMs: 900_000 },  // 10 per 15 min
  '/api/auth/forgot':    { limit: 3,    windowMs: 3_600_000 }, // 3 per hour
};

const defaultConfig: RouteConfig = { limit: 60, windowMs: 60_000 };

export function getRouteConfig(pathname: string): RouteConfig {
  // Exact match first
  if (routeConfig[pathname]) return routeConfig[pathname];

  // Prefix match
  for (const [route, config] of Object.entries(routeConfig)) {
    if (pathname.startsWith(route)) return config;
  }

  return defaultConfig;
}
```

## Per-route helper for Route Handlers

For fine-grained control inside individual Route Handlers:

```ts
// lib/with-rate-limit.ts
import { headers } from 'next/headers';
import { auth } from '@/lib/auth';
import { rateLimit } from './rate-limit';

export async function withRateLimit(
  prefix: string,
  limit: number,
  windowMs: number,
): Promise<{ limited: boolean; response?: Response }> {
  const session = await auth();
  const ip = (await headers()).get('x-forwarded-for') ?? 'anonymous';
  const identifier = session?.user?.id ?? ip;

  const result = await rateLimit(`${prefix}:${identifier}`, limit, windowMs);

  if (!result.success) {
    return {
      limited: true,
      response: Response.json(
        { error: 'Rate limit exceeded', retryAfter: result.reset },
        {
          status: 429,
          headers: {
            'Retry-After': String(Math.ceil((result.reset - Date.now()) / 1000)),
          },
        },
      ),
    };
  }

  return { limited: false };
}
```

```ts
// app/api/ai/generate/route.ts
export async function POST(req: Request) {
  const { limited, response } = await withRateLimit('ai:generate', 10, 60_000);
  if (limited) return response!;

  // ... rest of handler
}
```

## Burst allowance

Some users have legitimate bursts — file uploads, batch operations. Allow a short burst above the sustained limit:

```ts
export async function rateLimitWithBurst(
  key: string,
  sustainedLimit: number,
  burstLimit: number,
  windowMs: number,
): Promise<RateLimitResult> {
  // Short burst window (10s)
  const burstResult = await rateLimit(`${key}:burst`, burstLimit, 10_000);
  if (!burstResult.success) {
    return { success: false, remaining: 0, reset: burstResult.reset };
  }

  // Sustained window
  return rateLimit(key, sustainedLimit, windowMs);
}
```

## Headers for client-side handling

Always return rate limit headers so clients can back off gracefully:

```ts
// Client-side: respect Retry-After
async function apiRequest(url: string, options?: RequestInit) {
  const response = await fetch(url, options);

  if (response.status === 429) {
    const retryAfter = Number(response.headers.get('Retry-After') ?? 60);
    console.warn(`Rate limited. Retry after ${retryAfter}s`);
    await new Promise((resolve) => setTimeout(resolve, retryAfter * 1000));
    return apiRequest(url, options);  // retry once
  }

  return response;
}
```

The rule: rate limit at the edge (middleware) for broad coverage, inside Route Handlers for endpoints that need custom limits. Always return headers so clients can adapt.
