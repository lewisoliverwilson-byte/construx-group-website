Server Actions are async functions that run on the server and can be called from Client Components. They replace API routes for most mutation operations. The key property: they work with the HTML `<form>` element directly — no JavaScript required for basic functionality.

## The basic pattern

```tsx
// app/posts/new/page.tsx — Server Component with inline action
import { redirect } from 'next/navigation';
import { db } from '@/lib/db';
import { posts } from '@/db/schema';
import { auth } from '@/auth';

export default function NewPostPage() {
  async function createPost(formData: FormData) {
    'use server';

    const session = await auth();
    if (!session) throw new Error('Unauthorized');

    const title = formData.get('title') as string;
    const content = formData.get('content') as string;

    const [post] = await db.insert(posts)
      .values({ title, content, authorId: session.user.id })
      .returning({ id: posts.id, slug: posts.slug });

    redirect(`/posts/${post.slug}`);
  }

  return (
    <form action={createPost}>
      <input name="title" placeholder="Title" required />
      <textarea name="content" placeholder="Content" required />
      <button type="submit">Publish</button>
    </form>
  );
}
```

## Validation with Zod

```tsx
// actions/create-post.ts
'use server';

import { z } from 'zod';
import { db } from '@/lib/db';
import { posts } from '@/db/schema';
import { auth } from '@/auth';

const createPostSchema = z.object({
  title: z.string().min(3).max(200),
  content: z.string().min(10),
  status: z.enum(['draft', 'published']).default('draft'),
});

type ActionState = {
  errors?: Record<string, string[]>;
  message?: string;
};

export async function createPost(
  prevState: ActionState,
  formData: FormData,
): Promise<ActionState> {
  const session = await auth();
  if (!session) return { message: 'Unauthorized' };

  const parsed = createPostSchema.safeParse({
    title: formData.get('title'),
    content: formData.get('content'),
    status: formData.get('status'),
  });

  if (!parsed.success) {
    return { errors: parsed.error.flatten().fieldErrors };
  }

  await db.insert(posts).values({
    ...parsed.data,
    authorId: session.user.id,
  });

  return { message: 'Post created' };
}
```

```tsx
// Client Component using useActionState
'use client';

import { useActionState } from 'react';
import { createPost } from '@/actions/create-post';

export function NewPostForm() {
  const [state, formAction, isPending] = useActionState(createPost, {});

  return (
    <form action={formAction}>
      <div>
        <input name="title" placeholder="Title" />
        {state.errors?.title && (
          <p className="text-red-500 text-sm">{state.errors.title[0]}</p>
        )}
      </div>
      <div>
        <textarea name="content" placeholder="Content" />
        {state.errors?.content && (
          <p className="text-red-500 text-sm">{state.errors.content[0]}</p>
        )}
      </div>
      <button type="submit" disabled={isPending}>
        {isPending ? 'Publishing...' : 'Publish'}
      </button>
      {state.message && <p>{state.message}</p>}
    </form>
  );
}
```

## File uploads

Server Actions handle `multipart/form-data` natively:

```tsx
'use server';

import { S3Client, PutObjectCommand } from '@aws-sdk/client-s3';

const s3 = new S3Client({ region: process.env.AWS_REGION });

export async function uploadAvatar(formData: FormData) {
  const session = await auth();
  if (!session) throw new Error('Unauthorized');

  const file = formData.get('avatar') as File;
  if (!file || file.size === 0) return { error: 'No file selected' };

  const allowedTypes = ['image/jpeg', 'image/png', 'image/webp'];
  if (!allowedTypes.includes(file.type)) return { error: 'Invalid file type' };

  if (file.size > 5 * 1024 * 1024) return { error: 'File too large (max 5MB)' };

  const buffer = Buffer.from(await file.arrayBuffer());
  const key = `avatars/${session.user.id}/${Date.now()}.${file.type.split('/')[1]}`;

  await s3.send(
    new PutObjectCommand({
      Bucket: process.env.AWS_S3_BUCKET!,
      Key: key,
      Body: buffer,
      ContentType: file.type,
    }),
  );

  await db.update(users)
    .set({ avatarKey: key })
    .where(eq(users.id, session.user.id));

  return { success: true, key };
}
```

## Calling actions from event handlers

Actions don't have to be triggered by forms:

```tsx
'use client';

import { deletePost } from '@/actions/delete-post';
import { useTransition } from 'react';

export function DeleteButton({ postId }: { postId: string }) {
  const [isPending, startTransition] = useTransition();

  return (
    <button
      onClick={() => {
        startTransition(() => {
          deletePost(postId);
        });
      }}
      disabled={isPending}
    >
      {isPending ? 'Deleting...' : 'Delete'}
    </button>
  );
}
```

`startTransition` wraps the action so React can keep the UI responsive while it runs.

## Revalidation after mutations

```ts
'use server';

import { revalidatePath, revalidateTag } from 'next/cache';

export async function publishPost(postId: string) {
  await db.update(posts)
    .set({ publishedAt: new Date() })
    .where(eq(posts.id, postId));

  revalidatePath('/posts');              // invalidate posts list
  revalidatePath(`/posts/${postId}`);   // invalidate this post's page
  revalidateTag('posts');               // invalidate all cached posts data
}
```

## Actions vs API routes

Use Server Actions for:
- Form submissions
- Mutations triggered by user interaction
- Operations where progressive enhancement matters

Keep API routes for:
- Webhook endpoints (Stripe, GitHub, etc.)
- Operations called by external services
- Endpoints used by mobile apps or third parties
- Streaming responses

Server Actions are not HTTP endpoints — they're called via a POST to a Next.js internal route. Third-party services can't call them.
