Columnar storage. Vectorised execution. Milliseconds over billions of rows.

## Schema design — events table

```sql
-- ClickHouse DDL — MergeTree engine with partitioning
CREATE TABLE events
(
    event_id    UUID         DEFAULT generateUUIDv4(),
    user_id     String,
    event_name  LowCardinality(String),    -- LowCardinality for repeated strings
    properties  Map(String, String),
    page_url    String,
    session_id  String,
    ip          IPv4,
    country     LowCardinality(String),
    ts          DateTime64(3),             -- millisecond precision
    date        Date MATERIALIZED toDate(ts)
)
ENGINE = MergeTree()
PARTITION BY toYYYYMM(date)               -- partition by month
ORDER BY (user_id, ts)                    -- sort key = primary index
SETTINGS index_granularity = 8192;

-- Materialized view: pre-aggregate page views per day
CREATE MATERIALIZED VIEW daily_page_views
ENGINE = SummingMergeTree()
ORDER BY (date, page_url)
AS SELECT
    date,
    page_url,
    count() AS views,
    uniqExact(user_id) AS unique_visitors
FROM events
WHERE event_name = 'pageview'
GROUP BY date, page_url;
```

## Analytical queries

```sql
-- Top pages last 7 days — scans only date, page_url, user_id columns
SELECT
    page_url,
    count()                            AS views,
    uniqExact(user_id)                 AS unique_visitors,
    round(uniqExact(session_id) / count() * 100, 1) AS bounce_rate_pct
FROM events
WHERE
    event_name = 'pageview'
    AND date >= today() - 7
GROUP BY page_url
ORDER BY views DESC
LIMIT 20;

-- Funnel analysis — how many users complete checkout
SELECT
    countIf(event_name = 'page_view'    AND page_url LIKE '%/checkout%') AS step_1,
    countIf(event_name = 'form_submit'  AND page_url LIKE '%/checkout%') AS step_2,
    countIf(event_name = 'order_created')                                 AS step_3,
    countIf(event_name = 'payment_success')                               AS step_4
FROM events
WHERE
    date >= today() - 30
    AND user_id IN (
        SELECT user_id FROM events
        WHERE event_name = 'page_view' AND page_url LIKE '%/checkout%'
        AND date >= today() - 30
    );

-- Session duration distribution
SELECT
    user_id,
    session_id,
    dateDiff('second', min(ts), max(ts)) AS session_duration_sec
FROM events
WHERE date = today() - 1
GROUP BY user_id, session_id
ORDER BY session_duration_sec DESC
LIMIT 100;
```

## ClickHouse HTTP API — ingest from Node.js

```ts
// lib/clickhouse.ts
import { createClient } from '@clickhouse/client';

const ch = createClient({
  host:     process.env.CLICKHOUSE_HOST!,
  username: process.env.CLICKHOUSE_USER!,
  password: process.env.CLICKHOUSE_PASSWORD!,
  database: 'analytics',
});

export async function trackEvent(events: Event[]) {
  await ch.insert({
    table:  'events',
    values: events,
    format: 'JSONEachRow',
  });
}

export async function queryTopPages(days: number) {
  const result = await ch.query({
    query: `
      SELECT page_url, count() AS views, uniqExact(user_id) AS uniques
      FROM events
      WHERE event_name = 'pageview' AND date >= today() - {days:UInt8}
      GROUP BY page_url ORDER BY views DESC LIMIT 20
    `,
    query_params: { days },
    format: 'JSONEachRow',
  });
  return result.json<{ page_url: string; views: number; uniques: number }[]>();
}
```

## Deploy with Docker

```yaml
services:
  clickhouse:
    image: clickhouse/clickhouse-server:24.12-alpine
    ports:
      - '8123:8123'   # HTTP API
      - '9000:9000'   # Native protocol
    volumes:
      - clickhouse-data:/var/lib/clickhouse
      - ./clickhouse/config.xml:/etc/clickhouse-server/config.d/config.xml
    ulimits:
      nofile: { soft: 262144, hard: 262144 }
    environment:
      CLICKHOUSE_DB:       analytics
      CLICKHOUSE_USER:     construx
      CLICKHOUSE_PASSWORD: $CLICKHOUSE_PASSWORD
```

ClickHouse vs PostgreSQL for analytics: PostgreSQL is row-oriented — great for OLTP (lookups by primary key, transactions). Scanning 1 billion rows for an aggregate query takes minutes. ClickHouse is column-oriented — reads only the columns it needs, compresses them 5-10x, and vectorises the computation. The crossover point: more than ~10 million rows and more than a few analytical queries per second. The Construx default: PostgreSQL for transactional data; ClickHouse for event analytics, metrics, and access logs.
