Middleware is the right place for: authentication checks, A/B testing, feature flags, locale detection, rate limiting headers, and URL rewrites. It runs once per request, before any route handler, at the CDN edge.

## Basic structure

```ts
// middleware.ts (root of the project)
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  return NextResponse.next();
}

// Control which paths the middleware runs on
export const config = {
  matcher: [
    // Match everything except static files and Next.js internals
    '/((?!_next/static|_next/image|favicon.ico|.*\\.(?:svg|png|jpg|gif|webp)).*)',
  ],
};
```

## Authentication

```ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const PROTECTED = ['/dashboard', '/api/protected', '/admin'];

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const isProtected  = PROTECTED.some((path) => pathname.startsWith(path));

  if (!isProtected) return NextResponse.next();

  const sessionCookie = request.cookies.get('session')?.value;

  if (!sessionCookie) {
    const url = request.nextUrl.clone();
    url.pathname = '/login';
    url.searchParams.set('redirect', pathname);
    return NextResponse.redirect(url);
  }

  // Verify the session (JWT decode — no DB call at edge)
  try {
    const payload = verifyJwt(sessionCookie);
    const response = NextResponse.next();
    // Pass user info to the route handler via header
    response.headers.set('x-user-id', payload.userId);
    return response;
  } catch {
    const url = request.nextUrl.clone();
    url.pathname = '/login';
    return NextResponse.redirect(url);
  }
}
```

Keep middleware fast: no database calls, no external HTTP requests. JWT verification is ideal — it's a local crypto operation.

## Locale detection

```ts
import { match }       from '@formatjs/intl-localematcher';
import Negotiator      from 'negotiator';
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const LOCALES  = ['en', 'fr', 'de', 'es'];
const DEFAULT  = 'en';

function getLocale(request: NextRequest): string {
  const headers     = { 'accept-language': request.headers.get('accept-language') ?? '' };
  const languages   = new Negotiator({ headers }).languages();
  return match(languages, LOCALES, DEFAULT);
}

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const hasLocale    = LOCALES.some((l) => pathname.startsWith(`/${l}/`) || pathname === `/${l}`);

  if (hasLocale) return NextResponse.next();

  const locale = getLocale(request);
  const url    = request.nextUrl.clone();
  url.pathname = `/${locale}${pathname}`;
  return NextResponse.redirect(url);
}
```

## A/B testing

```ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const EXPERIMENT = 'hero-v2';

export function middleware(request: NextRequest) {
  const variant = request.cookies.get(EXPERIMENT)?.value;

  if (variant) {
    return NextResponse.rewrite(
      new URL(variant === 'b' ? '/landing-b' : '/', request.url),
    );
  }

  // Assign variant and set cookie
  const newVariant = Math.random() < 0.5 ? 'a' : 'b';
  const response   = NextResponse.rewrite(
    new URL(newVariant === 'b' ? '/landing-b' : '/', request.url),
  );
  response.cookies.set(EXPERIMENT, newVariant, {
    maxAge: 60 * 60 * 24 * 30,  // 30 days
    sameSite: 'lax',
    httpOnly: true,
  });
  return response;
}

export const config = { matcher: ['/'] };
```

Rewrites change what page is served without changing the URL the user sees. Redirects change the URL. Use rewrite for A/B; use redirect for locale/auth.

## Rate limiting with Upstash Redis

```ts
import { Ratelimit } from '@upstash/ratelimit';
import { Redis }     from '@upstash/redis';
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const ratelimit = new Ratelimit({
  redis:     Redis.fromEnv(),
  limiter:   Ratelimit.slidingWindow(10, '10 s'),  // 10 req per 10s
  analytics: true,
});

export async function middleware(request: NextRequest) {
  const ip  = request.ip ?? '127.0.0.1';
  const key = `ratelimit:${ip}`;

  const { success, remaining, reset } = await ratelimit.limit(key);

  if (!success) {
    return new NextResponse('Too Many Requests', {
      status: 429,
      headers: {
        'X-RateLimit-Remaining': String(remaining),
        'X-RateLimit-Reset':     String(reset),
        'Retry-After':           String(Math.ceil((reset - Date.now()) / 1000)),
      },
    });
  }

  return NextResponse.next();
}

export const config = { matcher: ['/api/:path*'] };
```

Upstash provides serverless Redis optimised for edge — no connection pool, HTTP-based, fast from any Cloudflare/Vercel edge node.

## Security headers

```ts
export function middleware(request: NextRequest) {
  const response = NextResponse.next();

  response.headers.set('X-Frame-Options', 'DENY');
  response.headers.set('X-Content-Type-Options', 'nosniff');
  response.headers.set('Referrer-Policy', 'strict-origin-when-cross-origin');
  response.headers.set(
    'Content-Security-Policy',
    "default-src 'self'; script-src 'self' 'unsafe-inline'; style-src 'self' 'unsafe-inline';",
  );
  response.headers.set(
    'Permissions-Policy',
    'camera=(), microphone=(), geolocation=()',
  );

  return response;
}
```

Adding security headers in middleware applies them globally to every response — no need to add them to each API route or page.
