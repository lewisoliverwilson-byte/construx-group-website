REST APIs with TypeScript have a type gap: the server knows the response shape, the client has to duplicate or guess it. tRPC closes that gap by sharing the router type directly. No code generation, no schema files, no OpenAPI — just TypeScript inference across the network boundary.

## Setup

```bash
npm install @trpc/server @trpc/client @trpc/react-query @tanstack/react-query zod
```

## Router definition

```ts
// server/trpc.ts
import { initTRPC } from '@trpc/server';
import { z } from 'zod';

const t = initTRPC.create();

export const router = t.router;
export const publicProcedure = t.procedure;
```

```ts
// server/routers/products.ts
import { z } from 'zod';
import { router, publicProcedure } from '../trpc';
import { db } from '@/lib/db';

export const productsRouter = router({
  list: publicProcedure
    .input(z.object({
      category: z.string().optional(),
      limit: z.number().min(1).max(100).default(20),
      cursor: z.string().optional(),
    }))
    .query(async ({ input }) => {
      const items = await db.query.products.findMany({
        where: input.category
          ? (p, { eq }) => eq(p.category, input.category!)
          : undefined,
        limit: input.limit + 1,
        cursor: input.cursor ? { id: input.cursor } : undefined,
      });

      let nextCursor: string | undefined;
      if (items.length > input.limit) {
        const next = items.pop();
        nextCursor = next?.id;
      }

      return { items, nextCursor };
    }),

  byId: publicProcedure
    .input(z.object({ id: z.string() }))
    .query(async ({ input }) => {
      const product = await db.query.products.findFirst({
        where: (p, { eq }) => eq(p.id, input.id),
      });

      if (!product) throw new Error('Product not found');
      return product;
    }),

  update: publicProcedure
    .input(z.object({
      id: z.string(),
      title: z.string().min(1).optional(),
      price: z.number().positive().optional(),
    }))
    .mutation(async ({ input }) => {
      const { id, ...data } = input;
      await db.update(products).set(data).where(eq(products.id, id));
      return { success: true };
    }),
});
```

```tsx
// app/providers.tsx
'use client';

import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { httpBatchLink } from '@trpc/client';
import { useState } from 'react';
import { trpc } from '@/lib/trpc';

export function TRPCProvider({ children }: { children: React.ReactNode }) {
  const [queryClient] = useState(() => new QueryClient());
  const [trpcClient] = useState(() =>
    trpc.createClient({
      links: [
        httpBatchLink({
          url: '/api/trpc',
        }),
      ],
    })
  );

  return (
    <trpc.Provider client={trpcClient} queryClient={queryClient}>
      <QueryClientProvider client={queryClient}>
        {children}
      </QueryClientProvider>
    </trpc.Provider>
  );
}
```

## Using tRPC in components

```tsx
// components/ProductList.tsx
'use client';

import { trpc } from '@/lib/trpc';

export function ProductList({ category }: { category?: string }) {
  const { data, isLoading, error } = trpc.products.list.useQuery({
    category,
    limit: 20,
  });

  if (isLoading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;

  return (
    <ul>
      {data?.items.map((product) => (
        // product is fully typed — TypeScript knows the shape
        <li key={product.id}>{product.title} — £{product.price}</li>
      ))}
    </ul>
  );
}
```

```tsx
// components/ProductEditor.tsx
'use client';

import { trpc } from '@/lib/trpc';

export function ProductEditor({ productId }: { productId: string }) {
  const utils = trpc.useUtils();
  const updateMutation = trpc.products.update.useMutation({
    onSuccess: () => {
      // Invalidate cached query to refetch fresh data
      utils.products.byId.invalidate({ id: productId });
      utils.products.list.invalidate();
    },
  });

  const handleSave = (title: string, price: number) => {
    updateMutation.mutate({ id: productId, title, price });
  };

  return (
    <div>
      {/* form fields */}
      <button
        onClick={() => handleSave('Updated Title', 29.99)}
        disabled={updateMutation.isPending}
      >
        {updateMutation.isPending ? 'Saving...' : 'Save'}
      </button>
      {updateMutation.isError && (
        <p className="text-red-500">{updateMutation.error.message}</p>
      )}
    </div>
  );
}
```

## Authentication context

```ts
// server/trpc.ts
import { initTRPC, TRPCError } from '@trpc/server';
import { verifySession } from '@/lib/auth';
import type { NextRequest } from 'next/server';

interface Context {
  userId: string | null;
}

export const createContext = async (opts: { req: NextRequest }): Promise<Context> => {
  const session = await verifySession(opts.req);
  return { userId: session?.userId ?? null };
};

const t = initTRPC.context<Context>().create();

export const router = t.router;
export const publicProcedure = t.procedure;

// Throws 401 if not authenticated
export const protectedProcedure = t.procedure.use(({ ctx, next }) => {
  if (!ctx.userId) throw new TRPCError({ code: 'UNAUTHORIZED' });
  return next({ ctx: { userId: ctx.userId } });
});
```

// server/routers/scans.ts — protected routes

## Server-side calls (Server Components)

```ts
// app/products/[id]/page.tsx
import { createCaller } from '@trpc/server';
import { appRouter } from '@/server/routers/_app';

export default async function ProductPage({ params }: { params: { id: string } }) {
  const caller = appRouter.createCaller({ userId: null });
  const product = await caller.products.byId({ id: params.id });

  // product is typed — same type as the client gets
  return <h1>{product.title}</h1>;
}
```

The type shared between server and client is `AppRouter`. Change a procedure's input or output, and TypeScript will surface every mismatched call site immediately — in the server handler, the client hook, and the server-side caller. No manual sync required.
