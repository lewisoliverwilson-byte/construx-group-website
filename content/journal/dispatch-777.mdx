Thanos, Cortex, Mimir, and Grafana Cloud all accept remote write. This gives you years of retention, global query federation, and high availability without running Prometheus at scale. For most teams, Grafana Cloud's free tier or a single Thanos sidecar covers the complete observability stack."

Prometheus with unlimited retention. Remote write to Thanos, Mimir, or Grafana Cloud.

## Remote write to Grafana Cloud

```yaml
# prometheus.yml
global:
  scrape_interval:     15s
  evaluation_interval: 15s

remote_write:
  - url: https://prometheus-prod-13-prod-us-east-0.grafana.net/api/prom/push
    basic_auth:
      username: <grafana-cloud-user-id>
      password: <grafana-cloud-api-key>
    # Backpressure handling
    queue_config:
      capacity:          10000
      max_shards:        30
      max_samples_per_send: 1000
    # Metadata sending (labels exposed in Grafana)
    metadata_config:
      send: true

scrape_configs:
  - job_name: construx-api
    static_configs:
      - targets: ['localhost:9090']
```

## Expose metrics from Node.js

```ts
// src/lib/metrics.ts
import { Registry, Counter, Histogram, collectDefaultMetrics } from 'prom-client';

export const registry = new Registry();

// Collect Node.js runtime metrics (event loop lag, heap, GC)
collectDefaultMetrics({ register: registry });

export const httpRequestsTotal = new Counter({
  name:    'http_requests_total',
  help:    'Total HTTP requests',
  labelNames: ['method', 'route', 'status'],
  registers: [registry],
});

export const httpRequestDuration = new Histogram({
  name:    'http_request_duration_seconds',
  help:    'HTTP request duration in seconds',
  labelNames: ['method', 'route', 'status'],
  buckets: [0.01, 0.05, 0.1, 0.25, 0.5, 1, 2.5, 5],
  registers: [registry],
});
```

```ts
// src/app/api/metrics/route.ts — expose /metrics endpoint
import { registry } from '@/lib/metrics';

export async function GET() {
  return new Response(await registry.metrics(), {
    headers: { 'Content-Type': registry.contentType },
  });
}
```

```ts
// src/middleware.ts — instrument requests
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { httpRequestsTotal, httpRequestDuration } from '@/lib/metrics';

export function middleware(request: NextRequest) {
  const start    = Date.now();
  const response = NextResponse.next();
  const duration = (Date.now() - start) / 1000;

  httpRequestsTotal.inc({
    method: request.method,
    route:  request.nextUrl.pathname,
    status: String(response.status),
  });
  httpRequestDuration.observe(
    { method: request.method, route: request.nextUrl.pathname, status: String(response.status) },
    duration,
  );

  return response;
}
```

## Thanos sidecar (self-hosted long-term storage)

```yaml
# docker-compose.yml — Prometheus + Thanos sidecar + S3 storage
services:
  prometheus:
    image: prom/prometheus:v2.54.0
    command:
      - '--config.file=/etc/prometheus/prometheus.yml'
      - '--storage.tsdb.path=/prometheus'
      - '--storage.tsdb.min-block-duration=2h'
      - '--storage.tsdb.max-block-duration=2h'  # required for Thanos
      - '--web.enable-lifecycle'

  thanos-sidecar:
    image: thanosio/thanos:v0.36.0
    command:
      - sidecar
      - --tsdb.path=/prometheus
      - --prometheus.url=http://prometheus:9090
      - --objstore.config-file=/etc/thanos/objstore.yml
    volumes:
      - prometheus-data:/prometheus
      - ./thanos-objstore.yml:/etc/thanos/objstore.yml

  thanos-store:
    image: thanosio/thanos:v0.36.0
    command:
      - store
      - --objstore.config-file=/etc/thanos/objstore.yml
      - --http-address=0.0.0.0:19190

  thanos-query:
    image: thanosio/thanos:v0.36.0
    command:
      - query
      - --store=thanos-sidecar:19090
      - --store=thanos-store:19090
    ports: ['19192:19192']
```

```yaml
# thanos-objstore.yml — S3-compatible storage (Cloudflare R2)
type: S3
config:
  bucket:     construx-metrics
  endpoint:   <account-id>.r2.cloudflarestorage.com
  region:     auto
  access_key: <r2-access-key>
  secret_key: <r2-secret-key>
```

Thanos uploads Prometheus blocks to S3/R2 after they're sealed (2h), giving years of retention for the cost of object storage (~$0.015/GB/month on R2).

## PromQL — query across years of data

```promql
# Requests per second over last week
rate(http_requests_total[5m])

# 99th percentile latency
histogram_quantile(0.99, rate(http_request_duration_seconds_bucket[5m]))

# Error rate
rate(http_requests_total{status=~"5.."}[5m])
  /
rate(http_requests_total[5m])

# Week-over-week comparison (Thanos/Mimir extends range beyond 15d)
rate(http_requests_total[5m]) offset 1w
```
