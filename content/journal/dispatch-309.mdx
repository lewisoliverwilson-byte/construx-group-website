A multi-step form is several form screens with a shared state. The implementation challenges: where to keep the state, when to validate, how to handle back/forward navigation, and how to submit only when all steps are complete.

## State management

The simplest approach: a parent component holds all form state and passes it down as props with update callbacks:

```tsx
// app/onboarding/page.tsx
'use client';

import { useState } from 'react';
import { StepOne } from './steps/StepOne';
import { StepTwo } from './steps/StepTwo';
import { StepThree } from './steps/StepThree';

interface FormData {
  // Step 1
  name: string;
  email: string;
  // Step 2
  company: string;
  role: string;
  teamSize: string;
  // Step 3
  useCase: string;
  budget: string;
}

const INITIAL_DATA: FormData = {
  name: '', email: '', company: '', role: '',
  teamSize: '', useCase: '', budget: '',
};

const STEPS = ['About you', 'Your team', 'Use case'];

export default function OnboardingPage() {
  const [step, setStep] = useState(0);
  const [data, setData] = useState<FormData>(INITIAL_DATA);
  const [errors, setErrors] = useState<Partial<FormData>>({});

  const updateField = (field: keyof FormData, value: string) => {
    setData((d) => ({ ...d, [field]: value }));
    // Clear error when field is updated
    if (errors[field]) setErrors((e) => ({ ...e, [field]: undefined }));
  };

  const validateStep = (currentStep: number): boolean => {
    const newErrors: Partial<FormData> = {};
    if (currentStep === 0) {
      if (!data.name.trim()) newErrors.name = 'Required';
      if (!/\S+@\S+\.\S+/.test(data.email)) newErrors.email = 'Invalid email';
    }
    if (currentStep === 1) {
      if (!data.company.trim()) newErrors.company = 'Required';
      if (!data.role) newErrors.role = 'Required';
    }
    if (currentStep === 2) {
      if (!data.useCase) newErrors.useCase = 'Required';
    }
    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleNext = () => {
    if (validateStep(step)) {
      setStep((s) => Math.min(s + 1, STEPS.length - 1));
    }
  };

  const handleBack = () => {
    setStep((s) => Math.max(s - 1, 0));
  };

  const handleSubmit = async () => {
    if (!validateStep(step)) return;
    // All data is in `data` — submit it
    await fetch('/api/onboard', { method: 'POST', body: JSON.stringify(data) });
  };

  const stepProps = { data, updateField, errors };

  return (
    <div className="max-w-lg mx-auto">
      <ProgressBar steps={STEPS} current={step} />

      {step === 0 && <StepOne {...stepProps} />}
      {step === 1 && <StepTwo {...stepProps} />}
      {step === 2 && <StepThree {...stepProps} />}

      <div className="flex justify-between mt-8">
        {step > 0 && (
          <button onClick={handleBack}>Back</button>
        )}
        {step < STEPS.length - 1 ? (
          <button onClick={handleNext}>Continue</button>
        ) : (
          <button onClick={handleSubmit}>Submit</button>
        )}
      </div>
    </div>
  );
}
```

## Individual step components

Each step receives the shared state and renders only its fields:

```tsx
// steps/StepOne.tsx
interface StepProps {
  data: FormData;
  updateField: (field: keyof FormData, value: string) => void;
  errors: Partial<FormData>;
}

export function StepOne({ data, updateField, errors }: StepProps) {
  return (
    <div className="space-y-4">
      <div>
        <label className="font-mono text-[10px] uppercase tracking-wider text-text-dim">
          Your name
        </label>
        <input
          value={data.name}
          onChange={(e) => updateField('name', e.target.value)}
          className={errors.name ? 'border-red-500' : ''}
        />
        {errors.name && <p className="text-xs text-red-400 mt-1">{errors.name}</p>}
      </div>

      <div>
        <label className="font-mono text-[10px] uppercase tracking-wider text-text-dim">
          Email address
        </label>
        <input
          type="email"
          value={data.email}
          onChange={(e) => updateField('email', e.target.value)}
          className={errors.email ? 'border-red-500' : ''}
        />
        {errors.email && <p className="text-xs text-red-400 mt-1">{errors.email}</p>}
      </div>
    </div>
  );
}
```

Steps are thin — they render fields and call `updateField`. Validation lives in the parent.

## Progress bar component

```tsx
function ProgressBar({ steps, current }: { steps: string[]; current: number }) {
  return (
    <div className="mb-8">
      <div className="flex items-center gap-2">
        {steps.map((label, i) => (
          <div key={label} className="flex items-center gap-2 flex-1">
            <div
              className={cn(
                'h-6 w-6 rounded-full flex items-center justify-center font-mono text-[10px] font-bold flex-shrink-0 transition-all',
                i < current ? 'bg-construx text-black' :
                i === current ? 'border-2 border-construx text-construx' :
                'border border-border text-text-dim'
              )}
            >
              {i < current ? '✓' : i + 1}
            </div>
            <span className={cn('text-sm', i <= current ? 'text-text-base' : 'text-text-dim')}>
              {label}
            </span>
            {i < steps.length - 1 && (
              <div className={cn('flex-1 h-px', i < current ? 'bg-construx' : 'bg-border')} />
            )}
          </div>
        ))}
      </div>
    </div>
  );
}
```

## Persisting across page refreshes

For long forms where users might navigate away, persist state to `localStorage`:

```ts
function usePersistedFormState(key: string, initial: FormData) {
  const [data, setData] = useState<FormData>(() => {
    if (typeof window === 'undefined') return initial;
    const stored = localStorage.getItem(key);
    return stored ? JSON.parse(stored) : initial;
  });

  const updateData = (updates: Partial<FormData>) => {
    setData((d) => {
      const next = { ...d, ...updates };
      localStorage.setItem(key, JSON.stringify(next));
      return next;
    });
  };

  const clearData = () => {
    localStorage.removeItem(key);
    setData(initial);
  };

  return { data, updateData, clearData };
}
```

Clear the persisted state on successful submission so the next user (or next session) starts fresh.

## The submission pattern

Submit on the final step, not per-step. Partial submissions create data integrity issues. If a step fails mid-way, you have an incomplete record.

If steps genuinely need individual saves (e.g., a setup wizard that configures live resources), make each step its own endpoint with idempotent semantics — the user can re-run any step without creating duplicates.
