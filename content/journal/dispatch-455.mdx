TypeScript types are erased at runtime. Your `User` type doesn't exist when the API response lands — it's just JSON that you told the compiler to trust. Zod bridges this: define a schema, get the TypeScript type *and* a runtime parser that throws on invalid data.

## Setup

```bash
npm install zod
```

## The core pattern

```ts
import { z } from 'zod';

// Define schema — TypeScript type is inferred automatically
const UserSchema = z.object({
  id: z.string().cuid(),
  email: z.string().email(),
  name: z.string().min(2).max(100),
  role: z.enum(['admin', 'user', 'viewer']),
  age: z.number().int().min(18).optional(),
  createdAt: z.coerce.date(),  // Accepts string, number, or Date — coerces to Date
});

// Extract TypeScript type
type User = z.infer<typeof UserSchema>;

// Parse (throws on invalid data)
const user = UserSchema.parse(rawData);

// Safe parse (returns { success, data } | { success, error })
const result = UserSchema.safeParse(rawData);
if (result.success) {
  console.log(result.data.email);  // TypeScript knows the type
} else {
  console.error(result.error.issues);  // Detailed error info
}
```

## API request validation

```ts
// app/api/products/route.ts
import { z } from 'zod';

const CreateProductSchema = z.object({
  title: z.string().min(1, 'Title is required').max(200),
  description: z.string().min(10).max(5000),
  price: z.number().positive('Price must be positive').max(99999),
  currency: z.enum(['GBP', 'USD', 'EUR']).default('GBP'),
  tags: z.array(z.string().max(30)).max(10).default([]),
  published: z.boolean().default(false),
});

type CreateProduct = z.infer<typeof CreateProductSchema>;

export async function POST(req: Request) {
  const body = await req.json().catch(() => null);

  if (!body) {
    return Response.json({ error: 'Invalid JSON body' }, { status: 400 });
  }

  const result = CreateProductSchema.safeParse(body);

  if (!result.success) {
    const fieldErrors = result.error.flatten().fieldErrors;
    return Response.json(
      { error: 'Validation failed', fields: fieldErrors },
      { status: 422 },
    );
  }

  const product = result.data;  // TypeScript knows the exact shape
  // product.price is number, product.currency is 'GBP' | 'USD' | 'EUR', etc.
  
  const created = await db.insert(products).values(product).returning();
  return Response.json(created[0], { status: 201 });
}
```

## Shared schemas between client and server

```ts
// lib/schemas/product.ts — shared, import in both client and server
import { z } from 'zod';

export const ProductSchema = z.object({
  id: z.string().uuid(),
  title: z.string(),
  price: z.number(),
  currency: z.enum(['GBP', 'USD', 'EUR']),
  published: z.boolean(),
  createdAt: z.coerce.date(),
});

export const CreateProductSchema = ProductSchema.omit({ id: true, createdAt: true });
export const UpdateProductSchema = CreateProductSchema.partial();  // All fields optional

export type Product = z.infer<typeof ProductSchema>;
export type CreateProduct = z.infer<typeof CreateProductSchema>;
export type UpdateProduct = z.infer<typeof UpdateProductSchema>;
```

## Environment variable validation

```ts
// lib/env.ts — run at startup, fail fast if config is wrong
import { z } from 'zod';

const EnvSchema = z.object({
  DATABASE_URL: z.string().url(),
  JWT_SECRET: z.string().min(32, 'JWT_SECRET must be at least 32 characters'),
  RESEND_API_KEY: z.string().startsWith('re_'),
  ANTHROPIC_API_KEY: z.string().startsWith('sk-ant-'),
  NEXT_PUBLIC_SITE_URL: z.string().url().optional(),
  NODE_ENV: z.enum(['development', 'test', 'production']).default('development'),
  PORT: z.coerce.number().int().positive().default(3000),
});

const result = EnvSchema.safeParse(process.env);

if (!result.success) {
  console.error('❌ Invalid environment variables:');
  console.error(result.error.flatten().fieldErrors);
  process.exit(1);
}

export const env = result.data;
// env.PORT is number, env.NODE_ENV is 'development' | 'test' | 'production', etc.
```

## Transformations and preprocessing

```ts
const SearchParamsSchema = z.object({
  q: z.string().trim().min(1).optional(),
  page: z.coerce.number().int().positive().default(1),
  limit: z.coerce.number().int().min(1).max(100).default(20),
  sort: z.enum(['asc', 'desc']).default('desc'),
  tags: z.union([
    z.string().transform((s) => s.split(',')),  // 'a,b,c' → ['a', 'b', 'c']
    z.array(z.string()),
  ]).default([]),
});

export async function GET(req: Request) {
  const url = new URL(req.url);
  const rawParams = Object.fromEntries(url.searchParams.entries());

  const params = SearchParamsSchema.parse(rawParams);
  // params.page is number, params.tags is string[], params.limit is 1-100
}
```

## Discriminated unions for event types

```ts
const WebhookEventSchema = z.discriminatedUnion('type', [
  z.object({
    type: z.literal('payment.succeeded'),
    data: z.object({
      paymentIntentId: z.string(),
      amount: z.number(),
      currency: z.string(),
    }),
  }),
  z.object({
    type: z.literal('payment.failed'),
    data: z.object({
      paymentIntentId: z.string(),
      errorCode: z.string(),
    }),
  }),
  z.object({
    type: z.literal('subscription.cancelled'),
    data: z.object({
      subscriptionId: z.string(),
      cancelledAt: z.coerce.date(),
    }),
  }),
]);

type WebhookEvent = z.infer<typeof WebhookEventSchema>;

function handleWebhook(event: WebhookEvent) {
  switch (event.type) {
    case 'payment.succeeded':
      // TypeScript knows event.data.amount exists here
      console.log(`Payment of ${event.data.amount}`);
      break;
    case 'payment.failed':
      // TypeScript knows event.data.errorCode exists here
      console.log(`Failed: ${event.data.errorCode}`);
      break;
  }
}
```

## Custom validators

```ts
const SlugSchema = z.string()
  .min(3)
  .max(60)
  .regex(/^[a-z0-9-]+$/, 'Slug must be lowercase alphanumeric with hyphens')
  .refine((s) => !s.startsWith('-') && !s.endsWith('-'), {
    message: 'Slug cannot start or end with a hyphen',
  });

const PasswordSchema = z.string()
  .min(12, 'Password must be at least 12 characters')
  .refine((p) => /[A-Z]/.test(p), 'Password must contain an uppercase letter')
  .refine((p) => /[0-9]/.test(p), 'Password must contain a number')
  .refine((p) => /[^a-zA-Z0-9]/.test(p), 'Password must contain a special character');

const ConfirmPasswordSchema = z
  .object({
    password: PasswordSchema,
    confirmPassword: z.string(),
  })
  .refine((data) => data.password === data.confirmPassword, {
    message: "Passwords don't match",
    path: ['confirmPassword'],  // Field the error attaches to
  });
```

## Integration with React Hook Form

```tsx
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { CreateProductSchema, type CreateProduct } from '@/lib/schemas/product';

export default function ProductForm() {
  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm<CreateProduct>({
    resolver: zodResolver(CreateProductSchema),
    defaultValues: { currency: 'GBP', published: false, tags: [] },
  });

  const onSubmit = async (data: CreateProduct) => {
    await fetch('/api/products', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    });
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input {...register('title')} />
      {errors.title && <p>{errors.title.message}</p>}

      <input {...register('price', { valueAsNumber: true })} type="number" />
      {errors.price && <p>{errors.price.message}</p>}

      <button type="submit" disabled={isSubmitting}>
        Create product
      </button>
    </form>
  );
}
```

Define the schema once. Get client-side form validation, server-side API validation, TypeScript types, and shared contract between the two — all from the same source. No duplication, no drift.
