TypeScript all the way down. No HCL. No YAML. Real types.

## Install and configure

```bash
# Install Pulumi CLI
brew install pulumi/tap/pulumi

# AWS credentials
export AWS_PROFILE=construx-prod

# Create a new TypeScript project
pulumi new typescript --name construx-infra --stack prod
```

## Stack definition

```ts
// index.ts
import * as aws      from '@pulumi/aws';
import * as awsx     from '@pulumi/awsx';
import * as pulumi   from '@pulumi/pulumi';

const cfg   = new pulumi.Config();
const stage = pulumi.getStack();  // 'prod' | 'staging'

// VPC with public + private subnets across 3 AZs
const vpc = new awsx.ec2.Vpc('construx-vpc', {
  cidrBlock:          '10.0.0.0/16',
  numberOfAvailabilityZones: 3,
  subnetStrategy:     'Auto',
  tags: { Name: `construx-${stage}`, Environment: stage },
});

// RDS PostgreSQL — private subnet only
const dbSg = new aws.ec2.SecurityGroup('db-sg', {
  vpcId:  vpc.vpcId,
  ingress: [{ protocol: 'tcp', fromPort: 5432, toPort: 5432, cidrBlocks: ['10.0.0.0/16'] }],
  egress:  [{ protocol: '-1', fromPort: 0, toPort: 0, cidrBlocks: ['0.0.0.0/0'] }],
});

const db = new aws.rds.Instance('construx-db', {
  engine:              'postgres',
  engineVersion:       '16.4',
  instanceClass:       cfg.require('dbInstanceClass'),  // from Pulumi config
  allocatedStorage:    100,
  dbName:              'construx',
  username:            'construx_admin',
  password:            cfg.requireSecret('dbPassword'),
  dbSubnetGroupName:   new aws.rds.SubnetGroup('db-subnet-group', {
    subnetIds: vpc.privateSubnetIds,
  }).name,
  vpcSecurityGroupIds: [dbSg.id],
  skipFinalSnapshot:   stage !== 'prod',
  tags: { Environment: stage },
});

// ECS Fargate — API service
const cluster = new aws.ecs.Cluster('construx-cluster', {
  tags: { Environment: stage },
});

const apiService = new awsx.ecs.FargateService('api', {
  cluster: cluster.arn,
  taskDefinitionArgs: {
    containers: {
      api: {
        image:  cfg.require('apiImage'),
        cpu:    512,
        memory: 1024,
        portMappings: [{ containerPort: 3000, hostPort: 3000, protocol: 'tcp' }],
        environment: [
          { name: 'DATABASE_URL', value: pulumi.interpolate`postgresql://${db.username}:${cfg.requireSecret('dbPassword')}@${db.endpoint}/construx` },
          { name: 'NODE_ENV',     value: stage },
        ],
      },
    },
  },
  networkConfiguration: {
    subnets:        vpc.privateSubnetIds,
    assignPublicIp: false,
  },
});

// Outputs — available via pulumi stack output
export const vpcId        = vpc.vpcId;
export const dbEndpoint   = db.endpoint;
export const clusterArn   = cluster.arn;
```

## Component abstraction

```ts
// components/construx-service.ts
import * as aws    from '@pulumi/aws';
import * as awsx   from '@pulumi/awsx';
import * as pulumi from '@pulumi/pulumi';

interface ConstructxServiceArgs {
  image:   pulumi.Input<string>;
  cpu:     number;
  memory:  number;
  env:     Record<string, pulumi.Input<string>>;
  cluster: pulumi.Input<string>;
  subnets: pulumi.Input<string>[];
}

// Reusable component — any service in the stack uses this
export class ConstruxService extends pulumi.ComponentResource {
  public readonly url: pulumi.Output<string>;

  constructor(name: string, args: ConstructxServiceArgs, opts?: pulumi.ComponentResourceOptions) {
    super('construx:index:Service', name, {}, opts);

    const lb = new awsx.lb.ApplicationLoadBalancer(`${name}-lb`, {}, { parent: this });

    new awsx.ecs.FargateService(`${name}-svc`, {
      cluster: args.cluster,
      taskDefinitionArgs: {
        containers: {
          [name]: {
            image:        args.image,
            cpu:          args.cpu,
            memory:       args.memory,
            portMappings: [lb.defaultTargetGroup],
            environment:  Object.entries(args.env).map(([name, value]) => ({ name, value })),
          },
        },
      },
    }, { parent: this });

    this.url = lb.loadBalancer.dnsName;
    this.registerOutputs({ url: this.url });
  }
}
```

## Workflow

```bash
# Preview changes — shows what will be created/updated/deleted
pulumi preview

# Apply changes
pulumi up

# Output a stack value
pulumi stack output dbEndpoint

# Destroy (with confirmation)
pulumi destroy
```

Pulumi vs Terraform: Terraform has HCL (a purpose-built language), a massive provider ecosystem, and excellent state management. Pulumi uses real languages — better for teams who find HCL limiting, or who want to test infrastructure logic. The Construx default: Pulumi for new AWS infrastructure where TypeScript composition matters; Terraform when a provider or module only exists for Terraform.
