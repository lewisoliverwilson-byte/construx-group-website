Decorators annotate code with metadata and transform it at definition time. A method decorator receives the method and can wrap it. A class decorator receives the constructor and can modify the class. The result is code that separates concerns cleanly — the core logic stays uncluttered, the cross-cutting behaviour (logging, auth, caching) lives in decorators.

## Enabling decorators (TypeScript 5.0+)

```json
// tsconfig.json — no experimentalDecorators needed for stage-3
{
  "compilerOptions": {
    "target": "ES2022",
    "lib":    ["ES2022"]
  }
}
```

Stage-3 decorators (TypeScript 5.0+) are the standard. If you see `experimentalDecorators: true` in older configs, that's the legacy syntax — the two are incompatible.

## Class decorator

```ts
function Singleton<T extends new (...args: any[]) => {}>(Base: T, context: ClassDecoratorContext) {
  let instance: InstanceType<T> | null = null;

  return class extends Base {
    constructor(...args: any[]) {
      if (instance) return instance;
      super(...args);
      instance = this as unknown as InstanceType<T>;
    }
  };
}

@Singleton
class Database {
  private connection: Connection;

  constructor() {
    this.connection = createConnection();
  }
}

const db1 = new Database();
const db2 = new Database();
console.log(db1 === db2);  // true
```

## Method decorator: logging

```ts
function Log(target: unknown, context: ClassMethodDecoratorContext) {
  const name = String(context.name);

  return function (this: unknown, ...args: unknown[]) {
    console.time(name);
    const result = (target as Function).call(this, ...args);
    if (result instanceof Promise) {
      return result.finally(() => console.timeEnd(name));
    }
    console.timeEnd(name);
    return result;
  };
}

class OrderService {
  @Log
  async processOrder(orderId: string) {
    await chargeCustomer(orderId);
    await fulfillOrder(orderId);
    await sendConfirmation(orderId);
  }
}
```

## Method decorator: memoisation

```ts
function Memo(target: unknown, context: ClassMethodDecoratorContext) {
  const cache = new Map<string, unknown>();

  return function (this: unknown, ...args: unknown[]) {
    const key = JSON.stringify(args);
    if (cache.has(key)) return cache.get(key);
    const result = (target as Function).call(this, ...args);
    cache.set(key, result);
    return result;
  };
}

class PricingService {
  @Memo
  calculatePrice(productId: string, quantity: number): number {
    // Expensive calculation
    return basePrice(productId) * quantity * discount(quantity);
  }
}
```

## Accessor decorator: validated property

```ts
function NonEmpty(target: unknown, context: ClassAccessorDecoratorContext) {
  return {
    get(this: unknown): string {
      return (target as any).get.call(this);
    },
    set(this: unknown, value: string) {
      if (!value.trim()) throw new Error(`${String(context.name)} cannot be empty`);
      (target as any).set.call(this, value);
    },
  };
}

class User {
  @NonEmpty accessor email: string = '';
  @NonEmpty accessor name: string  = '';
}

const user = new User();
user.email = '';  // throws: "email cannot be empty"
```

## Practical pattern: route registration

```ts
const routes: Array<{ method: string; path: string; handler: Function }> = [];

function Get(path: string) {
  return function (target: unknown, context: ClassMethodDecoratorContext) {
    context.addInitializer(function (this: unknown) {
      routes.push({ method: 'GET', path, handler: target as Function });
    });
  };
}

function Post(path: string) {
  return function (target: unknown, context: ClassMethodDecoratorContext) {
    context.addInitializer(function (this: unknown) {
      routes.push({ method: 'POST', path, handler: target as Function });
    });
  };
}

class PostsController {
  @Get('/posts')
  async list(req: Request): Promise<Response> {
    const posts = await db.post.findMany();
    return Response.json(posts);
  }

  @Post('/posts')
  async create(req: Request): Promise<Response> {
    const data = await req.json();
    const post  = await db.post.create({ data });
    return Response.json(post, { status: 201 });
  }
}

// Register all routes after class definition
for (const { method, path, handler } of routes) {
  app.on(method, path, handler);
}
```

## Decorator metadata (stage-3)

```ts
import 'reflect-metadata';

function Validate(schema: ZodSchema) {
  return function (target: unknown, context: ClassMethodDecoratorContext) {
    context.metadata ??= {};
    (context.metadata as Record<string, unknown>)[String(context.name)] = schema;
  };
}

// Read metadata later
const metadata = Symbol.metadata;
const schemas  = MyClass[metadata];  // all @Validate schemas indexed by method name
```

Decorator metadata uses `Symbol.metadata` — a standard proposal. It lets decorators attach typed metadata to classes that middleware and frameworks can read.

## When to use decorators

Decorators fit cross-cutting concerns that need to be applied to many methods or classes without changing their core logic. Good candidates: logging, caching, auth checks, validation, ORM column definitions, dependency injection. Poor candidates: complex business logic, operations with side effects that should be explicit.
