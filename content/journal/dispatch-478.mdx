Webhooks flip the polling model: instead of your server asking "did anything happen?", the third-party service tells you when it does. The challenge is that the POST arrives with no retry guarantee, the payload could be malicious, and the same event can arrive twice. Handling this correctly is a small but non-trivial engineering problem.

## The basic handler

```ts
// app/api/webhooks/stripe/route.ts
import Stripe from 'stripe';
import { headers } from 'next/headers';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
const webhookSecret = process.env.STRIPE_WEBHOOK_SECRET!;

export async function POST(req: Request) {
  const body = await req.text();
  const signature = (await headers()).get('stripe-signature');

  if (!signature) {
    return new Response('No signature', { status: 400 });
  }

  let event: Stripe.Event;

  try {
    event = stripe.webhooks.constructEvent(body, signature, webhookSecret);
  } catch (err) {
    return new Response(`Webhook signature verification failed: ${err}`, { status: 400 });
  }

  // Route to handlers
  switch (event.type) {
    case 'checkout.session.completed':
      await handleCheckoutCompleted(event.data.object as Stripe.Checkout.Session);
      break;
    case 'customer.subscription.deleted':
      await handleSubscriptionCancelled(event.data.object as Stripe.Subscription);
      break;
    case 'invoice.payment_failed':
      await handlePaymentFailed(event.data.object as Stripe.Invoice);
      break;
    default:
      // Unhandled event type — acknowledge receipt anyway
      break;
  }

  return new Response(null, { status: 200 });
}
```

Always use `req.text()` not `req.json()` for the raw body. Stripe's signature verification runs on the exact bytes received — parsing to JSON first mutates the whitespace and breaks the hash comparison.

## Idempotency

Webhook delivery is at-least-once. The same event can arrive multiple times (retries after a 5xx, network issues, race conditions). Your handler must be idempotent — processing an event twice should have the same outcome as processing it once.

```ts
// db/schema.ts
export const processedWebhooks = pgTable('processed_webhooks', {
  id: text('id').primaryKey(),  // event ID from the service
  processedAt: timestamp('processed_at', { withTimezone: true }).notNull().defaultNow(),
});
```

```ts
async function handleCheckoutCompleted(session: Stripe.Checkout.Session) {
  // Check if already processed
  const existing = await db.query.processedWebhooks.findFirst({
    where: (w, { eq }) => eq(w.id, session.id),
  });

  if (existing) return;  // Already handled

  // Process the event
  await db.transaction(async (tx) => {
    await tx.update(users)
      .set({ subscriptionStatus: 'active', stripeCustomerId: session.customer as string })
      .where(eq(users.email, session.customer_email!));

    // Mark as processed — atomic with the update
    await tx.insert(processedWebhooks).values({ id: session.id });
  });

  await sendWelcomeEmail(session.customer_email!);
}
```

The deduplication check and the actual update are in a single database transaction. If the transaction fails, the webhook isn't marked as processed — so the retry will re-run it safely.

## Resend webhook handling

```ts
// app/api/webhooks/resend/route.ts
import { Webhook } from 'svix';

const webhookSecret = process.env.RESEND_WEBHOOK_SECRET!;

export async function POST(req: Request) {
  const body = await req.text();
  const headerPayload = await headers();

  const svix = new Webhook(webhookSecret);

  let event: { type: string; data: Record<string, unknown> };

  try {
    event = svix.verify(body, {
      'svix-id': headerPayload.get('svix-id') ?? '',
      'svix-timestamp': headerPayload.get('svix-timestamp') ?? '',
      'svix-signature': headerPayload.get('svix-signature') ?? '',
    }) as typeof event;
  } catch {
    return new Response('Verification failed', { status: 400 });
  }

  switch (event.type) {
    case 'email.delivered':
      await db.update(emailLogs)
        .set({ status: 'delivered', deliveredAt: new Date() })
        .where(eq(emailLogs.resendId, event.data.email_id as string));
      break;
    case 'email.bounced':
      await handleBounce(event.data.email_id as string, event.data.to as string);
      break;
  }

  return new Response(null, { status: 200 });
}
```

## GitHub webhook handling

```ts
// app/api/webhooks/github/route.ts
import { createHmac, timingSafeEqual } from 'crypto';

const secret = process.env.GITHUB_WEBHOOK_SECRET!;

function verifyGitHubSignature(payload: string, signature: string | null): boolean {
  if (!signature?.startsWith('sha256=')) return false;

  const hmac = createHmac('sha256', secret);
  hmac.update(payload);
  const digest = `sha256=${hmac.digest('hex')}`;

  return timingSafeEqual(Buffer.from(digest), Buffer.from(signature));
}

export async function POST(req: Request) {
  const body = await req.text();
  const event = (await headers()).get('x-github-event');
  const signature = (await headers()).get('x-hub-signature-256');

  if (!verifyGitHubSignature(body, signature)) {
    return new Response('Unauthorized', { status: 401 });
  }

  const payload = JSON.parse(body);

  if (event === 'push' && payload.ref === 'refs/heads/master') {
    await triggerDeployment(payload.repository.name, payload.after);
  }

  return new Response(null, { status: 200 });
}
```

Use `timingSafeEqual` for signature comparison — regular string equality is vulnerable to timing attacks.

## Responding quickly

Webhook timeouts are strict. Stripe waits 30 seconds, then retries. If your handler does heavy work (sending emails, triggering scans, calling external APIs), acknowledge the webhook immediately and queue the work:

```ts
export async function POST(req: Request) {
  // ... verify signature

  // Enqueue the job — returns in milliseconds
  await queue.add('process-checkout', { sessionId: event.data.object.id });

  // Acknowledge immediately
  return new Response(null, { status: 200 });
}
```

The queue worker does the slow work. The webhook endpoint is just a receiver. This pattern also gives you retry logic if the slow work fails — without relying on Stripe's webhook retries.

## Local testing

Use the Stripe CLI to forward events to your local server:

```bash
stripe listen --forward-to localhost:3000/api/webhooks/stripe
```

This prints a webhook secret to use in your `.env.local`. Events triggered in the Stripe dashboard or via the CLI are forwarded to your local endpoint with valid signatures.
