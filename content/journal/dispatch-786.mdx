date: "2031-05-29"
excerpt: "Playwright is Microsoft's browser automation library for end-to-end testing. It runs tests across Chromium, Firefox, and WebKit with a single API, auto-waits for elements before interacting with them (eliminating most flakiness), and has built-in network interception, screenshot diffing, and trace recording. For applications where button clicks, form submissions, and navigation flows need to be tested against real browser rendering, Playwright is the current best option."

Auto-wait eliminates flaky tests. Record traces, not guesses.

## Install and configure

```bash
bun add -D @playwright/test
bunx playwright install --with-deps chromium firefox webkit
```

```ts
// playwright.config.ts
import { defineConfig, devices } from '@playwright/test';

export default defineConfig({
  testDir:        './tests/e2e',
  fullyParallel:  true,
  forbidOnly:     !!process.env.CI,
  retries:        process.env.CI ? 2 : 0,
  workers:        process.env.CI ? 1 : undefined,
  reporter:       'html',
  use: {
    baseURL:        'http://localhost:3000',
    trace:          'on-first-retry',
    screenshot:     'only-on-failure',
    video:          'retain-on-failure',
  },
  projects: [
    { name: 'chromium', use: { ...devices['Desktop Chrome'] } },
    { name: 'firefox',  use: { ...devices['Desktop Firefox'] } },
    { name: 'webkit',   use: { ...devices['Desktop Safari'] } },
    { name: 'mobile',   use: { ...devices['iPhone 15'] } },
  ],
  webServer: {
    command:            'bun run dev',
    url:                'http://localhost:3000',
    reuseExistingServer: !process.env.CI,
  },
});
```

## Writing tests

```ts
// tests/e2e/auth.spec.ts
import { test, expect } from '@playwright/test';

test.describe('authentication', () => {
  test('user can sign up and log in', async ({ page }) => {
    await page.goto('/signup');

    await page.fill('[name=email]', 'test@construxgroup.io');
    await page.fill('[name=password]', 'secure-password-123');
    await page.click('button[type=submit]');

    // Playwright auto-waits — no sleep, no arbitrary timeout
    await expect(page).toHaveURL('/dashboard');
    await expect(page.locator('h1')).toHaveText('Welcome back');
  });

  test('invalid credentials show error', async ({ page }) => {
    await page.goto('/login');

    await page.fill('[name=email]', 'wrong@example.com');
    await page.fill('[name=password]', 'wrong-password');
    await page.click('button[type=submit]');

    await expect(page.locator('[role=alert]')).toContainText('Invalid credentials');
    await expect(page).toHaveURL('/login');  // Still on login page
  });
});
```

## Network interception

```ts
test('checkout with Stripe mock', async ({ page }) => {
  // Intercept the Stripe API call
  await page.route('**/api/payments/create-intent', async (route) => {
    await route.fulfill({
      status: 200,
      body:   JSON.stringify({ clientSecret: 'pi_test_mock_secret' }),

  await page.fill('[name=card-number]', '4242 4242 4242 4242');

  await expect(page.locator('[data-testid=success]')).toBeVisible();
```

## Visual regression — screenshot comparison

```ts
test('landing page matches snapshot', async ({ page }) => {
  await page.goto('/');
  await page.waitForLoadState('networkidle');

  // First run: creates snapshot. Subsequent runs: diffs against it.
  await expect(page).toHaveScreenshot('landing.png', {
    maxDiffPixelRatio: 0.02,  // Allow 2% pixel difference
    animations:        'disabled',
  });
});

// Component-level snapshot
test('button variants', async ({ page }) => {
  await page.goto('/design-system/buttons');
  const buttons = page.locator('[data-testid=button-grid]');
  await expect(buttons).toHaveScreenshot('buttons.png');
});
```

## Fixtures — shared test setup

```ts
// fixtures.ts
import { test as base } from '@playwright/test';

type Fixtures = {
  loggedInPage: Page;
};

export const test = base.extend<Fixtures>({
  loggedInPage: async ({ page }, use) => {
    await page.goto('/login');
    await page.fill('[name=email]', process.env.TEST_USER_EMAIL!);
    await page.fill('[name=password]', process.env.TEST_USER_PASSWORD!);
    await page.click('button[type=submit]');
    await page.waitForURL('/dashboard');
    await use(page);  // Test runs with this authenticated page
  },
});

// tests/dashboard.spec.ts
import { test } from './fixtures';

test('user sees their projects', async ({ loggedInPage }) => {
  await loggedInPage.goto('/dashboard/projects');
  await expect(loggedInPage.locator('[data-testid=project-list]')).toBeVisible();
});
```

## CI — GitHub Actions

```yaml
- name: Run Playwright tests
  run: bunx playwright test

- uses: actions/upload-artifact@v4
  if: always()
  with:
    name: playwright-report
    path: playwright-report/
    retention-days: 30
```

Playwright vs Cypress: Playwright runs across all browsers natively (Cypress only added Firefox/WebKit recently), is faster in CI with full parallelism, and has better network mocking. Cypress has a better DX for visual debugging locally. The Construx default: Playwright for all E2E tests.
